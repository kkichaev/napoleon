Функция ПодключениеКСерверу()
	сервер = СоздатьОбъект("GRSoft.Server");
	рзд = Найти(АдресСервера, ":");
	адрес = Лев(АдресСервера, рзд-1);
	порт = Число(Прав(АдресСервера, СтрДлина(АдресСервера) - рзд));
	рез = сервер.Connect(адрес, порт);
	
	Если рез = 0 Тогда
		Предупреждение(сервер.ErrorMessage);
		Возврат "";
	КонецЕсли;
	
	Возврат сервер;
КонецФункции
              
Функция ДобавитьИнвентаризацию(док)
	докЗапись = СоздатьОбъект("Документ.Реализация");
	товар = СоздатьОбъект("Справочник.Номенклатура");
	
	докЗапись.Новый();
	
	докЗапись.ДатаДок = ТекущаяДата();
	докЗапись.Автор = глПользователь;
	
	докЗапись.Комментарий = "Инвентаризация с ТСД";
	
	
	Для сч = 0 По док.items.Count-1 Цикл
		item = док.items.Get(сч);
		товар.НайтиПоКоду(СокрЛП(item.id));
		
		докЗапись.НоваяСтрока();
		
		докЗапись.Номенклатура = товар.ТекущийЭлемент();
		докЗапись.Количество = item.inputQty;
		докЗапись.КоличествоСТСД_Сиброн  = item.inputQty;

		Если item.isBaseUnit = 1 Тогда
			докЗапись.Единица = товар.БазоваяЕдиница;
			докЗаписьКоэффициент = 1;
		Иначе
			докЗапись.Единица = товар.ОсновнаяЕдиница;
			докЗаписьКоэффициент = товар.ОсновнаяЕдиница.Коэффициент;
		КонецЕсли;
	КонецЦикла;
	
	докЗапись.Записать();
	
	Сообщить("Обработан " + Строка(докЗапись));
	Возврат 0;
КонецФункции

Функция ДобавитьДокументСТСД(док)
	рет = 0;
	
	докОбъект = "";
	типДок = док.type;
	Если типДок = "Invent" Тогда
		рет = ДобавитьИнвентаризацию(док);
		возврат рет;
	ИначеЕсли типДок = "Delivery" Тогда
		докОбъект = "Документ.Реализация";
	ИначеЕсли типДок = "Income" Тогда
		докОбъект = "Документ.ПоступлениеТМЦ";
	КонецЕсли;
	
	Если докОбъект <> "" Тогда
		докЗапись = СоздатьОбъект(докОбъект);
		Если докЗапись.НайтиПоНомеру(док.number, док.date) = 1 Тогда
			докЗапись.ВыбратьСтроки();
			статусТСД = Перечисление.СтатусВТСД_Сиброн.ОбработанВТСД_сошл;
			Пока докЗапись.ПолучитьСтроку() = 1 Цикл
				
				Для сч = 0 По док.items.Count-1 Цикл
					item = док.items.Get(сч);
					Если докЗапись.Номенклатура.Код = item.id Тогда
						докЗапись.КоличествоСТСД_Сиброн = item.inputQty;
						Прервать;
					КонецЕсли;
				КонецЦикла;
				
				Если докЗапись.КоличествоСТСД_Сиброн <> докЗапись.Количество Тогда
					статусТСД = Перечисление.СтатусВТСД_Сиброн.ОбработанВТСД_расх;
				КонецЕсли;
			КонецЦикла;
			докЗапись.СтатусВТСД = статусТСД;
			докЗапись.Записать();
			рет = 1;
			Сообщить("Обработан " + Строка(докЗапись));
		Иначе
			Сообщить("Не найден " + докОбъект + " № " + док.number + " от " + док.date);
		КонецЕсли;
	КонецЕсли;
	
	Возврат рет;
КонецФункции

Функция ToDate(стрДата)
	Возврат Сред(стрДата,7,2) + "/" + Сред(стрДата,5,2) + "/" + Лев(стрДата, 4) + " " + Сред(стрДата,9,2) + ":" + Сред(стрДата,11,2) + ":" + Сред(стрДата,13,2);
КонецФункции

Процедура ПриемДокументов()
	сервер = ПодключениеКСерверу();

	Если сервер = "" Тогда
		Возврат ;
	КонецЕсли;	
	
	доки = сервер.Get("NewDocuments", "");
	Если доки = 0 Тогда
		Возврат;
	КонецЕсли;
	
	handled = сервер.New("HandledDocs");
	Для сч = 0 По доки.Count -1 Цикл
		док = доки.Get(сч);
		where = "userid='" + док.userid + "' and created=ToDate('" + ToDate(док.DateToString("created")) + "')";
		
		новДокс = сервер.Get("WhOutDoc", where);
		Если (новДокс <> 0) и (новДокс.Count = 1) Тогда
			текДок = новДокс.Get(0);
			Если ДобавитьДокументСТСД(текДок) = 1 Тогда
				hd = handled.New();
				cr = текДок.DateToString("created");
				hd.DateFromString("created", cr);;
				hd.userid = текДок.userid;
			КонецЕсли;
		КонецЕсли;
	КонецЦикла;
	
	Если handled.Count > 0 Тогда
		handled.WriteDirect();
	КонецЕсли;
	
КонецПроцедуры

Процедура ВыгрузитьПользователей(сервер)
	users = сервер.Get("Agents", "");
	агенты = СоздатьОбъект("Справочник.ПользователиТСД_Сиброн");
	агенты.ВыбратьЭлементы();
	Пока агенты.ПолучитьЭлемент() = 1 Цикл
		Если агенты.ПометкаУдаления() = 1 Тогда
			Продолжить;
		КонецЕсли;
		
		Нашли = 0;
		Для сч = 0 По users.Count -1 Цикл
			user = users.Get(сч);
			Если user.id = агенты.Код Тогда
				user.name = агенты.Наименование;
				Нашли = 1;
				Прервать;
			КонецЕсли;
		КонецЦикла;
		Если Нашли = 0 Тогда
			user = users.New();
			user.name = агенты.Наименование;
			user.id = агенты.Код;
		КонецЕсли;
	КонецЦикла;
	
	Если users.Count > 0 Тогда
		users.WriteDirect();
	КонецЕсли;
КонецПроцедуры

Процедура ВыгрузитьДокументы(сервер)
	
	docs = сервер.New("WHDocs");
	
	Документы.ВыбратьСтроки();
	Пока Документы.ПолучитьСтроку() = 1 Цикл
		док = Документы.Документ;
		дест = "";
		докЗапись = "";
		дест = docs.New();

		Если док.Вид() = "Реализация" Тогда
			дест.type = "Delivery";
			докЗапись = СоздатьОбъект("Документ.Реализация");
		ИначеЕсли док.Вид() = "ПоступлениеТМЦ" Тогда
			дест.type = "Income";
			докЗапись = СоздатьОбъект("Документ.ПоступлениеТМЦ");
		КонецЕсли;
		
		дест.id = док.Контрагент.Код;
		дест.number = док.НомерДок;
		дест.date = док.ДатаДок;
		дест.name = док.Контрагент.Наименование;
		
		док.ВыбратьСтроки();
		Пока док.ПолучитьСтроку() = 1 Цикл
			стр = дест.items.New();
			стр.id =  док.Номенклатура.Код;
			стр.qty = док.Количество;
			стр.packCoef = док.Коэффициент;
			стр.isBaseUnit = ?(док.Единица = док.Номенклатура.БазоваяЕдиница, 1, 0);
		КонецЦикла;		

		Если ПустаяСтрока(док.СтатусВТСД) = 1 Тогда
			докЗапись.НайтиДокумент(док); 
			докЗапись.СтатусВТСД = Перечисление.СтатусВТСД_Сиброн.ВыгруженВТСД;
			докЗапись.Записать();
		КонецЕсли;
	КонецЦикла;
	
	Если docs.Count > 0 Тогда
		docs.WriteDirect();
	КонецЕсли;
КонецПроцедуры

Процедура ВыгрузитьТовары(сервер)
	дест = сервер.New("Price");
	
	товар = СоздатьОбъект("Справочник.Номенклатура");
	товар.ВыбратьЭлементы();
	Пока товар.ПолучитьЭлемент() = 1 Цикл
		Если (товар.ПометкаУдаления() = 1) или (товар.ЭтоГруппа() = 1) Тогда
			Продолжить;
		КонецЕсли;
		
		стр = дест.New();
		стр.id = товар.Код;
		стр.name = товар.Наименование;
		стр.qtyInPack = товар.ОсновнаяЕдиница.Коэффициент;
		стр.packName = товар.ОсновнаяЕдиница.ОКЕИ.Наименование;
		стр.barcode = СокрЛП(товар.БазоваяЕдиница.ШтрихКод);
		стр.boxBarcode = СокрЛП(товар.ОсновнаяЕдиница.ШтрихКод);
	КонецЦикла;
	
	Если дест.Count > 0 Тогда
		дест.WriteDirect();
	КонецЕсли;
