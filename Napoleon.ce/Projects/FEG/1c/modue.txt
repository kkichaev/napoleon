Функция ПолучитьТаблицуКоманд()

   // Создадим пустую таблицу команд и колонки в ней
   Команды = Новый ТаблицаЗначений;

   // Как будет выглядеть описание печатной формы для пользователя
   Команды.Колонки.Добавить("Представление", Новый ОписаниеТипов("Строка")); 

   // Имя нашего макета, что бы могли отличить вызванную команду в обработке печати
   Команды.Колонки.Добавить("Идентификатор", Новый ОписаниеТипов("Строка"));

   // Тут задается, как должна вызваться команда обработки
   // Возможные варианты:
   // - ОткрытиеФормы - в этом случае в колонке идентификатор должно быть указано имя формы, которое должна будет открыть система
   // - ВызовКлиентскогоМетода - вызвать клиентскую экспортную процедуру из модуля формы обработки
   // - ВызовСерверногоМетода - вызвать серверную экспортную процедуру из модуля объекта обработки
   Команды.Колонки.Добавить("Использование", Новый ОписаниеТипов("Строка"));

   // Следующий параметр указывает, необходимо ли показывать оповещение при начале и завершению работы обработки. Не имеет смысла при открытии формы
   Команды.Колонки.Добавить("ПоказыватьОповещение", Новый ОписаниеТипов("Булево"));

   // Для печатной формы должен содержать строку ПечатьMXL 
   Команды.Колонки.Добавить("Модификатор", Новый ОписаниеТипов("Строка"));
   Возврат Команды;

Процедура ДобавитьСкидку(ДБФ, Скидка)
	нкд = Скидка.ХранилищеНастроекКомпоновкиДанных.Получить();
	Если ТипЗнч(нкд) = Тип("Структура") и нкд.Свойство("ОтборПоНоменклатуре") <> Неопределено Тогда
		Для каждого эл из нкд.ОтборПоНоменклатуре.Отбор.Элементы Цикл
			Если Не эл.Использование Тогда
				Продолжить;
			КонецЕсли;
			
			Если Строка(эл.ЛевоеЗначение) = "Номенклатура" и ТипЗнч(эл.ПравоеЗначение) = Тип("СписокЗначений") Тогда
				Если эл.ВидСравнения = ВидСравненияКомпоновкиДанных.ВСпискеПоИерархии Тогда
					ЗаписьСкидкиНоменклатуры(ДБФ, Скидка, 0, эл.ПравоеЗначение);
				ИначеЕсли эл.ВидСравнения = ВидСравненияКомпоновкиДанных.НеВСпискеПоИерархии Тогда
					ЗаписьСкидкиНоменклатуры(ДБФ, Скидка, 1, эл.ПравоеЗначение);
				КонецЕсли;
				
				Прервать;
			КонецЕсли;
		КонецЦикла;	
	КонецЕсли;

Функция ЗагрузкаЗаявокАгента(АдресЗаказ, КодАгента, ПринятыеЗаказы)
	СчДок = 0;
	
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	Заказ_имя = КаталогВременныхФайлов() + "TORD.dbf";
	ЗаказДД = ПолучитьИзВременногоХранилища(АдресЗаказ);
	ЗаказДД.Записать(Заказ_имя);
	Заказ = Новый XBase(Заказ_имя);
	
	Заказ.Кодировка = КодировкаXBase.OEM;
	
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				СообщитьИнт("Повторный заказ " + значЗаказ);
			Иначе			
				стр = Найти(Заказ.ID, Символ(9));
				код = Заказ.ID;
				кодСоглашения = "";
				Если стр <> 0 Тогда
					кодСоглашения =	Прав(Заказ.ID, СтрДлина(Заказ.ID) - стр);
					код = Лев(Заказ.ID, стр-1);
				КонецЕсли;
				Документ = СоздатьЗаказ(Заказ, код, кодСоглашения);
				
				Если МожноМенятьЦену Тогда
					ВидЦеныДокумента = ПолучитьЦену(Заказ.CTYPE);
				Иначе
					ВидЦеныДокумента = Неопределено;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ, ВидЦеныДокумента);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		ЗаписатьДокумент(Документ);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

Процедура ЗагрузитьЗаявки() Экспорт 
	СчДок = 0;
	СообщитьИнт("Поиск файлов " + ПапкаОбмена + "\ORD*.DBF");
	
	Рез = НайтиФайлы(ПапкаОбмена, "ORD*.DBF");

		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");
		КодАгента = Сред(имяФайла, 4, разделитель-4);
		
		временныйФайл = ПапкаОбмена + "\TORD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);

		СчДок = СчДок + ЗагрузкаЗаявокАгента(АдресЗаказ, КодАгента, ПринятыеЗаказы);
		
		СохранитьОбработанныеДокументы(КодАгента, ПринятыеЗаказы, "Order");
		СохранитьПринятыеДокументы(КодАгента, ПринятыеЗаказы);

Процедура ЗагрузитьПКО() Экспорт

		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");

		временныйФайл = КаталогВременныхФайлов() + "TORD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		ЗаписатьДок = 0;
		ПринятыеДок = ЗагрузитьПринятыеДокументы(КодАгента, Истина, "Incass");
		
		Док = Новый XBase(временныйФайл);
		Док.Кодировка = КодировкаXBase.OEM;
		
		КодДок = "";
		ТекСум = 0;
		КодОрг = "";
		ТекДок = Неопределено;
		
		Док.Первая();		
		Пока Док.ВКонце() = 0 Цикл			
			//
			// Проверим признак начала заявки
			//
			Если Док.ORDFLAG <> 0 или ПустаяСтрока(Док.NUM_I) Тогда
				//
				// Запишем предыдущий документ
				//
				Если ТекДок <> Неопределено Тогда
					Если ТекСум > 0 Тогда
						ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
					КонецЕсли;
					Если ЗаписатьДокументСервер(ТекДок) Тогда
						СообщитьОДокументе(ТекДок);
						ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
						ЗаписатьДокументСервер(ТекДок);
					КонецЕсли;
				КонецЕсли;
				
				ТекДок = Неопределено;
				ТекСум = Док.SUM;
				КодДок = СокрЛП(Док.CREATED);
				значДок = ПринятыеДок.НайтиПоЗначению(КодДок);
				Если значДок <> Неопределено Тогда			
					Сообщить("Повторный документ " + значДок);
					ТекСум = 0;
				Иначе
					ТекДок = СоздатьПКО(Док);
				КонецЕсли;
			КонецЕсли;
			
			Если ТекДок <> Неопределено Тогда 
				КодОрг = Док.ID;
				Сумма = Док.SUM_I;
				ДокОсн = Документы.РеализацияТоваровУслуг.ПустаяСсылка();
				Если ПустаяСтрока(Док.NUM_I) Тогда
					Сумма = Док.SUM;
				Иначе
					ДокОсн = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Док.NUM_I, Док.DATE_I);
				КонецЕсли;
				
				ДобавитьДокументОснование(ТекДок, ДокОсн, Сумма);
				ТекСум = ТекСум - Сумма;
			КонецЕсли;	
			Док.Следующая();
		КонецЦикла;
		
		Если ТекДок <> Неопределено Тогда
			Если ТекСум > 0 Тогда
				ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
			КонецЕсли;
			Если ЗаписатьДокументСервер(ТекДок) Тогда
				СообщитьОДокументе(ТекДок);
				ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
			КонецЕсли;
		КонецЕсли;
		
		Док.ЗакрытьФайл();
		
			
		СохранитьОбработанныеДокументы(КодАгента, ПринятыеДок, "Incass");
		СохранитьПринятыеДокументы(КодАгента, ПринятыеДок, "Incass");
	КонецЦикла;
