Функция ДобавитьРасшифровку(докСП, ДокИст, сумма, накладная)
	организация = накладная.Организация;
	Если организация.Пустая() Тогда
		организации = ПолучитьНастройкуАгента(ДокИст.userid, "Организация");
		Если организации <> Неопределено и организации.Количество() > 0 Тогда
			организация = организации.Получить(0);

		Документ.Договор = накладная.Договор;
	КонецЕсли;

		Документ.Контрагент = накладная.Контрагент;
	КонецЕсли;

				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	
	рп = Документ.РасшифровкаПлатежа.Добавить();
	рп.Партнер = Документ.партнер;
	рп.Сумма = сумма;

	рп.Организация = Документ.Организация;
	рп.Подразделение = Документ.Подразделение;
	рп.ВалютаВзаиморасчетов = Документ.Валюта;
	рп.Заказ = Документ.Договор;
	
	рп.Заказ = накладная.Договор;
	рп.ОснованиеПлатежа = накладная;
		
	Возврат рп;

			докИст = докс.Get(сч);

			фильтр = "userid='"+КодАгента+"' and created=ToDate('" + датаСтр + "')";
			
			док1с = сервер.Get("OrderCommitted", фильтр);
			Если док1с <> Неопределено и док1с.Count > 0 Тогда
				заказИст = док1с.Get(0);
				заказ = Документы.ЗаказКлиента.НайтиПоНомеру(заказИст.number, Дата(заказИст.dateToString("date")));
				
				Если не заказ.Пустая() Тогда

					Если файлСчет <> Неопределено Тогда
						фс = фси.New();
						фс.id = докИст.fileId;
						фс.type = "application/pdf";
						фс.data.Read(файлСчет);
					Иначе	
						ретСтр = ретСтр + "FailCrBill;";
					КонецЕсли;
				Иначе 
					ретСтр = ретСтр + "NoOrd;";
				КонецЕсли;
			КонецЕсли;
		КонецЦикла;
		
		Если фси.Count > 0 Тогда
			фси.WriteDirect();
			ретСтр = ретСтр + "Done;";
