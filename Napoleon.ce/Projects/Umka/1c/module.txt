Процедура ВыполнитьВыгрузкуОстатков(НастройкиОтборовСКД = Неопределено, Каталог = "", ВариантНастройкиИзСпискаНастроек = 0,
		  ВыбранаНастройкаИзСпискаНастроек = Ложь) Экспорт
	
	// ++ 03.12.2020 НЭТИ Васильченко О.А. Если выбран вариант список настроек, то обрабатываем отборы СКД

		ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Не выбран каталог");

	КонецЕсли;			
	
	УстановленныйВариантНастройкиИзСпискаНастроек = ВариантНастройкиИзСпискаНастроек;
	Если УстановленныйВариантНастройкиИзСпискаНастроек Тогда
		
		Если ТипЗнч(НастройкиОтборовСКД) = Тип("Массив") Тогда
			СчетчикТекущейНастройки = 1;
			ВсегоНастроекВГрупповойВыгрузке = НастройкиОтборовСКД.Количество();
			Для каждого ЭлементНастройкиМассива Из НастройкиОтборовСКД Цикл

				СчетчикТекущейНастройки = СчетчикТекущейНастройки + 1;
			КонецЦикла;	
		Иначе	
			ВыполнитьВыгрузкуОстатковПоНастройке(НастройкиОтборовСКД, ВыбранаНастройкаИзСпискаНастроек);
			ВыполнитьВыгрузкуВФайлы();
		КонецЕсли;
						
	Иначе
		
		УстановленныйКаталогФайлов = ЭтотОбъект.Каталог;
		Если Не ЗначениеЗаполнено(Склады) Тогда
			ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Не выбран склад");
			Возврат;
		КонецЕсли;
					
		Если Не ЗначениеЗаполнено(ВидЦеныПоУмолчанию) Тогда
			ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Не выбран вид цены по умолчанию");
			Возврат;
		КонецЕсли;
			
		ВыбранныеФирмы = Фирмы.НайтиСтроки(Новый Структура("Выбрано", Истина));
		Если ВыбранныеФирмы.Количество() = 0 Тогда
			ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Не выбрана организация");
			Возврат;
		КонецЕсли;
			
		ВыгрузитьНастройки();
	    ВыполнитьВыгрузкуВФайлы();

		Возврат Ложь;
	КонецЕсли;

		Если ЭлементОтбора.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("Организация") Тогда

		ИначеЕсли ЭлементОтбора.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("Склад") Тогда
			Если НЕ ЭлементОтбора.Использование Тогда
				ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Не выбраны склады");
				Возврат Ложь;
			КонецЕсли;
		ИначеЕсли ЭлементОтбора.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("Агент") Тогда
			Если НЕ ЭлементОтбора.Использование Тогда
				ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Не выбраны агенты");
				Возврат Ложь;
			КонецЕсли;
		КонецЕсли;
	КонецЦикла;	
	
	Возврат Истина;

Процедура ВыгрузитьНастройкиОтборовСКД(СтруктураНастроекОтборовСКД)
	
	НастройкиОтборовСКД = СтруктураНастроекОтборовСКД.НастройкиОтборовСКД;
	МожноИзменятьЦенуИзОтборовСКД = СтруктураНастроекОтборовСКД.МожноИзменятьЦену;
	ВидЦеныИзОтборовСКД = СтруктураНастроекОтборовСКД.ВидЦеныПоУмолчанию;
	
	ОбщегоНазначенияКлиентСервер.СообщитьПользователю("Выгрузка настроек...");
	
	Настройки = Новый ТекстовыйДокумент;
	
	Настройки.ДобавитьСтроку("ВидЦены = Закупочная");
	
	СтрокаФайла = "МожноИзменятьЦену=" + ?(МожноИзменятьЦенуИзОтборовСКД, "1", "0");
	Настройки.ДобавитьСтроку(СтрокаФайла);
	
	Для Каждого ЭлементОтбора из НастройкиОтборовСКД.Отбор.Элементы Цикл	
		ЗаполнитьСписокОтборомСКД(ЭлементОтбора);
	КонецЦикла;
	
	СтрокаФайла = "Организация = " + СписокЗначенийВСтроку(СписокОрганизацийИзОтборовСКД);
	Настройки.ДобавитьСтроку(СтрокаФайла);
		
	ИмяФайла = УстановленныйКаталогФайлов + "\order.cfg";
	Настройки.Записать(ИмяФайла, КодировкаТекста.ANSI);

Процедура ЗаполнитьСписокОтборомСКД(ЭлементОтбораСКД)
	
	ИскомыйСписок = Новый СписокЗначений;
	
	Если ЭлементОтбораСКД.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("Организация") Тогда
		ТипЭлементаОтбора = "Справочник.Организации";	
	ИначеЕсли ЭлементОтбораСКД.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("Склад") Тогда
		ТипЭлементаОтбора = "Справочник.Склады";		
	ИначеЕсли ЭлементОтбораСКД.ЛевоеЗначение = Новый ПолеКомпоновкиДанных("Агент") Тогда
		ТипЭлементаОтбора = "Справочник.Пользователи";		
	КонецЕсли;
	
	Если ЭлементОтбораСКД.ВидСравнения = ВидСравненияКомпоновкиДанных.Равно Тогда
		ИскомыйСписок.Добавить(ЭлементОтбораСКД.ПравоеЗначение);	
	ИначеЕсли ЭлементОтбораСКД.ВидСравнения = ВидСравненияКомпоновкиДанных.НеРавно Тогда
		Запрос = Новый Запрос;
		Запрос.Текст = "ВЫБРАТЬ
		               |	Справочник.Ссылка КАК Ссылка
		               |ИЗ
		               |	" + ТипЭлементаОтбора + " КАК Справочник
		               |ГДЕ
		               |	Справочник.Ссылка <> &ИсключаемыйЭлемент" + ?(ТипЭлементаОтбора = "Справочник.Склады", " И НЕ Справочник.ЭтоГруппа", "");
		Запрос.УстановитьПараметр("ИсключаемыйЭлемент", ЭлементОтбораСКД.ПравоеЗначение);
		Выборка = Запрос.Выполнить().Выбрать();
		Пока Выборка.Следующий() Цикл
			ИскомыйСписок.Добавить(Выборка.Ссылка);
		КонецЦикла;
	ИначеЕсли ЭлементОтбораСКД.ВидСравнения = ВидСравненияКомпоновкиДанных.ВСписке Тогда
		ИскомыйСписок = ЭлементОтбораСКД.ПравоеЗначение;
	ИначеЕсли ЭлементОтбораСКД.ВидСравнения = ВидСравненияКомпоновкиДанных.НеВСписке Тогда
		Запрос = Новый Запрос;
		Запрос.Текст = "ВЫБРАТЬ
		               |	Справочник.Ссылка КАК Ссылка
		               |ИЗ

Процедура ЗаполнитьПартнераИСоглашение(Документ, КодПартнера, КодСоглашения, СоглашениеПоУмолчанию)
	Партнер = Справочники.Партнеры.НайтиПоКоду(КодПартнера);
	Если Партнер <> Неопределено Тогда
		Документ.Партнер = Партнер;
	КонецЕсли;
	
	Если КодСоглашения <> "" Тогда
		ид = Новый УникальныйИдентификатор(КодСоглашения);
		Документ.Соглашение = Справочники.СоглашенияСКлиентами.ПолучитьСсылку(ид);	
	ИначеЕсли ЗначениеЗаполнено(СоглашениеПоУмолчанию) Тогда
		Документ.Соглашение = СоглашениеПоУмолчанию;
	КонецЕсли;
	
	Если Документ.Соглашение.Пустая() Тогда
		//Документ.НалогообложениеНДС =  Перечисления.ТипыНалогообложенияНДС.ПродажаОблагаетсяНДС;
		Документ.ЦенаВключаетНДС = Истина;

		//Документ.НалогообложениеНДС =  Документ.Соглашение.НалогообложениеНДС;
		Документ.ЦенаВключаетНДС = Документ.Соглашение.ЦенаВключаетНДС;
		УсловияПродаж = ПродажиСервер.ПолучитьУсловияПродаж(Документ.Соглашение, Истина, Истина);
		Если ЗначениеЗаполнено(УсловияПродаж) И УсловияПродаж.Свойство("Типовое") И УсловияПродаж.Типовое <> Неопределено Тогда
			Документ.ЗаполнитьУсловияПродажПоСоглашению();
		КонецЕсли;
	КонецЕсли;

		ПодходящаяСтрокаЦеновойГруппыСоглашения = Документ.Соглашение.ЦеновыеГруппы.Найти(Товар.Номенклатура.ЦеноваяГруппа, "ЦеноваяГруппа");
		Если ПодходящаяСтрокаЦеновойГруппыСоглашения <> Неопределено Тогда
			Товар.ВидЦены = ПодходящаяСтрокаЦеновойГруппыСоглашения.ВидЦен;	
		КонецЕсли;
	КонецЕсли;
	// -- 03.12.2020 НЭТИ Васильченко О.А.
	
	Данные = Новый Структура;	
	мета = Документ.Метаданные();
	Для Каждого рек из мета.ТабличныеЧасти.Товары.Реквизиты Цикл
		Данные.Вставить(рек.Имя);
	КонецЦикла;
	ЗаполнитьЗначенияСвойств(Данные, Товар);
	Данные.Вставить("ТипНоменклатуры");
	Данные.Вставить("ХарактеристикиИспользуются");
	
	СтруктураДействий = Новый Структура;
	СтруктураДействий.Вставить("ЗаполнитьПризнакТипНоменклатуры", Новый Структура("Номенклатура", "ТипНоменклатуры"));
	СтруктураДействий.Вставить("ЗаполнитьПризнакХарактеристикиИспользуются", Новый Структура("Номенклатура", "ХарактеристикиИспользуются"));
	
	СтруктураДействий.Вставить("ПроверитьХарактеристикуПоВладельцу", Товар.Характеристика);
	СтруктураДействий.Вставить("ПроверитьЗаполнитьУпаковкуПоВладельцу", Товар.Упаковка);
	СтруктураДействий.Вставить("ПересчитатьКоличествоЕдиниц");
	СтруктураДействий.Вставить("ЗаполнитьТипНоменклатуры");
	СтруктураДействий.Вставить("ПроверитьЗаполнитьСклад", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияСкладаВСтрокеТЧ(Документ, Ложь));
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДС", 
		Новый Структура("НалогообложениеНДС, Дата", Документ.НалогообложениеНДС, Документ.Дата));
	
	Если ВидЦеныДокумента = Неопределено Тогда
		СтруктураПересчетаСуммы = ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруПересчетаСуммыНДСВСтрокеТЧ(Документ);
		
		СтруктураДействий.Вставить("ЗаполнитьУсловияПродаж", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияУсловийПродажВСтрокеТЧ(Документ));
		СтруктураДействий.Вставить("ПересчитатьСуммуНДС", СтруктураПересчетаСуммы);
		СтруктураДействий.Вставить("ПересчитатьСуммуСНДС", СтруктураПересчетаСуммы);
		СтруктураДействий.Вставить("ПересчитатьСумму");
		СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомАвтоматическойСкидки", Новый Структура("Очищать", Истина));
		Если ТипЗнч(Документ.Ссылка) = Тип("ДокументСсылка.РеализацияТоваровУслуг") Тогда
			СтруктураДействий.Вставить("ОчиститьСуммуВзаиморасчетов");
		КонецЕсли;
		СтруктураДействий.Вставить("ЗаполнитьПризнакТипНоменклатуры", Новый Структура("Номенклатура", "ТипНоменклатуры"));
	КонецЕсли;
	
	ОбработкаТабличнойЧастиСервер.ОбработатьСтрокуТЧ(Данные, СтруктураДействий, Неопределено);
	ЗаполнитьЗначенияСвойств(Товар, Данные);
	Если ТипЗнч(Документ.Ссылка) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
		Товар.ВариантОбеспечения = Перечисления.ВариантыОбеспечения.Отгрузить;
	КонецЕсли;
