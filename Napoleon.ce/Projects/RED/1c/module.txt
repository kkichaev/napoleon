Функция СжатьДанные(данные)

	тд = Новый ТекстовыйДокумент();
	тд.ДобавитьСтроку(данные);

	читать.Пропустить(14);
	crc = читать.ПрочитатьЦелое32();
	размерСжатый = читать.ПрочитатьЦелое32();
	размер = читать.ПрочитатьЦелое32();
	
	// fileNameLength + extraLentgh +fileName+extra = 8 for data filename
	читать.Пропустить(8);
	тело = читать.Прочитать(размерСжатый).ПолучитьДвоичныеДанные();
	
	читать.Закрыть();
	поток.Закрыть();
	
	зппоток = Новый ПотокВПамяти();
	запись = Новый ЗаписьДанных(зппоток);
	
	// GZip Header for data fileName
	запись.Записать(Base64Значение("H4sICAAAAAAEAGRhdGEA"));
	запись.Записать(тело);
	запись.ЗаписатьЦелое32(crc);
	запись.ЗаписатьЦелое32(размер);
	запись.Закрыть();

	Возврат зппоток.ЗакрытьИПолучитьДвоичныеДанные();

	Товар.Номенклатура = Номенклатура;

		Товар.СтавкаНДС = УправлениеНебольшойФирмойПовтИсп.ПолучитьСтавкуНДСБезНДС();

		Товар.СтавкаНДС = Справочники.СтавкиНДС.СтавкаНДС(Документ.Организация.ВидСтавкиНДСПоУмолчанию, Документ.Дата);

	СтруктураДанныеНоменклатуры = Новый Структура();
	СтруктураДанныеНоменклатуры.Вставить("Номенклатура", Товар.Номенклатура);

	СтруктураДанныеНоменклатуры.Вставить("Организация", Документ.Организация);
	СтруктураДанныеНоменклатуры.Вставить("НалогообложениеНДС", Документ.НалогообложениеНДС);
	СтруктураДанныеНоменклатуры.Вставить("СуммаВключаетНДС", Документ.СуммаВключаетНДС);
	СтруктураДанныеНоменклатуры.Вставить("ВалютаДокумента", Документ.ВалютаДокумента);
	СтруктураДанныеНоменклатуры.Вставить("ВидЦен", Документ.ВидЦен);
	СтруктураДанныеНоменклатуры.Вставить("СтавкаНДС", Товар.СтавкаНДС);
		
	коэф = 1;
	еи = Номенклатура.ЕдиницаИзмерения;
	Если СокрЛП(ДокИст.unit) <> "" Тогда
		теи = ПолучитьОбъект(Справочники.ЕдиницыИзмерения, ДокИст.unit);
		Если не теи.Пустая() Тогда

			КонецЕсли;
		КонецЕсли;
	КонецЕсли;

	Товар.ТипНоменклатурыЗапас = Истина;

							  		Товар.Сумма - (Товар.Сумма) / ((СтавкаНДС + 100) / 100),
							  		Товар.Сумма * СтавкаНДС / 100);

Процедура РаспределитьСумму(Документ, Сумма)
	коэф = Сумма / Документ.СуммаДокумента;
	сумНДС = Документ.РасшифровкаПлатежа.Итог("СуммаНДС") * коэф;
	
	Документ.СуммаДокумента = Сумма;
	Документ.СуммаУчета = Сумма;
	
	Для сч = 0 По Документ.РасшифровкаПлатежа.Количество() - 1 Цикл
		стр = Документ.РасшифровкаПлатежа[сч];
		Если сч = Документ.РасшифровкаПлатежа.Количество() - 1 Тогда
			стр.СуммаРасчетов = Сумма;
			стр.СуммаПлатежа = Сумма;
			стр.СуммаНДС = сумНДС;
		Иначе
			стрСум = стр.СуммаРасчетов * коэф;
			стрНДС = стр.СуммаНДС * коэф;
			сумНдс = сумНдс - стрНДС;
			Сумма = Сумма - стрСум;
			
			стр.СуммаРасчетов = стрСум;
			стр.СуммаПлатежа = стрСум;
			стр.СуммаНДС = стрНДС;
		КонецЕсли;
	КонецЦикла;

		данные.Добавить(Новый Структура("Тип,Данные", "Delivery", накладные));
