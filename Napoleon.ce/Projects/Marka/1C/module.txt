Функция ДобавитьАгента(Агенты, Агент, СвойствоПрефикс)
	Код = КодАгента(Агент);
	
	префикс = "";
	Если Не СвойствоПрефикс.Пустая()  Тогда
		Для каждого дс из Агент.ДополнительныеРеквизиты Цикл
			Если дс.Свойство = СвойствоПрефикс Тогда
				префикс = дс.Значение;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	
	Для Каждого стр Из Агенты Цикл	
		Если стр.Код = Код Тогда
			стр.Наименование = Агент.Наименование;
			стр.Префикс = префикс;
			Возврат Код;	
		КонецЕсли;
	КонецЦикла;
	
	стр = Агенты.Добавить();
	стр.Код = Код;
	стр.Наименование = Агент.Наименование;
	стр.Префикс = префикс;
	
	Возврат Код;

Функция ЗагрузкаИнкассацийАгента(АдресДок, КодАгента, ПринятыеДок, КодировкаANSI)
	Кодировка = ?(КодировкаANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(КодАгента);
	Документ = Неопределено;
	
	СчДок = 0;
	
	Док_имя = КаталогВременныхФайлов() + "TINC.dbf";
	ДокД = ПолучитьИзВременногоХранилища(АдресДок);
	ДокД.Записать(Док_имя);
	Док = Новый XBase(Док_имя);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Док.Кодировка = Кодировка;
	
	КодДок = "";
	ТекСум = 0;
	КодОрг = "";
	ТекДок = Неопределено;
	
	Док.Первая();		
	Пока Док.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Док.ORDFLAG <> 0 или ПустаяСтрока(Док.NUM_I) Тогда
			//
			// Запишем предыдущий документ
			//
			Если ТекДок <> Неопределено Тогда
				Если ТекСум > 0 Тогда
					ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
				КонецЕсли;
				Если ЗаписатьДокументСервер(ТекДок) Тогда
					СообщитьОДокументе(ТекДок);
					ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
					ЗаписатьДокументСервер(ТекДок, Истина);
				КонецЕсли;
			КонецЕсли;
			
			ТекДок = Неопределено;
			ТекСум = Док.SUM;
			КодДок = СокрЛП(Док.CREATED);
			значДок = ПринятыеДок.НайтиПоЗначению(КодДок);
			Если значДок <> Неопределено Тогда			
				СообщитьИнт("Повторный документ " + значДок);
				ТекСум = 0;
			Иначе
				ТекДок = СоздатьИнкассацию(Док.ID, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если ТекДок <> Неопределено Тогда 
			КодОрг = Док.ID;
			Сумма = Док.SUM_I;
			ДокОсн = Документы.РеализацияТоваровУслуг.ПустаяСсылка();
			Если ПустаяСтрока(Док.NUM_I) Тогда
				Сумма = Док.SUM;
			Иначе
				ДокОсн = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Док.NUM_I, Док.DATE_I);
			КонецЕсли;
			
			ДобавитьДокументОснование(ТекДок, ДокОсн, Сумма);
			ТекСум = ТекСум - Сумма;
		КонецЕсли;	
		Док.Следующая();
	КонецЦикла;
	
	Если ТекДок <> Неопределено Тогда
		Если ТекСум > 0 Тогда
			ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
		КонецЕсли;
		Если ЗаписатьДокументСервер(ТекДок) Тогда
			СообщитьОДокументе(ТекДок);
			ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
			ЗаписатьДокументСервер(ТекДок, Истина);
		КонецЕсли;
	КонецЕсли;
	
	Док.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

Процедура ЗаполнитьНоменклатуру(Документ, Товар)
	ТекущаяСтрока = Новый Структура;	
	мета = Документ.Метаданные();
	Для Каждого рек из мета.ТабличныеЧасти.Товары.Реквизиты Цикл
		ТекущаяСтрока.Вставить(рек.Имя);
	КонецЦикла;
	ЗаполнитьЗначенияСвойств(ТекущаяСтрока, Товар);
		
	ТекущаяСтрока.Вставить("ТипНоменклатуры", Перечисления.ТипыНоменклатуры.Товар);
	ТекущаяСтрока.Вставить("Артикул");
	ТекущаяСтрока.Вставить("ВариантОформленияПродажи");
	ТекущаяСтрока.Вставить("ХарактеристикиИспользуются");
	
	СтруктураПересчетаСуммы = ОбработкаТабличнойЧастиКлиентСервер.ПараметрыПересчетаСуммыНДСВСтрокеТЧ(Документ);
	
	СтруктураДействий = Новый Структура;
	СтруктураДействий.Вставить("ПроверитьХарактеристикуПоВладельцу", ТекущаяСтрока.Характеристика);
	СтруктураДействий.Вставить("ПроверитьЗаполнитьУпаковкуПоВладельцу", ТекущаяСтрока.Упаковка);
	СтруктураДействий.Вставить("ПересчитатьКоличествоЕдиниц");
	СтруктураДействий.Вставить("ПроверитьЗаполнитьСклад", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияСкладаВСтрокеТЧ(Документ, Истина));
	Если ЗначениеЗаполнено(Документ.Соглашение) Тогда
		СтруктураДействий.Вставить("ЗаполнитьУсловияПродаж", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияУсловийПродажВСтрокеТЧ(Документ));
	Иначе
		СтруктураДействий.Вставить("ЗаполнитьЦенуПродажи", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияЦеныВСтрокеТЧ(Документ));
	КонецЕсли;
	
	Если ТипЗнч(Документ.Ссылка) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
		СтруктураДействий.Вставить("ЗаполнитьСтавкуНДС", Новый Структура("НалогообложениеНДС, Дата, ПоДатеОтгрузки", Документ.НалогообложениеНДС, Документ.Дата, Ложь));
	Иначе	
		СтруктураДействий.Вставить("ЗаполнитьСтавкуНДС", Новый Структура("НалогообложениеНДС, Дата", Документ.НалогообложениеНДС, Документ.Дата));
		СтруктураДействий.Вставить("ЗаполнитьКодТНВЭД", Документ.НалогообложениеНДС);
	КонецЕсли;
	
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДСВозвратнойТары", Документ.ВернутьМногооборотнуюТару);
	СтруктураДействий.Вставить("ПересчитатьСуммуНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСуммуСНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСумму");
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомРучнойСкидки", Новый Структура("Очищать", Истина));
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомАвтоматическойСкидки", Новый Структура("Очищать", Истина));
	СтруктураДействий.Вставить("ЗаполнитьПризнакТипНоменклатуры", Новый Структура("Номенклатура", "ТипНоменклатуры"));
	СтруктураДействий.Вставить("ЗаполнитьПризнакАртикул", Новый Структура("Номенклатура", "Артикул"));
	СтруктураДействий.Вставить("ЗаполнитьПризнакВариантОформленияПродажи", Новый Структура("Номенклатура", "ВариантОформленияПродажи"));
	СтруктураДействий.Вставить("ЗаполнитьСодержание", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияСодержанияУслугиВСтрокеТЧ(Документ, Ложь));

	ОбработкаТабличнойЧастиСервер.ОбработатьСтрокуТЧ(ТекущаяСтрока, СтруктураДействий, Неопределено);

	ЗаполнитьЗначенияСвойств(Товар, ТекущаяСтрока);
КонецПроцедуры
	
Процедура ДобавитьТовар(Документ, ДокИст, ВидЦеныДокумента)
	Товар = Документ.Товары.Добавить();
	
	Номенклатура = ПолучитьНоменклатуру(ДокИст.ID_I);
	Товар.Номенклатура = Номенклатура.Ссылка;	
	
	Товар.Количество = ДокИст.QTY;
	Товар.КоличествоУпаковок = ДокИст.QTY;
	Товар.Упаковка = Номенклатура.ЕдиницаИзмерения;
	
	Товар.ВидЦены = ВидЦеныДокумента;
	Товар.Склад = Документ.Склад;
	
	ЗаполнитьНоменклатуру(Документ, Товар);
	Если Товар.Цена = 0 Тогда
		Товар.Цена = ДокИст.COST;
	КонецЕсли;
	
	Если ДокИст.PACK <> 0 Тогда
		
		Если Не Номенклатура.НаборУпаковок.Пустая() Тогда
			выб = Справочники.УпаковкиЕдиницыИзмерения.Выбрать(,Номенклатура.НаборУпаковок);
			Если выб.Следующий() Тогда
				Товар.Упаковка = выб.Ссылка;
				Товар.КоличествоУпаковок = Окр(ДокИст.QTY / выб.Числитель * выб.Знаменатель, 0, РежимОкругления.Окр15как20);
				Товар.Цена = Товар.Цена * выб.Числитель / выб.Знаменатель;
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
		
	ТекущаяСтрока = Новый Структура;	
	мета = Документ.Метаданные();
	Для Каждого рек из мета.ТабличныеЧасти.Товары.Реквизиты Цикл
		ТекущаяСтрока.Вставить(рек.Имя);
	КонецЦикла;
	ЗаполнитьЗначенияСвойств(ТекущаяСтрока, Товар);
	ТекущаяСтрока.Вставить("ТипНоменклатуры", Перечисления.ТипыНоменклатуры.Товар);
	ТекущаяСтрока.Вставить("ХарактеристикиИспользуются");
	
	СтруктураПересчетаСуммы = ОбработкаТабличнойЧастиКлиентСервер.ПараметрыПересчетаСуммыНДСВСтрокеТЧ(Документ);
	СтруктураДействий = Новый Структура;
	
	СтруктураДействий.Вставить("ПересчитатьСуммуНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСуммуСНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСумму");
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомРучнойСкидки", Новый Структура("Очищать", Ложь));
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомАвтоматическойСкидки", Новый Структура("Очищать", Истина));
	
	ОбработкаТабличнойЧастиСервер.ОбработатьСтрокуТЧ(ТекущаяСтрока, СтруктураДействий, Неопределено);
	ЗаполнитьЗначенияСвойств(Товар, ТекущаяСтрока);
	
	Если ТипЗнч(Документ.Ссылка) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
		Товар.ВариантОбеспечения = Перечисления.ВариантыОбеспечения.Отгрузить;
	КонецЕсли;
	//ОбработкаТабличнойЧастиСервер.ОбработатьСтрокуТЧ(Товар, СтруктураДействий, Неопределено);

Функция ЗагрузкаЗаявокАгента(АдресЗаказ, КодАгента, ПринятыеЗаказы,КодировкаANSI)
	Кодировка = ?(КодировкаANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ_имя = КаталогВременныхФайлов() + "TORD.dbf";
	ЗаказДД = ПолучитьИзВременногоХранилища(АдресЗаказ);
	ЗаказДД.Записать(Заказ_имя);
	Заказ = Новый XBase(Заказ_имя);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Заказ.Кодировка = Кодировка;
	
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда

				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				СообщитьИнт("Повторный заказ " + значЗаказ);
			Иначе			
				код = Заказ.ID;
				Документ = СоздатьЗаказ(Заказ, код, Агент);
				
				Если МожноМенятьЦену Тогда
					ВидЦеныДокумента = ПолучитьЦену(Заказ.CTYPE);
				Иначе
					ВидЦеныДокумента = Неопределено;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ, ВидЦеныДокумента);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		ЗаписатьДокумент(Документ, ВидЦеныДокумента);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;
