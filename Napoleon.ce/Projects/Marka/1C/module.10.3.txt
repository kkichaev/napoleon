//
// Выбираем фактический адрес, если его нет то юридический
//
Функция АдресаКонтрагента(Контрагент)
	Адреса = РегистрыСведений.КонтактнаяИнформация.СоздатьНаборЗаписей();
	УправлениеКонтактнойИнформацией.ПрочитатьКонтактнуюИнформацию(Адреса, Контрагент.Ссылка);
	
	рет = Новый Структура("Адрес,АдресКА", "", "");
	
	Попытка
		Для каждого Адрес из Адреса Цикл
			стр = Строка(Адрес.Вид);
			Если Найти(стр, "доставк") > 0 или Найти(стр, "Фактический") > 0 Тогда
				Если Адрес.Представление <> "" Тогда
					рет.Адрес = Адрес.Представление;
				Иначе
					рет.Адрес = УправлениеКонтактнойИнформацией.ПолучитьПредставлениеАдреса(Адрес);
				КонецЕсли;
			КонецЕсли;
			Если Найти(стр, "Юридический") > 0 Тогда
				Если Адрес.Представление <> "" Тогда
					рет.АдресКА = Адрес.Представление;
				Иначе
					рет.АдресКА = УправлениеКонтактнойИнформацией.ПолучитьПредставлениеАдреса(Адрес);
				КонецЕсли;
			КонецЕсли;
		КонецЦикла;
	Исключение
		//Сообщить("ошибка АдресКонтрагента " + Контрагент + ": " + ОписаниеОшибки());
	КонецПопытки;
	
	Если ПустаяСтрока(рет.Адрес) Тогда
		рет.Адрес = рет.АдресКА;
	ИначеЕсли ПустаяСтрока(рет.АдресКА) Тогда
		рет.АдресКА = рет.Адрес;
	КонецЕсли;
	
	Возврат рет;

Процедура ВыгрузкаКонтрагентов(ТипыЦен)
	СообщитьИнт("Выгружаем контрагентов по справочнику...");
		
	Менеджеры = ФайлМенеджеров();
	
	//
	// создаем таблицу значений для выгрузки контактов
	//
	Контакты = Новый ТаблицаЗначений;
	Контакты.Колонки.Добавить("ФИО", Новый ОписаниеТипов("Строка"));
	Контакты.Колонки.Добавить("Телефон", Новый ОписаниеТипов("Строка"));
	
	ВсеКАгенты = новый СписокЗначений;
	ВсеКА = ФайлОрганизаций(Истина);
	
	
	Для Каждого астр из Агенты Цикл
		ПроверитьИнформациюПоМенеджеру(Менеджеры, астр.Агент, астр.ПрефиксДокументов);
		
		ОргDBF = ФайлОрганизаций();
        КонтDBF = ФайлКонтактов(астр.Агент);
		
		Запрос = Новый Запрос;
		Запрос.Текст = 
		"ВЫБРАТЬ
		|	Контрагенты.Ссылка КАК Контрагент
		|ИЗ
		|	Справочник.Контрагенты КАК Контрагенты
		|ГДЕ
		|	Контрагенты.Ссылка В ИЕРАРХИИ(&Агент)";
		Запрос.УстановитьПараметр("Агент", астр.Агент);
		Выборка = Запрос.Выполнить().Выбрать();
				
		Пока Выборка.Следующий() Цикл
			Контрагент = Выборка.Контрагент;		
			кодОрг = Контрагент.Код;
			
			стрАдрес = АдресаКонтрагента(Контрагент);
			
			ТелефонКА = ЗаполнитьКонтакты(Контакты, Контрагент);
					
			//
			// добавляем информацию по контрагентам и контактам
			//
			ОргDBF.Добавить();
			ОргDBF.ID = КодОрг;
			ОргDBF.NAME = Контрагент.Наименование;
			ОргDBF.ADDRESS = стрАдрес.Адрес;
			
			ОргDBF.INN = ?(СтрДлина(СокрЛП(Контрагент.КПП)) = 0, Контрагент.ИНН, Контрагент.ИНН + "/" + Контрагент.КПП);
			
			ОргDBF.PHONE = ТелефонКА;
			ОргDBF.LEG_ADR = стрАдрес.АдресКА;
			
			ОргDBF.FULLNAME = Контрагент.НаименованиеПолное;
			ОргDBF.BANK = РСЧвСтроку(Контрагент.ОсновнойБанковскийСчет);
			ОргDBF.COSTYPE = 0;
			ОргDBF.OKPO = Контрагент.КодПоОКПО;
				
			договор = Контрагент.ОсновнойДоговорКонтрагента;
			Если договор <> Null и Не договор.Пустая() Тогда
				//
				// ТипЦены для контрагента выгружаем как позиция цены в order.cfg
				//
				Если договор.ТипЦен <> Неопределено Тогда
					тц = договор.ТипЦен;
					Если ТипЗнч(тц) = Тип("СправочникСсылка.ТипыЦенНоменклатурыКонтрагентов") Тогда
						тц = тц.ТипЦеныНоменклатуры;	
					КонецЕсли;
					Если тц.Пустая() Тогда
						тц = ТипЦеныПрайса;
					КонецЕсли;
					
					типЦены = ТипыЦен.НайтиПоЗначению(тц);
					сч = 0;
					Если типЦены = Неопределено Тогда
						сч = ТипыЦен.Количество();
						ТипыЦен.Добавить(тц, "COST" + Строка(сч+1));
					Иначе
						сч = ТипыЦен.Индекс(типЦены);
					КонецЕсли;
					ОргDBF.COSTYPE = сч;
				КонецЕсли;
			КонецЕсли;
			
			ОргDBF.Записать();		
			
			Если ВсеКАгенты.НайтиПоЗначению(Контрагент) = Неопределено ТОгда
				ВсеКАгенты.Добавить(Контрагент);
				ВсеКА.Добавить();
				ЗаполнитьЗначенияСвойств(ВсеКА, ОргDBF, "ID,ADDRESS,NAME,COSTYPE"); 
				ВсеКА.Записать();
			КонецЕсли;
			//
			// сохраняем информацию о контактах
			//
			Для Каждого Контакт из Контакты Цикл
				КонтDBF.Добавить();
				КонтDBF.ID = КодОрг;
				КонтDBF.FIO = Контакт.ФИО;
				КонтDBF.PHONE = Контакт.Телефон;
				КонтDBF.Записать();			
			КонецЦикла;	
		КонецЦикла;		
		ЗакрытьФайлыАгента(ОргDBF, КонтDBF, астр.Агент);
	КонецЦикла;
		
	ВсеКА.ЗакрытьФайл();
	ПереместитьФайл(ПапкаОбмена + "\TORGS.DBF", ПапкаОбмена + "\ORGS.DBF");  
	ЗакрытьФайлМенеджеров(Менеджеры);
КонецПроцедуры

Процедура ЗаполнитьРасчетныеЦены(стрТовара, цены, ВыгрЦены)
	перем ЦенаБазовая;
	Дата = ТекущаяДата();
	
	Для каждого стр из ВыгрЦены Цикл
		
		ТипЦен = стр.Значение;
		Если Не (ТипЦен.Рассчитывается и ЗначениеЗаполнено(ТипЦен.БазовыйТипЦен)) Тогда
			Продолжить;
		КонецЕсли;
		
		ЦенаБазовая = цены.Получить(ТипЦен.БазовыйТипЦен);
		Если ЦенаБазовая = Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		ПолученнаяЦена = 0;
		
		ВалютаТекущая = ТипЦен.ВалютаЦены;
		ВалютаБазовая = ТипЦен.БазовыйТипЦен.ВалютаЦены;
		
		Если ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоПроцентнойНаценкеНаБазовыйТип Тогда
			ПолученнаяЦена = ЦенаБазовая * (1 + стр.Значение.ПроцентСкидкиНаценки / 100);
			ВалютаТекущая = ВалютаБазовая;
		ИначеЕсли ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоВхождениюБазовойЦеныВДиапазон Тогда
			ПолученнаяЦена = Ценообразование.ПолучитьЦенуПоЦеновымДиапазонам(ТипЦен, Дата, ЦенаБазовая, ВалютаТекущая, ВалютаБазовая);
		КонецЕсли;
		
		ПолученнаяЦена = Ценообразование.ПересчитатьЦенуПриИзмененииВалюты(ПолученнаяЦена, ВалютаТекущая, ВалютаТекущая, 0, 1, Дата);
		ПолученнаяЦена = Ценообразование.ОкруглитьЦену(ПолученнаяЦена, ТипЦен.ПорядокОкругления, ТипЦен.ОкруглятьВБольшуюСторону);
		
		стрТовара.Цены.Вставить(стр.Представление, ПолученнаяЦена);

	КонецЦикла;
	
	цены.Очистить();

	|		ПО ЦеныНоменклатуры.Номенклатура = ТоварыНаСкладахОстатки.Номенклатура
	|
	|СГРУППИРОВАТЬ ПО
	|	ЦеныНоменклатуры.Цена,
	|	ЦеныНоменклатуры.Номенклатура.Ссылка,
	|	ЦеныНоменклатуры.ТипЦен.Ссылка,
	|	ЕСТЬNULL(ТоварыНаСкладахОстатки.КоличествоОстаток, 0),
	|	ЦеныНоменклатуры.ЕдиницаИзмерения
	|
	|УПОРЯДОЧИТЬ ПО
	|	Товар";
	
	
	Запрос = Новый Запрос;
	Запрос.Текст = ТекстЗапроса;
	Запрос.Параметры.Вставить("ТипыЦен", ВыгрЦены);
	Запрос.Параметры.Вставить("Дата", ТекущаяДата());
	Запрос.Параметры.Вставить("СкладОстатков", СкладОстатков);
	
	базовыеЦены = Новый Соответствие;
	текТовар = Неопределено;

		Если текТовар <> стр.Товар Тогда
			Если текТовар <> Неопределено Тогда
				ЗаполнитьРасчетныеЦены(стрТовара, базовыеЦены, ВыгрЦены);
			КонецЕсли;
			текТовар = стр.Товар;

			стрТовара.Цены.Вставить(типЦены.Представление, Цена);

Функция СоздатьЗаказ(Заказ)
	
	параметрыДоговора = ЗаполнениеДокументов.ПолучитьСтруктуруПараметровДляПолученияДоговораЗаказаПокупателя();
	
	Накладная = Документы.ЗаказПокупателя.СоздатьДокумент();
	
	Накладная.Дата = ТекущаяДата();
	Накладная.ДатаОтгрузки = Заказ.DATE;
	Накладная.ДатаОплаты = Заказ.PAYDATE;
	
	//Накладная.ИнициализироватьНовыйДокумент(Неопределено, Неопределено);
	
	контрагент = Справочники.Контрагенты.НайтиПоКоду(Заказ.ID);
	договор = контрагент.ОсновнойДоговорКонтрагента;

	Накладная.УчитыватьНДС = Истина;
	Накладная.СуммаВключаетНДС = Истина;
	
	Накладная.Грузополучатель = контрагент;
	Накладная.Контрагент = контрагент;
	
	ЗаполнениеДокументов.ПриИзмененииЗначенияКонтрагента(Накладная, параметрыДоговора);
	
	Если договор.Пустая() Тогда
		выбд = Справочники.ДоговорыКонтрагентов.Выбрать(,контрагент);
		Пока выбд.Следующий() Цикл
			Если выбд.ВидДоговора = Перечисления.ВидыДоговоровКонтрагентов.СПокупателем Тогда
				договор = выбд.Ссылка;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	Накладная.Организация = договор.Организация;
	Накладная.ДоговорКонтрагента = договор;
	Накладная.ТипЦен = Накладная.ДоговорКонтрагента.ТипЦен;
	Если Накладная.ТипЦен.Пустая() Тогда
		Накладная.ТипЦен = ТипЦеныПрайса;

		Накладная.ДатаОплаты = Накладная.ДатаОтгрузки + Накладная.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 86400;
	КонецЕсли;

		Накладная.ВалютаДокумента = Накладная.ДоговорКонтрагента.ВалютаВзаиморасчетов;
	Иначе
		Накладная.ВалютаДокумента = глЗначениеПеременной("ВалютаРегламентированногоУчета");;
	КонецЕсли;
	Курс = МодульВалютногоУчета.ПолучитьКурсВалюты(Накладная.ВалютаДокумента, Накладная.Дата);
	Накладная.КурсВзаиморасчетов = Курс.Курс;
	Накладная.КратностьВзаиморасчетов = Курс.Кратность;

Процедура ПриемЗаявок() 
	Перем Накладная, агент;
	
	Рез = НайтиФайлы(ПапкаОбмена, "ORD*.DBF");
		
	СчНакл = 0;
	
	Для Каждого файл из Рез Цикл
		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");

		КодСотр = Сред(имяФайла, 4, разделитель-4);
				
		временныйФайл = ПапкаОбмена + "\TORD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		Заказ = Новый XBase(временныйФайл);
		Заказ.Кодировка = КодировкаXBase.OEM;
		ОбработанныеЗаказы = ФайлОбработанныхЗаказов(КодСотр);
		
		ЗаписатьЗаказ = 0;
		ПринятыеЗаявки = ЗагрузитьПринятыеДокументы(КодСотр, Истина);
		Ном = "";
		
		Заказ.Первая();		
		Пока Заказ.ВКонце() = 0 Цикл
			//
			// Проверим признак начала заявки
			//
			Если Заказ.ORDFLAG <> 0 Тогда
				//
				// началась новая заявка, если надо записать заявка, запишем ее.
				//
				Если ЗаписатьЗаказ = 1 Тогда
					ЗаписатьЗаявку(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
					ЗаписатьЗаказ = 0;
				КонецЕсли;
				
				//
				// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
				//
				пз = ПринятыеЗаявки.НайтиПоЗначению(СокрЛП(Заказ.CREATED));
				Если пз <> Неопределено Тогда
					//
					// пропускаем этот заказ
					//
					ЗаписатьЗаказ = 0;
					Заказ.Следующая();
					Сообщить("Повторная заявка - накладная N " + пз.Представление);
					Продолжить;
				КонецЕсли;
				
				//
				// создаем новую заявку
				//
				Накладная = СоздатьЗаказ(Заказ);
				ЗаписатьЗаказ = 1;
				Ном = СокрЛП(Заказ.CREATED);
				СчНакл = СчНакл + 1;
			КонецЕсли;
			
			Если ЗаписатьЗаказ = 0 Тогда
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
						
			номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);

			товар = Накладная.Товары.Добавить();
			товар.Номенклатура = номенклатура.Ссылка;
			
			Если Заказ.PACK <> 0 Тогда
				товар.ЕдиницаИзмерения = Номенклатура.ЕдиницаХраненияОстатков;
			Иначе
				веи = Справочники.ЕдиницыИзмерения.Выбрать(,номенклатура);
				Пока веи.Следующий() Цикл
					Если веи.ЕдиницаПоКлассификатору = номенклатура.БазоваяЕдиницаИзмерения Тогда
						товар.ЕдиницаИзмерения = веи.Ссылка;
						Прервать;
					КонецЕсли;
				КонецЦикла;
			КонецЕсли;
			
			товар.Коэффициент = товар.ЕдиницаИзмерения.Коэффициент;
			
			коэф = товар.ЕдиницаИзмерения.Коэффициент;
			Если коэф = 0 Тогда
				коэф = 1;
			КонецЕсли;
			товар.Количество = Заказ.QTY / коэф;
			товар.Коэффициент = коэф;
			товар.Цена = Ценообразование.ПолучитьЦенуНоменклатуры(товар.Номенклатура,,Накладная.ТипЦен,
				Накладная.Дата,товар.ЕдиницаИзмерения);
			
			товар.СпособСписанияОстаткаТоваров = Перечисления.СпособыСписанияОстаткаТоваров.СоСклада;
			
			ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Накладная);
			
			Заказ.Следующая();
		КонецЦикла;
		
		Если ЗаписатьЗаказ = 1 Тогда
			ЗаписатьЗаявку(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
		КонецЕсли;
		
		СохранитьПринятыеДокументы(КодСотр, ПринятыеЗаявки);
		Заказ.ЗакрытьФайл();
		ЗакрытьФайлОбработанныхЗаявок(ОбработанныеЗаказы, КодСотр);
		
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
	Сообщить("Всего заявок " + СчНакл);
	
КонецПроцедуры

	Накладная.КурсВзаиморасчетов = СтруктураКурсаДокумента.Курс;

Функция ПриемРеализацийАгента(имяФайла, КодСотр)
	СчНакл = 0;
	
	временныйФайл = ПапкаОбмена + "\TSLS.DBF";
	ПереместитьФайл(имяФайла, временныйФайл);
	
	Заказ = Новый XBase(временныйФайл);
	Заказ.Кодировка = КодировкаXBase.OEM;
	ОбработанныеЗаказы = ФайлОбработанныхЗаказов(КодСотр);
	ПринятыеЗаявки = ЗагрузитьПринятыеДокументы(КодСотр, Истина);
	
	ЗаписатьДокумент = 0;
	Накладная = Неопределено;
	Ном = "";
	текПоз = 1;
	
	СкладАгента = СкладОстатков;
	Для Каждого стр из Агенты Цикл
		Если КодАгента(стр.Агент) = КодСотр Тогда
			СкладАгента = стр.Склад;
			Прервать;
		КонецЕсли;
	КонецЦикла;
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			Если ЗаписатьДокумент = 1 Тогда
				ЗаписатьРеализацию(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);				
				ЗаписатьДокумент = 0;
			КонецЕсли;
			текПоз = Заказ.НомерЗаписи();
			
			Док = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Заказ.NUMBER, Заказ.DATE);
			Если Не Док.Пустая() Тогда
				ЗаписатьДокумент = 0;
				Сообщить("Повторная накладная N " + Заказ.NUMBER);
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
			
			//
			// создаем новую заявку
			//
			ЗаписатьДокумент = 1;
			Ном = СокрЛП(Заказ.CREATED);
			СчНакл = СчНакл + 1;
			
			Накладная = СоздатьНакладную(Заказ, СкладАгента);
		КонецЕсли;
		
		Если ЗаписатьДокумент = 0 Тогда
			Заказ.Следующая();
			Продолжить;
		КонецЕсли;
		
		номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);
		товар = Накладная.Товары.Добавить();
		товар.Номенклатура = номенклатура.Ссылка;
		
		
		Если Заказ.PACK <> 0 Тогда
			товар.ЕдиницаИзмерения = Номенклатура.ЕдиницаХраненияОстатков;
		Иначе
			веи = Справочники.ЕдиницыИзмерения.Выбрать(,номенклатура);
			Пока веи.Следующий() Цикл
				Если веи.ЕдиницаПоКлассификатору = номенклатура.БазоваяЕдиницаИзмерения Тогда
					товар.ЕдиницаИзмерения = веи.Ссылка;
					Прервать;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;
		
		товар.Коэффициент = товар.ЕдиницаИзмерения.Коэффициент;
		
		коэф = товар.ЕдиницаИзмерения.Коэффициент;
		Если коэф = 0 Тогда
			коэф = 1;
		КонецЕсли;
		товар.Количество = Заказ.QTY / коэф;
		товар.Коэффициент = коэф;
		товар.Цена = Заказ.COST;
		
		ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар, Накладная);
		ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Накладная);
		ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Накладная);
		
		ОбработкаТабличныхЧастей.РассчитатьКоличествоМестТабЧасти(товар, Накладная);
				
		
		товар.Склад = Накладная.Склад;
		товар.КлючСтроки = УправлениеЗапасами.ПолучитьНовыйКлючСтроки(Накладная);
		товар.КлючСвязи  = 0;
		//

		Заказ.Следующая();
	КонецЦикла;
	
	Если ЗаписатьДокумент = 1 Тогда
		ЗаписатьРеализацию(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	УдалитьФайлы(временныйФайл);
	СохранитьПринятыеДокументы(КодСотр, ПринятыеЗаявки);
	
	Возврат СчНакл;
КонецФункции

	Для Каждого файл из Рез Цикл
		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");

		КодСотр = Сред(имяФайла, 4, разделитель-4);
		
		СкладАгента = СкладОстатков;
		Для Каждого стр из Агенты Цикл
			Если КодАгента(стр.Агент) = КодСотр Тогда

				Прервать;
			КонецЕсли;
		КонецЦикла;
		
		временныйФайл = ПапкаОбмена + "\TWRD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		Заказ = Новый XBase(временныйФайл);
		Заказ.Кодировка = КодировкаXBase.OEM;
		
		ЗаписатьДокумент = 0;
		Накладная = Неопределено;
		
		Приняты = Новый СписокЗначений;
		
		Заказ.Первая();		
		Пока Заказ.ВКонце() = 0 Цикл
			//
			// Проверим признак начала заявки
			//
			Если Заказ.ORDFLAG <> 0 Тогда
				Если ЗаписатьДокумент = 1 Тогда
					ЗаписатьПоступление(Накладная);
					ЗаписатьДокумент = 0;
				КонецЕсли;

				Если Приняты.НайтиПоЗначению(Ном) <> Неопределено Тогда
					Приняты.Добавить(Ном);
					ЗаписатьДокумент = 0;
				Иначе
					СчНакл = СчНакл + 1;				
					Накладная = СоздатьПоступление(Заказ, СкладАгента);
					ЗаписатьДокумент = 1;
				КонецЕсли;

				Заказ.Следующая();

			Если Заказ.QTY > 0 Тогда
				номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);

				товар.Количество = Заказ.QTY / номенклатура.ЕдиницаДляОтчетов.Коэффициент;
				товар.ЕдиницаИзмерения = номенклатура.ЕдиницаДляОтчетов;
				
				
				товар.ЕдиницаИзмеренияМест = номенклатура.ЕдиницаИзмеренияМест;
				Если Не товар.ЕдиницаИзмеренияМест.Пустая() Тогда
					товар.КоличествоМест = 
						Окр(товар.Количество / номенклатура.ЕдиницаИзмеренияМест.Коэффициент, 4, РежимОкругления.Окр15как20);
				КонецЕсли;
					
				товар.Цена = Ценообразование.ПолучитьЦенуНоменклатуры(товар.Номенклатура,,ТипЦеныПрайса);
			КонецЕсли;
			Заказ.Следующая();
		КонецЦикла;
		
		Если ЗаписатьДокумент = 1 Тогда
			ЗаписатьПоступление(Накладная);
		КонецЕсли;
		
		Заказ.ЗакрытьФайл();
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
	Сообщить("Всего заявок на борт (ПеремещениеТоваров): " + СчНакл);

				// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
				//
				пз = ПринятыеДок.НайтиПоЗначению(СокрЛП(Док.CREATED));
				Если пз <> Неопределено Тогда
					//
					// пропускаем этот заказ
					//
					ЗаписатьДок = 0;
					Док.Следующая();
					СообщитьИнт("Повторный ПКО N " + пз.Представление);
					Продолжить;
				КонецЕсли;
				
				//
				// создаем новую заявку
				//
				Накладная = СоздатьПКО(Док);
				ЗаписатьДок = 1;
				СумДок = Док.SUM;
				Ном = СокрЛП(Док.CREATED);
				СчНакл = СчНакл + 1;
			КонецЕсли;
			
			Если ЗаписатьДок = 0 Тогда
				Док.Следующая();
				Продолжить;
			КонецЕсли;
			
			номер = Док.NUM_I;
			Если Найти(номер, "Реализация товаров и услуг") > 0 Тогда
				тстр = СокрЛП(Прав(номер, СтрДлина(номер) - СтрДлина("Реализация товаров и услуг")));
				инд = Найти(тстр, " ");
				Если инд > 0 Тогда 
					номер = СокрЛП(Лев(тстр, инд));
				Иначе 					
					номер =  тстр;
				КонецЕсли;
			КонецЕсли;
			реализация = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(номер, Док.DATE_I);
			
			Если ЗначениеЗаполнено(реализация) Тогда
				//ДокПКО = Документы.ПриходныйКассовыйОрдер.СоздатьДокумент();
				НС = Накладная.РасшифровкаПлатежа.Добавить();
				НС.ДоговорКонтрагента = реализация.ДоговорКонтрагента;
				НС.ДокументРасчетовСКонтрагентом = реализация;
				НС.СуммаВзаиморасчетов = Док.SUM_I;
				НС.КурсВзаиморасчетов = 1;
				НС.КратностьВзаиморасчетов = 1;
				НС.СуммаПлатежа = НС.СуммаВзаиморасчетов;
				НС.СтатьяДвиженияДенежныхСредств = НС.ДоговорКонтрагента.ОсновнаяСтатьяДвиженияДенежныхСредств;
				
				СумДок = СумДок - Док.SUM_I;
			КонецЕсли;
			
			Док.Следующая();
		КонецЦикла;
		
		Если ЗаписатьДок = 1 Тогда
			ЗаписатьПКО(Накладная, СумДок, Ном, ПринятыеДок, ОбработанныеДокы);
		КонецЕсли;
		
		СохранитьПринятыеДокументы(КодСотр, ПринятыеДок, "Incass");
		Док.ЗакрытьФайл();
		ЗакрытьФайлОбработанныхЗаявок(ОбработанныеДокы, КодСотр);
		
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
	СообщитьИнт("Всего ПКО " + СчНакл);
КонецПроцедуры

Функция СохранитьНакладную(накладная, Документ, СуммаДолга, НомерЗаказа)
	суммаДок = Документ.Товары.Итог("Сумма");
	Если СуммаДолга < суммаДок Тогда
		суммаДок = СуммаДолга;
	КонецЕсли;
	Для Каждого товар из Документ.Товары Цикл
		накладная.Добавить();
		накладная.ID = Документ.Контрагент.Код;
		накладная.NUM = Документ.Номер;
		накладная.DATE = Документ.Дата;
		накладная.ID_I = товар.Номенклатура.Код;
		накладная.QTY = товар.Количество;
		накладная.SUM = товар.Сумма * Документ.КурсВзаиморасчетов;
		накладная.SUMD = суммаДок;
		накладная.PAYDATE = Документ.Дата + Документ.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 24 * 3600;
		накладная.CREATED = НомерЗаказа;
		накладная.Записать();
	КонецЦикла;
	
	Возврат СуммаДолга - суммаДок;

	|	И РеализацияТоваровУслуг.Дата МЕЖДУ &ДатНач И &ДатКнц
	|
	|УПОРЯДОЧИТЬ ПО
	|	Контрагент,
	|	Дата УБЫВ";

		текАгент = астр.Агент;
		оплаты = ФайлОплат();
		отгрузки = ФайлОтгрузок();
		ПринятыеЗаявки = ЗагрузитьПринятыеДокументы(КодАгента(текАгент), Ложь);
	
		Запрос = Новый Запрос;
		Запрос.Текст = ТекстЗапроса;
		Запрос.Параметры.Вставить("ДатКнц", ДатаКон);
		Запрос.Параметры.Вставить("ДатНач", ДатаНач);
		Запрос.УстановитьПараметр("ПапкаКонтрагентов", астр.Агент);

		валюта = глЗначениеПеременной("ВалютаУправленческогоУчета");

					оплаты.ID = текКА.Код;
					оплаты.NUM = "нач.сальдо";
					оплаты.SUM = суммаДолга;
					оплаты.DATE = ДатаНач;
					оплаты.Записать();
				КонецЕсли;
				суммаДолга = выборка.Долг;
				текКА = Выборка.Контрагент;
			КонецЕсли;
			
			Если ТипЗнч(Выборка.Документ) = Тип("ДокументСсылка.РеализацияТоваровУслуг") Тогда

					НомерЗаказа = НомерЗаказа.Представление;
				Иначе

			оплаты.NUM = "нач.сальдо";
