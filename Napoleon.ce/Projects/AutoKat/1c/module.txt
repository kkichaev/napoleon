Функция СтдТелефон(телефон)

	Если пс = "8" Тогда 		// 8905...
		рез = "+7" + Прав(телефон, СтрДлина(телефон)-1);
	ИначеЕсли пс = "+" Тогда 	// +7905...
		рез = телефон;
	Иначе                       // 905...
		рез = "+7" + телефон;

	рез = СтрЗаменить(рез, " ", "");
	рез = СтрЗаменить(рез, "(", "");
	рез = СтрЗаменить(рез, ")", "");
	рез = СтрЗаменить(рез, "-", "");
	Возврат рез;
