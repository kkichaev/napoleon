Функция ПриемЗаявок(Соединение)
	stmt = 
	"if (OBJECT_ID('MarkOrderHandled', 'P') IS NULL) 
	| exec('
	|  create procedure MarkOrderHandled(@userid nvarchar(50), @created bigint, @number nvarchar(50))
	|  as 
	|  begin 
	|  insert into [OrderProceeded]([created], [type], [userid]) values (@created, ''Order'', @userid)
	|  if exists( select * from [OrderCommitted] where [created] = @created and [userid] = @userid and [type] = ''Order'')
	|    update [OrderCommitted] set [number] = @number where [created] = @created and [userid] = @userid
	|  else
	|    insert into [OrderCommitted] ([userid], [created], [number], [type]) values (@userid, @created, @number, ''Order'')
	|  end
	| ')";
	
	ВыполнитьВыражение(Соединение, stmt);
	
	stmt = "
	|SELECT 
	|  dateadd(ss, o.created/ 10000000 - 11644473600,'01/01/1970') as created, 
	|  dateadd(ss, o.date/ 10000000 - 11644473600,'01/01/1970') as date, 
	|  o.id, o.firmCode, o.cash, o.userid, o.created as cr_id, o.remark,
	|  o.delivery,
	|  (cast(o.created as varchar(30)) + o.userid) as order_id,
	|  i.id as id_i, 
	|  i.cost as cost, i.qty as qty
	|FROM ([Order] o left join OrderCommitted oc on o.created = oc.created and o.userid = oc.userid) , [Order$items] i
	|  where o.created = i.[Order$created] and o.userid = i.[Order$userid] 
	|  and oc.created is null
	|";
	
	обработанныеДокументы = СоздатьКоманду(Соединение, "[MarkOrderHandled]", ТипКомандыПроцедура());
	
	СчДок = 0;
	
	заявки = Новый ComОбъект("ADODB.Recordset");
	заявки.Open(stmt, Соединение);
		
	текЗаявка = "";
	Накладная = Неопределено;
	кодДок = "";
	кодАгента = "";
	created = "";
	ЗаписатьЗаказ = Ложь;
	
	ЗаписанныеЗаказы = Новый Массив;
	
	Попытка
	
		Агент = Неопределено;
		Пока Не заявки.EOF Цикл
			идЗаявки = заявки.Fields.Item("order_id").Value;
			Если идЗаявки <> текЗаявка Тогда
				Если ЗаписатьЗаказ Тогда
					Если ЗаписатьЗаказ(Соединение, Накладная) Тогда
						СчДок = СчДок + 1;
						СообщитьОДокументе(Накладная);
						стрЗаказа = Новый Структура("Номер,Агент,Док", Накладная.Номер, КодАгента, created);
						ЗаписанныеЗаказы.Добавить(стрЗаказа);
					КонецЕсли;
					ЗаписатьЗаказ = Ложь;
				КонецЕсли;
				текЗаявка = идЗаявки;
				кодАгента = заявки.Fields.Item("userid").Value;
				created = заявки.Fields.Item("cr_id").Value;
				Агент = НайтиАгентаПоКоду(кодАгента);
				
				Попытка
					Накладная = СоздатьЗаказ(заявки, Агент);
					ЗаписатьЗаказ = Истина;
				Исключение
					Сообщить(ОписаниеОшибки());
					ЗаписатьЗаказ = Ложь;
				КонецПопытки;
			КонецЕсли;
			
			Если ЗаписатьЗаказ Тогда
				ДобавитьТовар(Накладная, заявки, ЦенаПрайса);
				Накладная.СуммаДокумента = Накладная.ПолучитьСуммуЗаказанныхСтрок();
			КонецЕсли;
			заявки.MoveNext();	
		КонецЦикла;
		
		Если ЗаписатьЗаказ Тогда
			Если ЗаписатьЗаказ(Соединение, Накладная) Тогда
				СчДок = СчДок + 1;
				СообщитьОДокументе(Накладная);
				стрЗаказа = Новый Структура("Номер,Агент,Док", Накладная.Номер, КодАгента, created);
				ЗаписанныеЗаказы.Добавить(стрЗаказа);
			КонецЕсли;
		КонецЕсли;
	Исключение
		ош = ОписаниеОшибки();
		ЗаписатьВЛог(Соединение, ош);
		Сообщить(ош);		
	КонецПопытки;
	заявки.Close();
	
	Соединение.BeginTrans();
	Для Каждого стр из ЗаписанныеЗаказы Цикл
		УстановитьПараметр(обработанныеДокументы, "@number", стр.Номер);				
		УстановитьПараметр(обработанныеДокументы, "@userid", стр.Агент);
		УстановитьПараметр(обработанныеДокументы, "@created", стр.Док);
		обработанныеДокументы.Execute();
	КонецЦикла;
	Соединение.CommitTrans();
	Возврат СчДок;
