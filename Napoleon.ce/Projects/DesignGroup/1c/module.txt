Функция ПолучитьТаблицуКоманд()

   // Как будет выглядеть описание печатной формы для пользователя
   Команды.Колонки.Добавить("Представление", Новый ОписаниеТипов("Строка"));

   // - ОткрытиеФормы - в этом случае в колонке идентификатор должно быть указано имя формы, которое должна будет открыть система
   // - ВызовКлиентскогоМетода - вызвать клиентскую экспортную процедуру из модуля формы обработки
   // - ВызовСерверногоМетода - вызвать серверную экспортную процедуру из модуля объекта обработки

   Команды.Колонки.Добавить("ПоказыватьОповещение", Новый ОписаниеТипов("Булево"));

Функция СоздатьЗаказ(ДокИст, КодДоговора, КодАдреса, Склад, Агент)
	
	Документ = Документы.ЗаказКлиента.СоздатьДокумент();
	Документ.Дата = ТекущаяДата();
	Документ.ЖелаемаяДатаОтгрузки = ДокИст.DATE;
	Документ.НеОтгружатьЧастями = Истина;
	Документ.ДатаОтгрузки = ДокИст.DATE;
	Документ.Менеджер = Агент;
	
	уид = Новый УникальныйИдентификатор(КодДоговора);
	Договор = Справочники.ДоговорыКонтрагентов.ПолучитьСсылку(уид);
	
	//уид = Новый УникальныйИдентификатор(ДокИст.ID_WH);
	//Склад = Справочники.Склады.ПолучитьСсылку(уид);
	
	Документ.Договор = Договор;
	Документ.Партнер = Договор.Партнер;
	Документ.Контрагент = Договор.Контрагент;
	Документ.Валюта = Договор.ВалютаВзаиморасчетов;
	Документ.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.РеализацияКлиенту;
	
	Если КодАдреса > 0 Тогда
		Для Каждого астр из Документ.Партнер.КонтактнаяИнформация Цикл
			Если астр.НомерСтроки = КодАдреса Тогда
				Документ.АдресДоставки = астр.Представление;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	
	Документ.Организация = Договор.Организация;	
	Документ.Склад = Склад;
	Документ.НалогообложениеНДС = Справочники.Организации.НалогообложениеНДС(Документ.Организация, Документ.Склад, Документ.Дата);
	
	Запрос = Новый Запрос("ВЫБРАТЬ
	                      |	СоглашенияСКлиентами.Ссылка
	                      |ИЗ
	                      |	Справочник.СоглашенияСКлиентами КАК СоглашенияСКлиентами
	                      |ГДЕ
	                      |	СоглашенияСКлиентами.Организация = &Организация
	                      |	И СоглашенияСКлиентами.Партнер = &Партнер");
	
	Запрос.УстановитьПараметр("Организация", Документ.Организация);
	Запрос.УстановитьПараметр("Партнер", Документ.Партнер);
	выб = Запрос.Выполнить().Выбрать();
	Если выб.Следующий() Тогда
		Документ.Соглашение = выб.Ссылка;
		Документ.ЦенаВключаетНДС = выб.Ссылка.ЦенаВключаетНДС;
	КонецЕсли;
	
	Документ.Статус = Перечисления.СтатусыЗаказовКлиентов.НеСогласован;
	
	Документ.Комментарий = ДокИст.REMARK;
	Документ.СпособДоставки = Перечисления.СпособыДоставки.Самовывоз;
	
	Возврат Документ;
КонецФункции

Процедура ДобавитьТоварВЗаказ(Документы, ДокИст, ВидЦеныДокумента)
	уид = Новый УникальныйИдентификатор(ДокИст.ID_I);
	Номенклатура = Справочники.Номенклатура.ПолучитьСсылку(уид);
	Кол_во = ДокИст.QTY;
	
	Запрос = новый Запрос;
	Запрос.Текст = 
	"ВЫБРАТЬ
	|	ТоварыНаСкладахОстатки.Склад,
	|	ТоварыНаСкладахОстатки.ВНаличииОстаток КАК Кол_во
	|ИЗ
	|	РегистрНакопления.ТоварыНаСкладах.Остатки(
	|			&ТекДата,
	|			Склад В (&Склады)
	|				И Номенклатура = &Товар) КАК ТоварыНаСкладахОстатки";
	
	Скл = Новый Массив;
	Для Каждого док из Документы Цикл
		Скл.Добавить(док.Склад);
	КонецЦикла;
	
	Запрос.УстановитьПараметр("Склады", Скл);
	Запрос.УстановитьПараметр("Товар", Номенклатура);
	Запрос.УстановитьПараметр("ТекДата", ТекущаяДата());
	
	ост = Запрос.Выполнить().Выгрузить();
	Для каждого док из Документы Цикл
		строки = ост.НайтиСтроки(Новый Структура("Склад", док.Склад));
		Для Каждого стр из строки Цикл
			текКол = Кол_во;
			Если текКол > стр.Кол_во Тогда
				текКол = стр.Кол_во;
			КонецЕсли;
			Если текКол > 0 Тогда
				ДобавитьТовар(док, ДокИст, ВидЦеныДокумента, Номенклатура, текКол);
				Кол_во = Кол_во - текКол;
				Если Кол_во = 0 Тогда
					Возврат;
				КонецЕсли;
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
