Процедура ЗаписатьПодтверждение(КодАгента, НовДок, ТипДок)
	Если ТипДок = Неопределено Тогда
		ТипДок = "Order";
	КонецЕсли;
	
	ДокСтр = ?(ТипДок = "Order", "Заказ клиента", "ПКО");
	
	// Write POD file
	имяФайла = Объект.ПапкаОбмена + "\POD" + КодАгента + ".DBF";
	имяВр = Объект.ПапкаОбмена + "\TPOD.DBF";
	ДБФ = Новый XBase();
	файл = Новый Файл(имяФайла);
	Если Не файл.Существует() Тогда
		ДБФ.Поля.Добавить("CREATED", "S", 25, 0);	
		ДБФ.Поля.Добавить("TYPE", "S", 25, 0);	
		ДБФ.СоздатьФайл(имяВр);
	Иначе
		ПереместитьФайл(имяФайла, имяВр);
		ДБФ.ОткрытьФайл(имяВр);
	КонецЕсли;
	
	ДБФ.Кодировка = КодировкаXBase.OEM;
	Для Каждого стр из НовДок Цикл
		ДБФ.Добавить();
		ДБФ.CREATED = стр.Значение;
		ДБФ.TYPE = ТипДок;
		
		СообщитьОДокументе(ДокСтр, стр.Представление);
		ДБФ.Записать();
	КонецЦикла;
	ДБФ.ЗакрытьФайл();	
	ПереместитьФайл(имяВр, имяФайла);
	
	
	// Write PRD file
	новФайл = Объект.ПапкаОбмена + "\PRD" + КодАгента + ".DBF";
	имяФайла = Объект.ПапкаОбмена + "\TPRD.DBF";
	ДБФ = Новый XBase();
	ДБФ.Поля.Добавить("CREATED",   "S", 20, 0);
	ДБФ.Поля.Добавить("NUMBER", "S", 20, 0);
	ДБФ.Поля.Добавить("TYPE", "S", 20, 0);
	ДБФ.СоздатьФайл(имяФайла);
	ДБФ.Кодировка = КодировкаXBase.OEM;
	
	ДатаНач = ДобавитьМесяц(ТекущаяДата(), -1);
	нач = Формат(ДатаНач, "ДФ=""ггггММдд000000""");
	
	датаПров = НачалоМесяца(ДобавитьМесяц(ТекущаяДата(), -1));
	
	файл = Новый Файл(новФайл);
	Если файл.Существует() Тогда
		Ист = Новый XBase();
		врФайл = Объект.ПапкаОбмена + "\TPRDSC.DBF";
		КопироватьФайл (новФайл, врФайл);
		Ист.ОткрытьФайл(врФайл);
		Ист.Кодировка = КодировкаXBase.OEM;
		
		Пока Не Ист.ВКонце() Цикл
			Если Дата(СокрЛП(Ист.CREATED)) > датаПров и СокрЛП(Ист.TYPE) <> ТипДок Тогда
				ДБФ.Добавить();
				ЗаполнитьЗначенияСвойств(ДБФ, Ист, "CREATED,NUMBER,TYPE");
				ДБФ.Записать();         
			КонецЕсли;
			Ист.Следующая();
		КонецЦикла;
		Ист.ЗакрытьФайл();
	КонецЕсли;
	
	Для Каждого стр из НовДок Цикл
		Если стр.Значение > нач Тогда
			ДБФ.Добавить();
			ДБФ.CREATED = стр.Значение;
			ДБФ.NUMBER = стр.Представление;
			ДБФ.TYPE = ТипДок;
			ДБФ.Записать();         
		КонецЕсли;
	КонецЦикла;
	
	ДБФ.ЗакрытьФайл();

	ПереместитьФайл(имяФайла, новФайл);

Функция ЗагрузитьПринятыеДокументы(КодАгента, ПоЗаявкам, ТипДок=Неопределено)

	имяФайла = Объект.ПапкаОбмена + "\PRD" + КодАгента + ".DBF";
	файл = Новый Файл(имяФайла);

		ДБФ = Новый XBase();

		ДБФ.Первая();
		Пока Не ДБФ.ВКонце() Цикл
			Если СокрЛП(ДБФ.TYPE) = ТипДок Тогда
				Если ПоЗаявкам Тогда
					ПринятыеДок.Добавить(СокрЛП(ДБФ.CREATED), СокрЛП(ДБФ.NUMBER));
				Иначе
					ПринятыеДок.Добавить(СокрЛП(ДБФ.NUMBER), СокрЛП(ДБФ.CREATED));
				КонецЕсли;
			КонецЕсли;
			ДБФ.Следующая();
		КонецЦикла;
		
		ДБФ.ЗакрытьФайл();
	КонецЕсли;
	
	Возврат ПринятыеДок;
