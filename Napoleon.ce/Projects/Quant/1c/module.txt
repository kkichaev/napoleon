	КодировкаБазы = ?(КодировкаБазыANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM);

	|УПОРЯДОЧИТЬ ПО

	|		ПО СоглашенияСКлиентами.СегментПартнеров = ПартнерыСегмента.Сегмент

	|	ПартнерыСегмента.Партнер = &Партнер
	|	И СоглашенияСКлиентами.Типовое = ИСТИНА";
	
	Запрос.Параметры.Вставить("Партнер", Партнер);
	
	Выб = Запрос.Выполнить().Выбрать();
	Пока выб.Следующий() цикл
		Если Не выб.Ссылка.Типовое Тогда
			соглашение = выб.Ссылка;
			прервать;	
		КонецЕсли;
		Если выб.Ссылка.Организация = Документ.Организация Тогда
			соглашение = выб.Ссылка;	
		КонецЕсли;
	КонецЦикла;
		
	Документ.Соглашение = соглашение;
	
	Если Документ.Соглашение.Пустая() Тогда
		Документ.ЦенаВключаетНДС = Истина;
		Документ.ЗаполнитьУсловияПродажПоУмолчанию();
	Иначе
		Документ.ЦенаВключаетНДС = Документ.Соглашение.ЦенаВключаетНДС;

	Номенклатура = Справочники.Номенклатура.НайтиПоКоду(ДокИст.ID_I);
	Товар.Номенклатура = Номенклатура;
	коэф = 1;
	//Если ДокИст.PACK <> 0 И не Номенклатура.алкОсновнаяУпаковка.Пустая() Тогда
	//	Товар.Упаковка = Номенклатура.алкОсновнаяУпаковка;
	//	коэф = Номенклатура.алкОсновнаяУпаковка.Числитель;
	//КонецЕсли;
	
	Товар.КоличествоУпаковок = ДокИст.QTY / коэф;
	Товар.ВидЦены = ВидЦеныДок;
	Товар.Цена = ДокИст.COST * коэф;
	Товар.Склад = СкладТовара(Номенклатура, складыСЗ);
	
	
	
	//
	////Товар.ДатаОтгрузки = ДокИст.DATE;
	//
	Данные = Новый Структура;	
	мета = Документ.Метаданные();
	Для Каждого рек из мета.ТабличныеЧасти.Товары.Реквизиты Цикл
		Данные.Вставить(рек.Имя);
	КонецЦикла;
	ЗаполнитьЗначенияСвойств(Данные, Товар);
	Данные.Вставить("ТипНоменклатуры");
	Данные.Вставить("ХарактеристикиИспользуются");
	
	СтруктураПересчетаСуммы = ОбработкаТабличнойЧастиКлиентСервер.ПараметрыПересчетаСуммыНДСВСтрокеТЧ(Документ);

		СтруктураДействий.Вставить("ПроверитьЗаполнитьУпаковкуПоВладельцу", Товар.Упаковка);
	КонецЕсли;
	СтруктураДействий.Вставить("ПересчитатьКоличествоЕдиниц");
	//СтруктураДействий.Вставить("ПроверитьЗаполнитьСклад", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияСкладаВСтрокеТЧ(Документ, Ложь));
	
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДС", ОбработкаТабличнойЧастиКлиентСервер.ПараметрыЗаполненияСтавкиНДС(Документ, Истина));
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДСВозвратнойТары", Документ.ВернутьМногооборотнуюТару);
	СтруктураДействий.Вставить("ПересчитатьСуммуНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСуммуСНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСумму");
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомРучнойСкидки", Новый Структура("Очищать", Истина));
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомАвтоматическойСкидки", Новый Структура("Очищать", Истина));

	ОбработкаТабличнойЧастиСервер.ОбработатьСтрокуТЧ(Данные, СтруктураДействий, Неопределено);
	ЗаполнитьЗначенияСвойств(Товар, Данные);
	Если ТипЗнч(Документ.Ссылка) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
//		Товар.ВариантОбеспечения = Перечисления.ВариантыОбеспечения.Отгрузить;
		Товар.ВариантОбеспечения = Перечисления.ВариантыОбеспечения.НеТребуется;

	//ОбработкаТабличнойЧастиСервер.ОбработатьСтрокуТЧ(Товар, СтруктураДействий, Неопределено);
КонецПроцедуры

Функция ЗагрузкаЗаявокАгента(временныйФайл, КодАгента, ПринятыеЗаказы, РезультатДок)
	складыСЗ = Новый СписокЗначений;
	складыСЗ.Добавить(СкладОстатков);
	Для каждого стр из Склады Цикл 
		складыСЗ.Добавить(стр.Склад);
	КонецЦикла;
	
	Кодировка = ?(КодировкаБазыANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ = Новый XBase(временныйФайл);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Заказ.Кодировка = Кодировка;
	
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ, ВидЦеныДокумента);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ, РезультатДок);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);                                            ф
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				//Сообщить("Повторный заказ " + значЗаказ);
			Иначе			
				код = СокрЛП(Заказ.ID);
				Документ = СоздатьЗаказ(Заказ, код, Агент);
				Попытка
					уид = Новый УникальныйИдентификатор(Заказ.PRCTYPE);
					ВидЦеныДокумента = Справочники.ВидыЦен.ПолучитьСсылку(уид);
				Исключение
					ВидЦеныДокумента = ВидыЦен[0];
				КонецПопытки;
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ, складыСЗ, ВидЦеныДокумента);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		ЗаписатьДокумент(Документ, ВидЦеныДокумента);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ, РезультатДок);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

Процедура ЗагрузитьПКО(РезультатДок)

		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		ПринятыеДок = ЗагрузитьПринятыеДокументы(КодАгента, Истина, "Incass");
		
		Кодировка = ?(КодировкаБазыANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
		Агент = НайтиАгентаПоКоду(КодАгента);
		Док = Новый XBase(временныйФайл);
		Док.Кодировка = Кодировка;
		Документ = Неопределено;
		СумДок = 0;
		КодДок = "";
		
		пропуститьДок = Ложь;
		Пока Не Док.ВКонце() Цикл
						
			Если Док.ORDFLAG = 1 или СокрЛП(Док.NUM_I) = "" Тогда
				Если Документ <> Неопределено Тогда
					Если СумДок > 0 Тогда
						ДобавитьСумму(Документ, СумДок);
					КонецЕсли;
					Если ЗаписатьДокументСервер(Документ) Тогда
						СообщитьОДокументе(Документ, РезультатДок);
						ПринятыеДок.Добавить(КодДок, Документ.Номер);
					КонецЕсли;
				КонецЕсли;
					
				Документ = Неопределено;
				КодДок = СокрЛП(Док.CREATED);
				значДок = ПринятыеДок.НайтиПоЗначению(КодДок);
				Если значДок <> Неопределено Тогда			
					пропуститьДок = Истина;
					Док.Следующая();
					Продолжить;
				Иначе
					пропуститьДок = Ложь;
					
					Документ = Документы.ПриходныйКассовыйОрдер.СоздатьДокумент();
						
					Документ.Дата = ТекущаяДата();
					//Документ.Менеджер = Агент;
					Документ.Подразделение = Документ.Подразделение;
					Документ.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.ПоступлениеОплатыОтКлиента;
					Документ.Комментарий = Док.REMARK;
					
					Документ.Организация = ПолучитьОрганизацию();
					
					выб = Справочники.Кассы.Выбрать(,Документ.Организация.Ссылка);
					Если выб.Следующий() Тогда
						Документ.Касса = выб.Ссылка;
					КонецЕсли;
					партнер = Справочники.Партнеры.НайтиПоКоду(СокрЛП(Док.ID));
					
					Документ.Партнер = партнер;
					Запрос = Новый Запрос;

					|	Контрагенты.Ссылка КАК Контрагент
					|ИЗ
					|	Справочник.Контрагенты КАК Контрагенты
					|ГДЕ
					|	Контрагенты.Партнер = &Партнер";
					Запрос.УстановитьПараметр("Партнер", партнер);
					выб = Запрос.Выполнить().Выбрать();
					Если выб.Следующий() Тогда
						Документ.Контрагент = выб.Контрагент;
					КонецЕсли;
					
					//ЗаполнитьКонтрагентаДоговор(Документ, Истина, Неопределено);

				Продолжить;
			КонецЕсли;

				Попытка  //Исправлено

						накл = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Док.NUM_I, Док.DATE_I);
