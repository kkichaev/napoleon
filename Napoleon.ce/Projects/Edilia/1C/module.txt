Функция СоздатьПроцедуруДобавитьАгента(ADOСоединение)
	ТекстЗапроса = 
	"IF EXISTS (SELECT *  FROM   sysobjects WHERE  id = object_id(N'[dbo].[AddAgent]') and OBJECTPROPERTY(id, N'IsProcedure') = 1)
	|BEGIN
	|	DROP PROCEDURE [dbo].[AddAgent];
	|END;";
	ADOСоединение.Execute(ТекстЗапроса);
	
	ТекстЗапроса = 
	"CREATE procedure [AddAgent](@id NVARCHAR(300), @name NVARCHAR(300), @division VARCHAR(300))
	|as
	|begin tran
	|update [Agents] with (serializable) set [name] = @name, [division] = @division where [id] = @id
	|if @@rowcount = 0
	|begin
 	|	insert into [Agents] ([id], [name], [division]) values (@id, @name, @division)
	|end;
	|commit tran;";
	ADOСоединение.Execute(ТекстЗапроса);
	
	добавитьАгента = Новый COMОбъект("ADODB.Command"); 
	добавитьАгента.ActiveConnection = ADOСоединение;
	добавитьАгента.CommandText ="AddAgent";
	добавитьАгента.CommandType=4; 
	добавитьАгента.Parameters.Append (добавитьАгента.CreateParameter("@id", 200, 1, 300));
	добавитьАгента.Parameters.Append (добавитьАгента.CreateParameter("@name", 200, 1, 300));
	добавитьАгента.Parameters.Append (добавитьАгента.CreateParameter("@division", 200, 1, 300));
	добавитьАгента.Prepared = true; 
	
	Возврат добавитьАгента;

	добавитьАгента.Execute();	
			
	Запрос = Новый Запрос;
	Запрос.Текст = "
	|ВЫБРАТЬ
	|	ДоговорыКонтрагентов.Ссылка КАК Договор,
	|	ДоговорыКонтрагентов.Владелец КАК Контрагент,
	|	ДоговорыКонтрагентов.ТипЦен,
	|	КонтактнаяИнформация.Представление КАК Адрес
	|ИЗ
	|	Справочник.ДоговорыКонтрагентов КАК ДоговорыКонтрагентов
	|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.КонтактнаяИнформация КАК КонтактнаяИнформация
	|		ПО ДоговорыКонтрагентов.Владелец = КонтактнаяИнформация.Объект
	|			И (КонтактнаяИнформация.Вид = &ВидКИ)
	|ГДЕ
	|	(ДоговорыКонтрагентов.Владелец.Ссылка В (&Контрагенты)
	|			ИЛИ ДоговорыКонтрагентов.Владелец.Ссылка В ИЕРАРХИИ (&Папки))
	|	И ДоговорыКонтрагентов.Команда = &Команда
	|	И ДоговорыКонтрагентов.ВидДоговора = ЗНАЧЕНИЕ(Перечисление.ВидыДоговоровКонтрагентов.СПокупателем)";
	
	Папки = Новый СписокЗначений;
	Контрагенты = Новый СписокЗначений;
	Для Каждого эл из Агент.Контрагенты Цикл
		Если эл.Контрагент.ЭтоГруппа Тогда
			Папки.Добавить(эл.Контрагент.Ссылка);
		Иначе
			Контрагенты.Добавить(эл.Контрагент.Ссылка);
		КонецЕсли;
	КонецЦикла;
	Запрос.Параметры.Вставить("Папки", Папки);
	Запрос.Параметры.Вставить("Контрагенты", Контрагенты);
	Запрос.Параметры.Вставить("Команда", Агент.Агент.Команда);
	Запрос.Параметры.Вставить("ВидКИ", Справочники.ВидыКонтактнойИнформации.ФактАдресКонтрагента);
	
	rs  = RSКонтрагенты(ADOСоединение);
	rscnt  = RSКонтакты(ADOСоединение);
	сч = 0;
	
	Контакты = Новый ТаблицаЗначений;
	Контакты.Колонки.Добавить("ФИО", Новый ОписаниеТипов("Строка"));
	Контакты.Колонки.Добавить("Телефон", Новый ОписаниеТипов("Строка"));
	
	выгрузили = Новый СписокЗначений;
	выб = Запрос.Выполнить().Выбрать();
	Пока выб.Следующий() Цикл
		код = Строка(выб.Контрагент.УникальныйИдентификатор());
		Если выгрузили.НайтиПоЗначению(код) <> Неопределено Тогда
			Продолжить;
		КонецЕсли;
		выгрузили.Добавить(код);
		
		Если выб.ТипЦен = Null или выб.ТипЦен.Пустая() Тогда
			текст = Строка(выб.Контрагент) + " не установлен тип цен в договоре контрагент не выгружен";
		#Если Клиент Тогда
			Сообщить(текст);
		#Иначе
			ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка, , , текст);
		#КонецЕсли
			Продолжить;		
		КонецЕсли;
		
		адрес = выб.Адрес;
		ЗаполнитьКонтакты(Контакты, выб.Контрагент);
		
		типЦены = ТипыЦен.НайтиПоЗначению(выб.ТипЦен);
		индЦены = 0;
		Если типЦены = Неопределено Тогда
			индЦены = ТипыЦен.Количество();
			ТипыЦен.Добавить(выб.ТипЦен);
		Иначе
			индЦены = ТипыЦен.Индекс(типЦены);
		КонецЕсли;
		
		rs.AddNew();
		rs.Fields("name").value = СокрЛП(СтрЗаменить(выб.Контрагент.Наименование, "'", "`"));
		rs.Fields("id").value = код;
		rs.Fields("costype").value = индЦены;
		rs.Fields("address").value = СокрЛП(адрес);
		rs.Fields("userid").value = кодАгента;
		rs.Fields("latitude").value = выб.Контрагент.Широта;
		rs.Fields("longitude").value = выб.Контрагент.Долгота;
		rs.Fields("debt").value = выб.Договор.ДопустимаяСуммаЗадолженности;
		rs.Fields("delay").value = выб.Договор.ДопустимоеЧислоДнейЗадолженности;
		rs.Fields("codeTT").value = выб.Контрагент.КодТТ.Наименование;
		
		Для каждого Контакт из Контакты Цикл
			rscnt.AddNew();
			rscnt.Fields("name").value = Контакт.ФИО;
			rscnt.Fields("Org$id").value = код;
			rscnt.Fields("phone").value = Контакт.Телефон;
			rscnt.Fields("Org$userid").value = кодАгента;
		КонецЦикла;
		
		Если сч >= 1000 Тогда
			сч = 0;	
			rs.UpdateBatch();
			rscnt.UpdateBatch();
		КонецЕсли;
		сч = сч + 1;		
	КонецЦикла;
	
	rs.UpdateBatch();
	rs.Close();
	rs = NULL; 
	
	rscnt.UpdateBatch();
	rscnt.Close();
	rscnt = NULL; 
	
	ТекстЗапроса = "
	|BEGIN TRANSACTION Update_Org;
	| DELETE FROM Org WHERE userid in ('" + кодАгента + "');

	|
	| INSERT INTO Org$contacts(Org$id,Org$userid,name,phone,id) select Org$id,Org$userid,name,phone,'id' from #tbOrgCont;
	| drop table #tbOrg;
	| drop table #tbOrgCont;
	|COMMIT TRANSACTION Update_Org;
	|";
	
	ВыполнитьЗапрос(ADOСоединение, ТекстЗапроса);

Процедура ОчисткаМаршрута(Агенты)
	НастройкаНаполеон = ПолучитьНастройки();
	 
		Если Агенты.ПометкаУдаления Тогда
			Возврат;
		КонецЕсли;
		СтруктураПараметров = Новый Структура; 
		СтруктураПараметров.Вставить("ПапкаОбмена",НастройкаНаполеон.ПапкаОбмена);
		СтруктураПараметров.Вставить("Агент",Агенты.Агент);
		Док = ПоискПоследнегоМаршрута(Агенты.Агент);
		Если Док <> Неопределено Тогда
			ТЗМаршрут = Док.Маршрут.Выгрузить();
			ВремТЗ = ТЗМаршрут.Скопировать();
			ВремТЗ.Свернуть("Контрагент");
			Для Каждого СтрВремТЗ Из ВремТЗ Цикл
				Если НЕ КонтрагентПринадлежитАгенту(СтрВремТЗ.Контрагент,Агенты) Тогда
					УдалитьКонтрагентаИзТЗ(ТЗМаршрут,СтрВремТЗ.Контрагент);
				КонецЕсли;
			КонецЦикла;
			ЧистыйМаршрут = Документы.Маршруты.СоздатьДокумент();
			ЧистыйМаршрут.Дата = ТекущаяДата();
			ЧистыйМаршрут.Менеджер = Док.Менеджер;
			ЧистыйМаршрут.Маршрут.Загрузить(ТЗМаршрут);
			ЧистыйМаршрут.Записать(РежимЗаписиДокумента.Проведение);
			
			СтруктураПараметров.Вставить("Маршрут",ЧистыйМаршрут.Маршрут.Выгрузить());
		Иначе 
			Возврат;
		КонецЕсли;

			ВозвратыИзНакладных.Вставить(номерДок, данныеНакладной);
		КонецЕсли;
		
		текКол = стр.Количество * стр.ЕдиницаИзмерения.Коэффициент;
		кол_во = данныеНакладной.Получить(стр.Номенклатура.Код);
		Если кол_во = Неопределено Тогда
			кол_во = текКол;
		Иначе
			кол_во = кол_во + текКол;

Процедура СохранитьНакладную(ADOСоединение, rs, rsi, Документ, created, ВозвратыИзНакладных, кодАгента, КодКоманды)
	данныеВозврата = ВозвратыИзНакладных.Получить(Документ.Номер);
	
	rs.AddNew();
	код = Строка(Документ.Контрагент.Ссылка.УникальныйИдентификатор());
	номер = Документ.Номер;
	дата = ДатаВSQL(Документ.Дата);
	rs.Fields("id").Value = код;
	rs.Fields("division").Value = КодКоманды;
	rs.Fields("number").Value = номер;
	rs.Fields("date").Value = дата;
	rs.Fields("payDate").Value = ДатаВSQL(Документ.Дата + Документ.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 24 * 3600);
	Если created <> 0 Тогда
		rs.Fields("created").Value = created;	
	КонецЕсли;

	Для Каждого товар из Документ.Товары Цикл
		rsi.AddNew();
		rsi.Fields("Delivery$id").Value = код;
		rsi.Fields("Delivery$number").Value = номер;
		rsi.Fields("Delivery$date").Value = дата;
		rsi.Fields("id").Value = Строка(товар.Номенклатура.Ссылка.УникальныйИдентификатор());
		rq = товар.Количество * товар.Коэффициент;
		
		rsi.Fields("qty").Value = rq;
		сумма = ?(Документ.СуммаВключаетНДС, товар.Сумма, товар.Сумма + товар.СуммаНДС);
		rsi.Fields("sum").Value = сумма * Документ.КурсВзаиморасчетов;
		
		Если данныеВозврата <> Неопределено Тогда
			колВозврат = данныеВозврата.Получить(товар.Номенклатура.Код);
			Если колВозврат <> Неопределено Тогда
				rq = rq - колВозврат;
				Если rq < 0 Тогда
					rq = 0;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
		rsi.Fields("retQty").Value = rq;
	КонецЦикла;
	
	rs.UpdateBatch();
	rsi.UpdateBatch();
	
	ТекстЗапроса = "

	| DELETE FROM Delivery WHERE number = '" + номер + "' and [id] = '" + код + "' and [date] = " + дата + ";
	| INSERT INTO Delivery ([id],[number],sumd,created,date,payDate,[division])
	|    SELECT [id],[number],sumd,created,date,payDate,[division] FROM #tbDlv;
	| INSERT INTO Delivery$items ([id],sum,qty,retQty,Delivery$id,Delivery$number,Delivery$date)
	|    SELECT [id],sum,qty,retQty,Delivery$id,Delivery$number,Delivery$date FROM #tbDlvI;
	| delete from  #tbDlv;
	| delete from  #tbDlvI;
	|COMMIT TRANSACTION Update_Delivery;
	|";

Функция СоздатьРасходнуюНакладную(выб, userid, настройки, агенты)
	Док = Документы.РеализацияТоваровУслуг.СоздатьДокумент();
	Док.ВидОперации = Перечисления.ВидыОперацийРеализацияТоваров.ПродажаКомиссия;
	Док.ВидПередачи = Перечисления.ВидыПередачиТоваров.СоСклада;
	
	Док.Организация = Справочники.Организации.НайтиПоКоду("000000001");
	
	уид = Новый УникальныйИдентификатор(СокрЛП(выб.Fields.Item("id").Value));
	Док.Контрагент = Справочники.Контрагенты.ПолучитьСсылку(уид);
	Док.Грузополучатель = Док.Контрагент;
	
	ТекДата = ТекущаяДата();
	ТекВремя = ТекДата - НачалоДня( ТекДата );
	
	Док.Дата = НачалоДня(ДатаВ1С(выб.Fields.Item("date").Value)) + ТекВремя;
	
	Агент = НайтиАгента(userid, агенты);
	
	Если ТипЗнч(Агент) = Тип("СправочникСсылка.Пользователи") Тогда
		Док.Ответственный = Агент;
		Док.Менеджер = Агент;
		//Док.Подразделение = Агент.Подразделение;
	Иначе	
		Док.Ответственный = Агент.Агент;
		Док.Менеджер = Агент.Агент;
		Док.Подразделение = Агент.Подразделение;
		Агент = Агент.Агент;
	КонецЕсли;
	
	Док.ОтражатьВУправленческомУчете = Истина;
	
	ЗаполнитьДоговор(Док, Агент, настройки);
	
	Если Док.ДоговорКонтрагента.ВидВзаиморасчетов.Код = Справочники.ВидыВзаиморасчетов.НаличныйРасчет.Код Тогда

		Док.АвторасчетНДС = Ложь;

		Док.АвторасчетНДС = Истина;

	Док.КурсВзаиморасчетов = Курс.Курс;
	Док.КратностьВзаиморасчетов = Курс.Кратность;

Функция СоздатьПКО(Док, суммаДок, ДатаДок, Агент, КассаАгента, комментарий, выб, настройки)
	
	докПКО = Документы.ПриходныйКассовыйОрдер.СоздатьДокумент();
	
	Если Док = Неопределено Тогда
		уид = Новый УникальныйИдентификатор(СокрЛП(выб.Fields.Item("id").Value));
		ДокПКО.Контрагент = Справочники.Контрагенты.ПолучитьСсылку(уид);

		докПКО.Подразделение = Агент.Подразделение;
		//ЗаполнитьДоговор(докПКО, Агент.Агент, настройки);
	Иначе
		ДокПКО.Организация  = Док.Организация;
		ДокПКО.ВидОперации  = Перечисления.ВидыОперацийПКО.ОплатаПокупателя;
		докПКО.Заполнить(Док.Ссылка);
	КонецЕсли;
	
	
	докПКО.СуммаДокумента = суммаДок;
	//докПКО.пкЗаказ      = Заказ.Ссылка;
	ДокПКО.Дата			= ТекущаяДата();
	ДокПКО.Ответственный = Агент.Агент;

	ДокПКО.Касса = КассаАгента;
	
	докПКО.ОтражатьВБухгалтерскомУчете = Ложь;

	СтруктураКурсаВалютаДокумента = МодульВалютногоУчета.ПолучитьКурсВалюты(докПКО.ВалютаДокумента, докПКО.Дата);

Функция ПриемПКО(ADOСоединение, настройки, агенты, КодАгента = Неопределено, ответы = Неопределено)
	minDate = ДатаВSQL(НачалоДня(ТекущаяДата()) - 7 * 24 * 3600);
	
	ТекстЗапроса = 
	"select * from [Incass$items] oi where oi.[Incass$created] = ? and oi.[Incass$userid] = ?";
	
	команда = Новый COMОбъект("ADODB.Command"); 
	команда.ActiveConnection = ADOСоединение;
	команда.CommandText = ТекстЗапроса;
	команда.CommandType = 1; 
	команда.Parameters.Append (команда.CreateParameter("@created", 20, 1, 8));
	команда.Parameters.Append (команда.CreateParameter("@userid", 200, 1, 300));
	команда.Prepared = true;
	
	ЗагружалиДок = Новый Массив;
	
	Пока Истина Цикл
		выб = ПолучитьНовыеДокументы(ADOСоединение, "Incass", "Incass", minDate, КодАгента);
		Если выб = Неопределено или выб.EOF Тогда
			Возврат Ложь;
		КонецЕсли;
		Док = Неопределено;
		ПринятыеДок = Новый Массив;
		
		Пока Истина Цикл
			created = выб.Fields.Item("created").Value;
			userid = выб.Fields.Item("userid").Value;
			
			кодДок = Строка(created) + userid;
			Если ЗагружалиДок.Найти(кодДок) = Неопределено Тогда
				ЗагружалиДок.Добавить(кодДок);
				Прервать;
			КонецЕсли;
			
			выб.MoveNext();
			Если выб.EOF Тогда
				Возврат Истина;
			КонецЕсли;
		КонецЦикла;		
		
		
		команда.Parameters(0).value = created;
		команда.Parameters(1).value = userid;
		
		комментарий = СокрЛП(выб.Fields.Item("remark").Value);
		Агент = НайтиАгента(userid, агенты);
		КассаАгента = Агент.Касса;
		строки = команда.Execute();
		добавили = Ложь;
		
		
		//ГВВ 0204
		суммаДок = 0;
		реализация = Неопределено;
		
		ДатаДок = ТекущаяДата();
		ДокПКО = СоздатьПКО(реализация, суммаДок, ДатаДок, Агент, КассаАгента, комментарий, выб, настройки);
		Если ДокПКО <> Неопределено и Не добавили Тогда
			ЗаписатьКоординаты(ДокПКО, выб);
		КонецЕсли;
		ДокПКО.Оплачено = Истина;
		ДокПКО.ВалютаДокумента = Константы.ВалютаРегламентированногоУчета.Получить();
		ДокПКО.РасшифровкаПлатежа.Очистить();
		
		Пока Не строки.EOF Цикл
			докНомер = СокрЛП(строки.Fields.Item("number").Value); 
			докДата = ДатаВ1С(строки.Fields.Item("date").Value); 
			
			//Если реализация = Неопределено Тогда
				реализация = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(докНомер, докДата);
			//КонецЕсли;
			
			суммаДок = суммаДок + строки.Fields.Item("sum").Value;
			
			Если ЗначениеЗаполнено(реализация) Тогда
				//ДокПКО = Документы.ПриходныйКассовыйОрдер.СоздатьДокумент();
				НС = ДокПКО.РасшифровкаПлатежа.Добавить();
				НС.ДоговорКонтрагента = реализация.ДоговорКонтрагента;
				НС.ДокументРасчетовСКонтрагентом = реализация;
				НС.СуммаВзаиморасчетов = строки.Fields.Item("sum").Value;
				НС.КурсВзаиморасчетов = 1;
				НС.КратностьВзаиморасчетов = 1;
				НС.СуммаПлатежа = НС.СуммаВзаиморасчетов;
				НС.СтатьяДвиженияДенежныхСредств = НС.ДоговорКонтрагента.ОсновнаяСтатьяДвиженияДенежныхСредств;
			КонецЕсли;
			строки.MoveNext();
		КонецЦикла;
		ДокПКО.СуммаДокумента = суммаДок;
		Если ПопыткаЗаписатьДокумент(ДокПКО, РежимЗаписиДокумента.Запись) Тогда
			Попытка
				ДокПКО.Записать(РежимЗаписиДокумента.Проведение);
				добавили = Истина;
				ПринятыеДок.Добавить(Новый Структура("created,userid,number", created, userid, ДокПКО.Номер));
			Исключение
			#Если Клиент Тогда
				текст = ОписаниеОшибки();
				Сообщить(текст);
			#Иначе
				ЗаписьЖурналаРегистрации("ОбменНаполеон OrderHandler", УровеньЖурналаРегистрации.Ошибка, , , ОписаниеОшибки());
			#КонецЕсли
			ПринятыеДок.Добавить(Новый Структура("created,userid,number", created, userid, ""));
			КонецПопытки;
		Иначе
		КонецЕсли;
	
		ЗаписатьОбработанныеДокументы(ADOСоединение, ПринятыеДок, "Incass");

	Возврат Истина;

Функция ПриемПосещений(ADOСоединение, настройки, агенты, КодАгента = Неопределено)
	minDate = ДатаВSQL(НачалоДня(ТекущаяДата()) - 7 * 24 * 3600);
	
	ТекстЗапроса = 
	"select name from [Visit$items] oi where oi.[Visit$date] = ? and oi.[Visit$userid] = ?";
	
	команда = Новый COMОбъект("ADODB.Command"); 
	команда.ActiveConnection = ADOСоединение;
	команда.CommandText = ТекстЗапроса;
	команда.CommandType = 1; 
	команда.Parameters.Append (команда.CreateParameter("@date", 20, 1, 8));
	команда.Parameters.Append (команда.CreateParameter("@userid", 200, 1, 300));
	команда.Prepared = true;
	
	ЗагружалиДок = Новый Массив;
	Пока Истина Цикл
		выб = ПолучитьНовыеДокументы(ADOСоединение, "Visit", "Visit", minDate, КодАгента);
		Если выб = Неопределено или выб.EOF Тогда
			Возврат Истина;
		КонецЕсли;
		
		Док = Неопределено;
		ПринятыеДок = Новый Массив;
		
		Пока Истина Цикл
			created = выб.Fields.Item("created").Value;
			userid = выб.Fields.Item("userid").Value;
			
			кодДок = Строка(created) + userid;
			Если ЗагружалиДок.Найти(кодДок) = Неопределено Тогда
				ЗагружалиДок.Добавить(кодДок);
				Прервать;
			КонецЕсли;
			
			выб.MoveNext();
			Если выб.EOF Тогда
				Возврат Истина;
			КонецЕсли;
		КонецЦикла;		
		

		команда.Parameters(0).value = created;
		команда.Parameters(1).value = userid;
		
		комментарий = СокрЛП(выб.Fields.Item("remark").Value);
		Агент = НайтиАгента(userid, агенты);
		
		строки = команда.Execute();
		добавили = Ложь;
		
		док = Документы.кпк_ПосещениеТорговойТочки.СоздатьДокумент();
		док.Ответственый = Агент.Агент;
		док.Дата = ДатаВ1С(выб.Fields.Item("date").Value);
		уид = Новый УникальныйИдентификатор(СокрЛП(выб.Fields.Item("id").Value));
		док.Контрагент = Справочники.Контрагенты.ПолучитьСсылку(уид);
		док.КомментарийКпк = комментарий;		
		
		сч = 1;
		Пока Не строки.EOF Цикл
			path = СокрЛП(строки.Fields.Item("name").Value); 
			стр = док.СсылкиНаИзображения.Добавить();
			стр.ИмяФайла = Строка(сч);
			стр.ПутьКФайлу = настройки.ПапкаОбмена + "\agents\" + path;
			
			сч = сч + 1;
			
			строки.MoveNext();
		КонецЦикла;
		
		Если ПопыткаЗаписатьДокумент(Док, РежимЗаписиДокумента.Запись) Тогда
			ПринятыеДок.Добавить(Новый Структура("created,userid,number", created, userid, док.Номер));
			ЗаписатьКоординаты(док, выб);
		КонецЕсли;
		
		ЗаписатьОбработанныеДокументы(ADOСоединение, ПринятыеДок, "Visit");

	КонецЦикла;
