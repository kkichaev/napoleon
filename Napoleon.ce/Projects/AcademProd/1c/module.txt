Функция ПолучитьТаблицуКоманд()

   // Создадим пустую таблицу команд и колонки в ней
   Команды = Новый ТаблицаЗначений;

   // Как будет выглядеть описание печатной формы для пользователя
   Команды.Колонки.Добавить("Представление", Новый ОписаниеТипов("Строка")); 

   // Имя нашего макета, что бы могли отличить вызванную команду в обработке печати
   Команды.Колонки.Добавить("Идентификатор", Новый ОписаниеТипов("Строка"));

   // Тут задается, как должна вызваться команда обработки
   // Возможные варианты:
   // - ОткрытиеФормы - в этом случае в колонке идентификатор должно быть указано имя формы, которое должна будет открыть система
   // - ВызовКлиентскогоМетода - вызвать клиентскую экспортную процедуру из модуля формы обработки
   // - ВызовСерверногоМетода - вызвать серверную экспортную процедуру из модуля объекта обработки
   Команды.Колонки.Добавить("Использование", Новый ОписаниеТипов("Строка"));

   // Следующий параметр указывает, необходимо ли показывать оповещение при начале и завершению работы обработки. Не имеет смысла при открытии формы
   Команды.Колонки.Добавить("ПоказыватьОповещение", Новый ОписаниеТипов("Булево"));

   // Для печатной формы должен содержать строку ПечатьMXL 
   Команды.Колонки.Добавить("Модификатор", Новый ОписаниеТипов("Строка"));
   Возврат Команды;

Функция ПолучитьНоменклатуруСкидки(ИсточникДанных) Экспорт
	ТЗ = Новый ТаблицаЗначений;

	Настройки = Неопределено;
	Если ИсточникДанных.ВариантОтбораНоменклатуры = Перечисления.ВариантыОтбораНоменклатурыДляРасчетаСкидокНаценок.ОтборКомпоновкиДанных
		Тогда
		ХранилищеНастроекКомпоновкиДанных = ИсточникДанных.ХранилищеНастроекКомпоновкиДанных; // ХранилищеЗначения
		Настройки = ХранилищеНастроекКомпоновкиДанных.Получить();
		
		Если ИсточникДанных.ВариантОтбораНоменклатуры = Перечисления.ВариантыОтбораНоменклатурыДляРасчетаСкидокНаценок.ОтборКомпоновкиДанных Тогда
			Если Настройки <> Неопределено И ОбщегоНазначенияУТКлиентСервер.ЕстьРеквизитОбъекта(Настройки, "ОтборПоНоменклатуре") Тогда
				настройкиКомпоновки = Настройки.ОтборПоНоменклатуре;
				
				СКД = ПолучитьМакет("Товары");
				
				ГруппировкаКомпоновкиДанных1 = НастройкиКомпоновки.Структура.Добавить(Тип("ГруппировкаКомпоновкиДанных"));
				ГруппировкаКомпоновкиДанных1.Использование = Истина;
				ПолеГруппировкиКомпоновкиДанных = ГруппировкаКомпоновкиДанных1.ПоляГруппировки.Элементы.Добавить(Тип("ПолеГруппировкиКомпоновкиДанных"));
				ПолеГруппировкиКомпоновкиДанных.Использование = Истина;
				ПолеГруппировкиКомпоновкиДанных.Поле = Новый ПолеКомпоновкиДанных("Номенклатура");
				
				АвтоВыбранноеПолеКомпоновкиДанных = ГруппировкаКомпоновкиДанных1.Выбор.Элементы.Добавить(Тип("АвтоВыбранноеПолеКомпоновкиДанных"));
				АвтоВыбранноеПолеКомпоновкиДанных.Использование = Истина;
				
				ВыбранноеПолеКомпоновкиДанных = НастройкиКомпоновки.Выбор.Элементы.Добавить(Тип("ВыбранноеПолеКомпоновкиДанных"));
				ВыбранноеПолеКомпоновкиДанных.Использование = Истина;
				ВыбранноеПолеКомпоновкиДанных.Поле = Новый ПолеКомпоновкиДанных("Номенклатура");
				
				КомпоновщикМакета = Новый КомпоновщикМакетаКомпоновкиДанных;
				МакетКомпоновки = КомпоновщикМакета.Выполнить(СКД, настройкиКомпоновки,,,Тип("ГенераторМакетаКомпоновкиДанныхДляКоллекцииЗначений"));
				
				ПроцессорКомпоновки = Новый ПроцессорКомпоновкиДанных;
				ПроцессорКомпоновки.Инициализировать(МакетКомпоновки);
				
				ПроцессорВывода = Новый ПроцессорВыводаРезультатаКомпоновкиДанныхВКоллекциюЗначений;
				ПроцессорВывода.УстановитьОбъект(ТЗ);
				ПроцессорВывода.Вывести(ПроцессорКомпоновки);
			КонецЕсли;
		КонецЕсли;		
	КонецЕсли;
	
	
	Возврат ТЗ;

Функция ПолучитьСтруктуруПолейТЧ(СтруктураДействий)

	Перем СтруктураПараметровДействия;

	СтруктураПолейТЧ = Новый Структура;
	СтруктураПолейТЧ.Вставить("Номенклатура");

	Если СтруктураДействий.Свойство("ОбработатьШтрихкоды", СтруктураПараметровДействия) Тогда
		СтруктураПолейТЧ.Вставить("Характеристика");
		Если Не СтруктураПараметровДействия.НеИспользоватьУпаковки Тогда
			СтруктураПолейТЧ.Вставить("Упаковка");
		КонецЕсли;
		СтруктураПолейТЧ.Вставить("ХарактеристикиИспользуются");
		СтруктураПолейТЧ.Вставить(СтруктураПараметровДействия.ИмяКолонкиКоличество, 0);
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьСерийныеНомераПоВладельцу") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("КлючСвязиСерийныхНомеров");
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьХарактеристикуПоВладельцу") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("Характеристика");
		СтруктураПолейТЧ.Вставить("ХарактеристикиИспользуются");
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьЗаполнитьУпаковкуПоВладельцу") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("Упаковка");
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьСтатьюАналитикуРасходов") Тогда

Процедура ДобавитьТовар(Документ, ДокИст)
	Товар = Документ.Товары.Добавить();
	
	Номенклатура = ПолучитьОбъект(Справочники.Номенклатура, ДокИст.ID_I);
	Товар.Номенклатура = Номенклатура;
	
	//Если ЗначениеЗаполнено(Документ.Соглашение.ВидЦен) Тогда
	//	Товар.ВидЦены = Документ.Соглашение.ВидЦен;
	//Иначе
	//	Товар.ВидЦены = ТипЦеныПрайс;
	//КонецЕсли;
	
	Товар.КоличествоУпаковок = ДокИст.QTY;
	
	Данные = Новый Структура;	
	мета = Документ.Метаданные();
	Для Каждого рек из мета.ТабличныеЧасти.Товары.Реквизиты Цикл
		Данные.Вставить(рек.Имя);
	КонецЦикла;
	ЗаполнитьЗначенияСвойств(Данные, Товар);
	Данные.Вставить("ТипНоменклатуры");
	Данные.Вставить("ХарактеристикиИспользуются");
	Данные.Вставить("НомерСтроки", Документ.Товары.Количество());
	
	СтруктураПересчетаСуммы = ОбработкаТабличнойЧастиКлиентСервер.ПараметрыПересчетаСуммыНДСВСтрокеТЧ(Документ);
	
	СтруктураДействий = Новый Структура;
	
	Если Товар.Упаковка.Пустая() Тогда
		СтруктураДействий.Вставить("ПроверитьЗаполнитьУпаковкуПоВладельцу", Товар.Упаковка);
	КонецЕсли;
	СтруктураДействий.Вставить("ПересчитатьКоличествоЕдиниц");
	СтруктураДействий.Вставить("ПроверитьЗаполнитьСклад", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияСкладаВСтрокеТЧ(Документ, Ложь));
	
	СтруктураДействий.Вставить("ЗаполнитьУсловияПродаж"
		, ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияУсловийПродажВСтрокеТЧ(Документ));
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДС", Новый Структура("НалогообложениеНДС, Организация, Дата, ПоДатеОтгрузки"
		, Документ.НалогообложениеНДС
		, Документ.Организация
		, Документ.Дата
		, Истина));
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДСВозвратнойТары", Документ.ВернутьМногооборотнуюТару);
	СтруктураДействий.Вставить("ПересчитатьСуммуНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСуммуСНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСумму");
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомРучнойСкидки", Новый Структура("Очищать", Истина));
	СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомАвтоматическойСкидки", Новый Структура("Очищать", Истина));

	ПараметрыДокумента = Документы.ЗаказКлиента.ПараметрыДокументаДляДействийОбеспечения(Документ);

Функция СоздатьЗаказ(ДокИст, Агент)
	партнер = ПолучитьОбъект(Справочники.Партнеры, ДокИст.ID);

	Запрос = Новый Запрос;
	Запрос.Текст = 
	"ВЫБРАТЬ
	|	СоглашенияСКлиентами.Ссылка КАК Ссылка
	|ИЗ
	|	Справочник.СоглашенияСКлиентами КАК СоглашенияСКлиентами
	|ГДЕ
	|	СоглашенияСКлиентами.ПометкаУдаления = ЛОЖЬ
	|	И СоглашенияСКлиентами.Партнер = &Партнер"
	;
	Запрос.УстановитьПараметр("Партнер", Партнер);
	выб = Запрос.Выполнить().Выбрать();
	Если выб.Следующий() Тогда
		соглашение = выб.Ссылка;
	Иначе
		Запрос = Новый Запрос;
		Запрос.Текст = 
		"ВЫБРАТЬ
		|	СоглашенияСКлиентами.Ссылка КАК Ссылка
		|ИЗ
		|	Справочник.СоглашенияСКлиентами КАК СоглашенияСКлиентами
		|ГДЕ
		|	СоглашенияСКлиентами.ПометкаУдаления = ЛОЖЬ
		|	И СоглашенияСКлиентами.Типовое = ИСТИНА"
		;
		
		выб = Запрос.Выполнить().Выбрать();
		Если выб.Следующий() Тогда
			соглашение = выб.Ссылка;
		Иначе
			соглашение = Справочники.СоглашенияСКлиентами.ПустаяСсылка();
		КонецЕсли;
	КонецЕсли;
	
	Документ = Документы.ЗаказКлиента.СоздатьДокумент();
	Документ.Дата = ТекущаяДата();
	Документ.ДатаОтгрузки = ДокИст.DATE;
	Документ.НеОтгружатьЧастями = Истина;
	
	Документ.Менеджер = Агент;
	Документ.Автор = Агент;
	Документ.Подразделение = Документ.Менеджер.Подразделение;

Функция ЗагрузкаЗаявокАгента(временныйФайл, КодАгента, ПринятыеЗаказы, РезультатДок)
	Агент = ПолучитьОбъект(Справочники.Пользователи, КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ = Новый XBase(временныйФайл);
	
	Заказ.Кодировка = КодировкаXBase.OEM;
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ, ВидЦеныДокумента);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ, РезультатДок);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				//Сообщить("Повторный заказ " + значЗаказ);
			Иначе			
				Документ = СоздатьЗаказ(Заказ, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		ЗаписатьДокумент(Документ, ВидЦеныДокумента);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ, РезультатДок);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

Функция ЗаписатьОтгрузку(Отгрузки, Реализация, Долг, ПринятыеЗаказы)
	
	долгДок = Реализация.СуммаДокумента;
	Если долгДок > Долг Тогда
		долгДок = Долг;
	КонецЕсли;	
		
	кодЗаказа = "";
	Если ЗначениеЗаполнено(Реализация.ЗаказКлиента) Тогда
		номерЗаказа = ПринятыеЗаказы.НайтиПоЗначению(Реализация.ЗаказКлиента.Номер);
		Если номерЗаказа <> Неопределено Тогда
			кодЗаказа = номерЗаказа.Представление;
		КонецЕсли;
	КонецЕсли;	
	Код = КодОбъекта(Реализация.Партнер);
	
	ДатаПлатежа = Неопределено;
	Для Каждого пл Из Реализация.ЭтапыГрафикаОплаты	Цикл
		Если ДатаПлатежа = Неопределено или ДатаПлатежа < пл.ДатаПлатежа Тогда
			ДатаПлатежа = пл.ДатаПлатежа;
		КонецЕсли;
	КонецЦикла;
	Если ДатаПлатежа = Неопределено Тогда
		ДатаПлатежа = Реализация.Дата + Реализация.Договор.СрокОплаты * 24 * 3600;
	КонецЕсли;
	
	Для Каждого товар Из Реализация.Товары Цикл
		Отгрузки.Добавить();
		
		Отгрузки.ID = Код;
		Отгрузки.NUM = Реализация.Номер;
		Отгрузки.SUMD = долгДок;
		Отгрузки.DATE = Реализация.Дата;
		Отгрузки.PAYDATE = ДатаПлатежа;
		Отгрузки.CREATED = кодЗаказа;
		
		Отгрузки.ID_I = КодОбъекта(товар.Номенклатура);
		Отгрузки.QTY = товар.Количество;
		Отгрузки.SUM = товар.Сумма;
		
		Отгрузки.Записать();
	КонецЦикла;
	
	Возврат Долг - долгДок;
КонецФункции
