Функция ПолучитьСтруктуруПолейТЧ(СтруктураДействий)

	Перем СтруктураПараметровДействия;

	СтруктураПолейТЧ = Новый Структура;
	СтруктураПолейТЧ.Вставить("Номенклатура");

	Если СтруктураДействий.Свойство("ОбработатьШтрихкоды", СтруктураПараметровДействия) Тогда
		СтруктураПолейТЧ.Вставить("Характеристика");
		Если Не СтруктураПараметровДействия.НеИспользоватьУпаковки Тогда
			СтруктураПолейТЧ.Вставить("Упаковка");
		КонецЕсли;
		СтруктураПолейТЧ.Вставить("ХарактеристикиИспользуются");
		СтруктураПолейТЧ.Вставить(СтруктураПараметровДействия.ИмяКолонкиКоличество, 0);
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьСерийныеНомераПоВладельцу") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("КлючСвязиСерийныхНомеров");
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьХарактеристикуПоВладельцу") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("Характеристика");
		СтруктураПолейТЧ.Вставить("ХарактеристикиИспользуются");
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьЗаполнитьУпаковкуПоВладельцу") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("Упаковка");
	КонецЕсли;

	Если СтруктураДействий.Свойство("ПроверитьСтатьюАналитикуРасходов") Тогда
		СтруктураПолейТЧ.Вставить("ТипНоменклатуры");
		СтруктураПолейТЧ.Вставить("СтатьяРасходов");
		СтруктураПолейТЧ.Вставить("АналитикаРасходов");
	КонецЕсли;
	
	Если СтруктураДействий.Свойство("ПересчитатьКоличествоЕдиниц") Тогда
		СтруктураПолейТЧ.Вставить("Упаковка");
		СтруктураПолейТЧ.Вставить("КоличествоУпаковок", 0);
		СтруктураПолейТЧ.Вставить("Количество", 0);
	КонецЕсли;

	Если СтруктураДействий.Свойство("ЗаполнитьЦенуПродажи") Тогда
		СтруктураПолейТЧ.Вставить("Номенклатура");
		СтруктураПолейТЧ.Вставить("Характеристика");
		СтруктураПолейТЧ.Вставить("Упаковка");
		СтруктураПолейТЧ.Вставить("ВидЦены");
		СтруктураПолейТЧ.Вставить("Цена", 0);
	КонецЕсли;

	Если СтруктураДействий.Свойство("ЗаполнитьНоменклатуруПоНоменклатуреПоставщика") Тогда
		СтруктураПолейТЧ.Вставить("НоменклатураПоставщика");
		СтруктураПолейТЧ.Вставить("Номенклатура");

Функция ЗагрузкаЗаявокАгента(временныйФайл, КодАгента, ПринятыеЗаказы, РезультатДок)
	Кодировка = ?(КодировкаБазыANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ = Новый XBase(временныйФайл);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Заказ.Кодировка = Кодировка;
	
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ, ВидЦеныДокумента);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ, РезультатДок);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);

			КонецЕсли;
