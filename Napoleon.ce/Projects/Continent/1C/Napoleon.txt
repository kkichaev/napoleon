Процедура HandleOrderRequest(КодАгента) Экспорт
	ПапкаОбмена = "C:\Works\Napoleon\Projects\Continent\!\Exp\";
	
	// находим файл с перемещением на борт
	рез = НайтиФайлы(ПапкаОбмена, "ORD" + КодАгента + ".DBF");
	
	Для Каждого файл из Рез Цикл
		
		разделитель = Найти(ВРег(файл.Имя), ".DBF");
		КодАгента= Сред(файл.Имя, 4, разделитель-4);

		имяФайла = ПапкаОбмена + "THRD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, имяФайла);
		
		
		// Создаем файл ответа
		выхФайл = "OMA.DBF";
		выхДок = Новый XBase();
		// поле CREATED входного документа
		выхДок.Поля.Добавить("CREATED", "S", 20, 0);
		
		// номер документа 1с
		выхДок.Поля.Добавить("NUMBER","S", 100, 0);
		
		// Примечание 
		выхДок.Поля.Добавить("REMARK","S", 200, 0);
		
		// Код товара
		выхДок.Поля.Добавить("ID_I",   "S", 40, 0);
		
		// Количество
		выхДок.Поля.Добавить("QTY",   "N", 10, 3);
		
		выхДок.СоздатьФайл(выхФайл);		
		выхДок.Кодировка = КодировкаXBase.OEM;
		
		
		дбф = Новый XBase(имяФайла);
		ДБФ.Кодировка = КодировкаXBase.OEM;
		
		
		ДБФ.Первая();		
		Пока ДБФ.ВКонце() = 0 Цикл
			выхДок.Добавить();
			
			выхДок.CREATED = ДБФ.CREATED;
			выхДок.ID_I = ДБФ.ID_I;
			выхДок.NUMBER = "123";
			выхДок.REMARK = "Test";
			выхДок.QTY = ДБФ.QTY;
			
			выхДок.Записать();
			
			ДБФ.Следующая();
		КонецЦикла;
		
		
		дбф.ЗакрытьФайл();
		выхДок.ЗакрытьФайл();
		
		ПереместитьФайл(выхФайл, ПапкаОбмена + "OA" + КодАгента + ".DBF");
	КонецЦикла;
