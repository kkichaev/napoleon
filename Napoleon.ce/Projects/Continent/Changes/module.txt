Процедура ЗаписатьДокументВБазу(ДатаДок, ИмяФайла, уидДокумента)
	сервер = Новый COMОбъект("GRSoft.Server");
	
	IP = "127.0.0.1"; // адрес сервера
	Port = 8888; // порт сервера
	
	рез = сервер.Connect(IP, Port);
	Если рез = 0 Тогда
		Сообщить(сервер.ErrorMessage);
		Возврат;
	КонецЕсли;
	
	докКол = сервер.New("Attachment");
	док = докКол.New();
	
	файл = Новый Файл(ИмяФайла);
	
	док.id = уидДокумента;
	док.fileName = ИмяФайла;
	док.name = файл.Имя;
	
	сервер.Write(докКол);
КонецПроцедуры
