&НаКлиенте
Процедура УстановитьОтборПоАгенту(ТекДанные)
// Создаем структуру отбора
	Отбор = Новый Структура;
	Отбор.Вставить("Агент", ТекДанные);
	// Задаем фиксированную структуру отбора для таблицы формы
	Элементы.Контрагенты.ОтборСтрок = Новый ФиксированнаяСтруктура(Отбор);
	Элементы.Склады.ОтборСтрок = Новый ФиксированнаяСтруктура(Отбор);
КонецПроцедуры

Процедура АгентыПриАктивизацииСтроки(Элемент)
	текДанные = Элемент.ТекущиеДанные;
	Если текДанные <> Неопределено Тогда
		текДанные = текДанные["Агент"];
	КонецЕсли;
	УстановитьОтборПоАгенту(текДанные); 
КонецПроцедуры

&НаКлиенте
Процедура УдалитьДанныеАгента(агент)
	строки = Объект.Контрагенты.НайтиСтроки(Новый Структура("Агент", агент));
	Для каждого стр из строки Цикл
		Объект.Контрагенты.Удалить(стр);
	КонецЦикла;
	строки = Объект.Склады.НайтиСтроки(Новый Структура("Агент", агент));
	Для каждого стр из строки Цикл
		Объект.Склады.Удалить(стр);
	КонецЦикла;
КонецПроцедуры

&НаКлиенте
Процедура ДобавитьКонтрагента(Команда)
	Если Элементы.Агенты.ТекущиеДанные = Неопределено Тогда
		Возврат;
	КонецЕсли;
	
	ПараметрыФормы = Новый Структура("РежимВыбора",Истина);
	ПараметрыФормы.Вставить("МножественныйВыбор",Истина);
	
	ОбработкаВыбора = Новый ОписаниеОповещения("ПриВыбореКонтрагента", ЭтаФорма,"");
	ОткрытьФорму("Справочник.Партнеры.ФормаВыбора", ПараметрыФормы, ЭтаФорма,,,,ОбработкаВыбора);
КонецПроцедуры
