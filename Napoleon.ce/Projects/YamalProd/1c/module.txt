Процедура ВыгрузкаКонтрагентов(ТипыЦен)

	ЗаписатьАгентов();
		
	контакты = Неопределено;
	партнеры = Неопределено;
	
	кагенты = СоздатьБазуПартнеров("TORGS.DBF");
		
	тблКа = Контрагенты.ВыгрузитьКолонки();
	тблКа.Сортировать("Агент");
	
	выгр = Новый Массив;
	
	Для Каждого а из Агенты Цикл
		отбор = Новый Структура("Агент", а.Агент);
		партнеры = СоздатьБазуПартнеров("TO.DBF");
		Контакты = СоздатьБазуКонтактов();
		
		партнерыАгента = новый Массив;
		Для каждого стр из Контрагенты.НайтиСтроки(отбор) Цикл
			партнерыАгента.Добавить(стр.Контрагент);
			
			код = КодОбъекта(стр.Контрагент);
			видЦены = ТипЦеныПрайс;
			
			инд = ТипыЦен.Найти(видЦены);
			Если инд = Неопределено Тогда 
				инд = типыЦен.Количество();
				типыЦен.Добавить(видЦены);
			КонецЕсли;
			
			партнеры.Добавить();
			партнеры.ID = код;
			партнеры.NAME = стр.Контрагент.Наименование;
			партнеры.ADDRESS = АдресКонтрагента(стр.Контрагент);
			партнеры.COSTYPE = инд;
			партнеры.Записать();
			
			Если выгр.Найти(код) = Неопределено Тогда
				выгр.Добавить(код);
				кагенты.Добавить();
				кагенты.ID = код;
				кагенты.NAME = стр.Контрагент.Наименование;
				кагенты.ADDRESS = АдресКонтрагента(стр.Контрагент);
				кагенты.COSTYPE = инд;
				кагенты.Записать();
			КонецЕсли;
			
			списокКонтактов = ВыгрузитьКонтакты(стр.Контрагент);
			ЗаписатьКонтакты(контакты, списокКонтактов, код);
			
		КонецЦикла;

		ЗакрытьБазуПартнеров(партнеры, контакты, а.Агент);
		ВыгрузитьСегменты(партнерыАгента, а.Агент);
	КонецЦикла;
	
	кагенты.ЗакрытьФайл();
	ПереместитьФайл(ПапкаОбмена + "\TORGS.DBF", ПапкаОбмена + "\ORGS.DBF");

Функция СоздатьЗаказ(ДокИст, Агент)
	партнер = ПолучитьОбъект(Справочники.Партнеры, ДокИст.ID);

	Документ = Документы.РеализацияТоваровУслуг.СоздатьДокумент();
	//Документ.Дата = ТекущаяДата();
	Документ.Дата = ДокИст.DATE;
	//Документ.НеОтгружатьЧастями = Истина;
	
	Документ.Менеджер = Агент;
	Документ.Автор = Агент;
	Документ.Подразделение = Документ.Менеджер.Подразделение; 
	
	Запрос = Новый Запрос;
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	СоглашенияСКлиентами.Ссылка КАК Соглашение
		|ИЗ
		|	Справочник.СоглашенияСКлиентами КАК СоглашенияСКлиентами
		|ГДЕ
		|	СоглашенияСКлиентами.Партнер = &Партнер
		|	И СоглашенияСКлиентами.ПометкаУдаления = ЛОЖЬ"
		;
	Запрос.УстановитьПараметр("Партнер", партнер);
	выб = Запрос.Выполнить().Выбрать();
	Если выб.Следующий() Тогда
		соглашение = выб.Соглашение;
	Иначе
		соглашение = Справочники.СоглашенияСКлиентами.ПустаяСсылка();
	КонецЕсли;
			
	Документ.Соглашение = соглашение;
	Документ.ЗаполнитьУсловияПродажПоСоглашению();	

	Документ.Организация = соглашение.Организация;
	Документ.Контрагент = соглашение.Контрагент;
	Документ.Партнер = Партнер;
	Если СтрДлина(СокрЛП(ДокИст.ID_WH)) = 0 Тогда
		отбор = Новый Структура("Агент", Агент);
		Для каждого стр из склады.НайтиСтроки(отбор) Цикл
			Документ.Склад = стр.Склад.Ссылка;
			Прервать;
		КонецЦикла;
	Иначе
		уид = Новый УникальныйИдентификатор(ДокИст.ID_WH);
		Документ.Склад = Справочники.Склады.ПолучитьСсылку(уид);
	КонецЕсли;
	Документ.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.РеализацияКлиенту;
	Документ.СпособДоставки = Перечисления.СпособыДоставки.ДоКлиента;
	Документ.Статус = Перечисления.СтатусыЗаказовКлиентов.КОтгрузке;
	//Документ.Приоритет = Справочники.Приоритеты.ПолучитьПриоритетПоУмолчанию(Неопределено);

	двыб = Справочники.ДоговорыКонтрагентов.Выбрать(,Документ.Контрагент);
	Если двыб.Следующий() Тогда
		Документ.Договор = двыб.Ссылка;
	КонецЕсли;
	
	Документ.Комментарий = ДокИст.REMARK;
	Документ.АдресДоставки = ФормированиеПечатныхФорм.ПолучитьАдресИзКонтактнойИнформации(Документ.Партнер);

	Возврат Документ;

Функция ЗагрузкаЗаявокАгента(временныйФайл, КодАгента, ПринятыеЗаказы, РезультатДок)
	Агент = ПолучитьОбъект(Справочники.Пользователи, КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ = Новый XBase(временныйФайл);
	
	Заказ.Кодировка = КодировкаXBase.OEM;
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ, ВидЦеныДокумента);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ, РезультатДок);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				//Сообщить("Повторный заказ " + значЗаказ);
			Иначе			
				Документ = СоздатьЗаказ(Заказ, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		ЗаписатьДокумент(Документ, ВидЦеныДокумента);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ, РезультатДок);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

Процедура СохранитьОбработанныеДокументы(КодАгента, ПринятыеЗаказы, ТипДок = "")
	имяФайла = ПапкаОбмена + "\POD" + КодАгента + ".DBF";
	имяВр = ПапкаОбмена + "\TPOD.DBF";

	ДБФ = Новый XBase();
	файл = Новый Файл(имяФайла);
	
	Если Не файл.Существует() Тогда
		ДБФ.Поля.Добавить("CREATED", "S", 25, 0);	
		ДБФ.Поля.Добавить("TYPE", "S", 25, 0);	
		ДБФ.СоздатьФайл(имяВр);
	Иначе
		ПереместитьФайл(имяФайла, имяВр);
		ДБФ.ОткрытьФайл(имяВр);
	КонецЕсли;
	
	ДБФ.Кодировка = КодировкаXBase.OEM;
	Для Каждого стр из ПринятыеЗаказы Цикл
		ДБФ.Добавить();
		ДБФ.CREATED = стр.Значение;
		ДБФ.TYPE = ТипДок;
		ДБФ.Записать();

Функция ЗаписатьОтгрузку(Отгрузки, Реализация, Долг, ПринятыеЗаказы)
	
	долгДок = Реализация.СуммаДокумента;
	Если долгДок > Долг Тогда
		долгДок = Долг;
	КонецЕсли;	
		
	кодЗаказа = "";
	номерЗаказа = ПринятыеЗаказы.НайтиПоЗначению(Реализация.Номер);
	Если номерЗаказа <> Неопределено Тогда
		кодЗаказа = номерЗаказа.Представление;
	КонецЕсли;

	Код = КодОбъекта(Реализация.Партнер);
	
	ДатаПлатежа = Неопределено;
	Для Каждого пл Из Реализация.ЭтапыГрафикаОплаты	Цикл
		Если ДатаПлатежа = Неопределено или ДатаПлатежа < пл.ДатаПлатежа Тогда
			ДатаПлатежа = пл.ДатаПлатежа;
		КонецЕсли;
	КонецЦикла;
	Если ДатаПлатежа = Неопределено Тогда
		ДатаПлатежа = Реализация.Дата + Реализация.Договор.СрокОплаты * 24 * 3600;
	КонецЕсли;
	
	Для Каждого товар Из Реализация.Товары Цикл
		Отгрузки.Добавить();
		
		Отгрузки.ID = Код;
		Отгрузки.NUM = Реализация.Номер;
		Отгрузки.SUMD = долгДок;
		Отгрузки.DATE = Реализация.Дата;
		Отгрузки.PAYDATE = ДатаПлатежа;
		Отгрузки.CREATED = кодЗаказа;
		
		Отгрузки.ID_I = КодОбъекта(товар.Номенклатура);
		Отгрузки.QTY = товар.Количество;
		Отгрузки.SUM = товар.Сумма;
		
		Отгрузки.Записать();
	КонецЦикла;
	
	Возврат Долг - долгДок;
КонецФункции
