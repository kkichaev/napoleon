
&НаКлиенте
Процедура ПриОткрытии(Отказ)
	Элементы.Версия.Заголовок = "Версия: " + ВерсияМодуля();
	Элементы.Информация.Заголовок = ИнфоМодуля();
	
	ОбновитьТаблицыЗначений();
	
	Если ПараметрЗапуска = "ВыгрузитьПрайс" Тогда
		ВыгрузитьОстатки(Неопределено);
		ЗавершитьРаботуСистемы();	
	ИначеЕсли ПараметрЗапуска = "ЗагрузитьЗаявки" Тогда
		ЗагрузитьЗаявки(Неопределено);
		ЗавершитьРаботуСистемы();	
	ИначеЕсли ПараметрЗапуска = "ВыгрузитьДолги" Тогда
		ВыгрузитьДолги(Неопределено);
		ЗавершитьРаботуСистемы();	
	КонецЕсли;
КонецПроцедуры

&НаСервере
Процедура ОбновитьТаблицыЗначений()
	ОбновитьТаблицуЗначений(ВидыЦен, Справочники.ВидыЦен);
	ОбновитьТаблицуЗначений(Фирмы, Справочники.Организации);
	
	// из списка цен удалим которые на для продажи
	сч = 0;
	Пока сч < ВидыЦен.Количество() Цикл
		стр = ВидыЦен.Получить(сч);
		Если стр.Объект.ИспользоватьПриПродаже = Ложь Тогда
			ВидыЦен.Удалить(стр);
		Иначе
			сч = сч + 1;
		КонецЕсли;
	КонецЦикла;
	
КонецПроцедуры

&НаСервере
Процедура ОбновитьТаблицуЗначений(таблица, справочник)	
	// Проверим виды цен и удалим тех которых нет
	сч = 0;
	Пока сч < таблица.Количество() Цикл
		стр = таблица.Получить(сч);
		
		Код = "";
		
		Попытка
			Код = стр.Объект.Код;
		Исключение
		КонецПопытки;
		
		Если справочник.НайтиПоКоду(Код) = Ложь Тогда
			таблица.Удалить(стр);
		Иначе
			сч = сч + 1;
		КонецЕсли;
	КонецЦикла;
	
	// добавим новые виды цен
	Отбор = новый Структура;
	Выборка = справочник.Выбрать();
	Пока Выборка.Следующий() Цикл
		Элемент = Выборка.ПолучитьОбъект();
		
		Отбор.Вставить("Объект", Элемент.Ссылка);
		нстр = таблица.НайтиСтроки(Отбор);
		Если нстр.Количество() = 0 Тогда
			стр = таблица.Добавить();
			стр.Объект = Элемент.Ссылка;
		КонецЕсли;
	КонецЦикла;
КонецПроцедуры

	ИмяФайла = ПапкаОбмена + "\USERS.DBF";
	файл = Новый Файл(ИмяФайла);

	ЗаписатьДД(данные.ОбщийФайл, "ORGS.DBF");
	Для каждого каСтр из данные.Контрагенты Цикл
		стр = каСтр.Значение;
		ЗаписатьДД(стр.Контакты, "C" + стр.Код + ".DBF");
		ЗаписатьДД(стр.Контрагенты, "O" + стр.Код + ".DBF");

	данные = ВыгрузитьПапкиТоваров(КодировкаБазыANSI);
	ЗаписатьДД(данные, "FOLDERS.DBF");
	
	данные = ВыгрузитьТовары(КодировкаБазыANSI);	
	ЗаписатьДД(данные, "WAREHOUS.DBF");
	
	Сообщить("Завершено", СтатусСообщения.Информация);

		ТДД = ПолучитьИзВременногоХранилища(агентыДБФ);
	
		ИмяФайла = КаталогВременныхФайлов()+ "\TUSERS.DBF";
		ТДД.Записать(ИмяФайла);
	
		дбф = Новый XBase;
		дбф.ОткрытьФайл(ИмяФайла);
		//дбф.Кодировка = КодировкаXBase.OEM;
		дбф.Кодировка = кодировка;

		Пока ДБФ.ВКонце() = 0 Цикл

			стр.Логин = СокрЛП(ДБФ.LOGIN);
			стр.Пароль = СокрЛП(ДБФ.PASSWORD);
			ДБФ.Следующая();

	"ВЫБРАТЬ РАЗРЕШЕННЫЕ

	|	Товар";

	// выберем товар с ценами
	//
	Запрос = Новый Запрос;

			КонецЕсли;
			Если стр.Множитель >	 1 Тогда
				Товары.INPACK = стр.Множитель;
			Иначе
				Товары.INPACK = 1;

Процедура ЗаполнитьНоменклатуру(Документ, Товар)
	ТекущаяСтрока = Новый Структура;	
	мета = Документ.Метаданные();
	Для Каждого рек из мета.ТабличныеЧасти.Товары.Реквизиты Цикл
		ТекущаяСтрока.Вставить(рек.Имя);
	КонецЦикла;
	ЗаполнитьЗначенияСвойств(ТекущаяСтрока, Товар);
		
	ТекущаяСтрока.Вставить("ТипНоменклатуры", Перечисления.ТипыНоменклатуры.Товар);
	ТекущаяСтрока.Вставить("Артикул");
	ТекущаяСтрока.Вставить("ВариантОформленияПродажи");
	ТекущаяСтрока.Вставить("ХарактеристикиИспользуются");
	ТекущаяСтрока.Вставить("Содержание");
	
	СтруктураПересчетаСуммы = ОбработкаТабличнойЧастиКлиентСервер.ПараметрыПересчетаСуммыНДСВСтрокеТЧ(Документ);
	
	СтруктураДействий = Новый Структура;
	СтруктураДействий.Вставить("ПроверитьХарактеристикуПоВладельцу", ТекущаяСтрока.Характеристика);
	СтруктураДействий.Вставить("ПроверитьЗаполнитьУпаковкуПоВладельцу", ТекущаяСтрока.Упаковка);
	СтруктураДействий.Вставить("ПересчитатьКоличествоЕдиниц");
	СтруктураДействий.Вставить("ПроверитьЗаполнитьСклад", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияСкладаВСтрокеТЧ(Документ, Истина));
	Если ЗначениеЗаполнено(Документ.Соглашение) Тогда
		СтруктураДействий.Вставить("ЗаполнитьУсловияПродаж", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияУсловийПродажВСтрокеТЧ(Документ));
	Иначе
		СтруктураДействий.Вставить("ЗаполнитьЦенуПродажи", ОбработкаТабличнойЧастиКлиентСервер.ПолучитьСтруктуруЗаполненияЦеныВСтрокеТЧ(Документ));
	КонецЕсли;
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДС", 
	Новый Структура(
		"НалогообложениеНДС, Дата, ПоДатеОтгрузки, Организация",
		Документ.НалогообложениеНДС, Документ.Дата, Ложь, Документ.Организация));
	СтруктураДействий.Вставить("ЗаполнитьСтавкуНДСВозвратнойТары", Документ.ВернутьМногооборотнуюТару);
	СтруктураДействий.Вставить("ПересчитатьСуммуНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСуммуСНДС", СтруктураПересчетаСуммы);
	СтруктураДействий.Вставить("ПересчитатьСумму");

	Товар.КоличествоУпаковок = ДокИст.QTY;

	Товар.Склад = Документ.Склад;
	
	ЗаполнитьНоменклатуру(Документ, Товар);

				Товар.Цена = Товар.Цена * выб.Числитель;

		СтруктураДействий.Вставить("ПересчитатьСумму");

		СтруктураДействий.Вставить("ПересчитатьСуммуСУчетомАвтоматическойСкидки", Новый Структура("Очищать", Истина));

Функция ЗагрузкаЗаявокАгента(АдресЗаказ, КодАгента, Агенты, ПринятыеЗаказы,КодировкаANSI)
	Кодировка = ?(КодировкаANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(Агенты, КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ_имя = КаталогВременныхФайлов() + "TORD.dbf";
	ЗаказДД = ПолучитьИзВременногоХранилища(АдресЗаказ);
	ЗаказДД.Записать(Заказ_имя);
	Заказ = Новый XBase(Заказ_имя);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Заказ.Кодировка = Кодировка;
	
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ, ВидЦеныДокумента);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				Сообщить("Повторный заказ " + значЗаказ);
			Иначе			
				стр = Найти(Заказ.ID, Символ(9));
				код = Заказ.ID;
				кодСоглашения = "";
				Если стр <> 0 Тогда
					кодСоглашения =	Прав(Заказ.ID, СтрДлина(Заказ.ID) - стр);
					код = Лев(Заказ.ID, стр-1);
				КонецЕсли;
				Если ЗагружатьКакЗаказКлиента Тогда
					Документ = СоздатьЗаказ(Заказ, код, кодСоглашения, Агент);
				Иначе
					Документ = СоздатьРеализацию(Заказ, код, кодСоглашения, Агент);
				КонецЕсли;
				
				Если МожноМенятьЦену Тогда
					ВидЦеныДокумента = ПолучитьЦену(Заказ.CTYPE);
				ИначеЕсли Не ЦенаИзСоглашения Тогда
					ВидЦеныДокумента = ВидЦеныВДокументе;
				Иначе
					ВидЦеныДокумента = Неопределено;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ, ВидЦеныДокумента);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		ЗаписатьДокумент(Документ, ВидЦеныДокумента);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;

Функция ЗагрузкаИнкассацийАгента(АдресДок, КодАгента, агенты, ПринятыеДок, КодировкаANSI)
	Кодировка = ?(КодировкаANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = Справочники.Пользователи.НайтиПоНаименованию(НайтиАгентаПоКоду(Агенты, КодАгента));
	Документ = Неопределено;
	
	СчДок = 0;
	
	Док_имя = КаталогВременныхФайлов() + "TINC.dbf";
	ДокД = ПолучитьИзВременногоХранилища(АдресДок);
	ДокД.Записать(Док_имя);
	Док = Новый XBase(Док_имя);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Док.Кодировка = Кодировка;
	
	КодДок = "";
	ТекСум = 0;
	КодОрг = "";
	ТекДок = Неопределено;
	
	Док.Первая();		
	Пока Док.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Док.ORDFLAG <> 0 или ПустаяСтрока(Док.NUM_I) Тогда
			//
			// Запишем предыдущий документ
			//
			Если ТекДок <> Неопределено Тогда
				Если ТекСум > 0 Тогда
					ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
				КонецЕсли;
				Если ЗаписатьДокументСервер(ТекДок) Тогда
					СообщитьОДокументе(ТекДок);
					ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
					ЗаписатьДокументСервер(ТекДок, Истина);
				КонецЕсли;
			КонецЕсли;
			
			ТекДок = Неопределено;
			ТекСум = Док.SUM;
			КодДок = СокрЛП(Док.CREATED);
			значДок = ПринятыеДок.НайтиПоЗначению(КодДок);
			Если значДок <> Неопределено Тогда			
				Сообщить("Повторный документ " + значДок);
				ТекСум = 0;
			Иначе
				ТекДок = СоздатьИнкассацию(Док.ID, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если ТекДок <> Неопределено Тогда 
			КодОрг = Док.ID;
			Сумма = Док.SUM_I;
			ДокОсн = Документы.РеализацияТоваровУслуг.ПустаяСсылка();
			Если ПустаяСтрока(Док.NUM_I) Тогда
				Сумма = Док.SUM;
			Иначе
				ДокОсн = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Док.NUM_I, Док.DATE_I);
			КонецЕсли;
			
			ДобавитьДокументОснование(ТекДок, ДокОсн, Сумма);
			ТекСум = ТекСум - Сумма;
		КонецЕсли;	
		Док.Следующая();
	КонецЦикла;
	
	Если ТекДок <> Неопределено Тогда
		Если ТекСум > 0 Тогда
			ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
		КонецЕсли;
		Если ЗаписатьДокументСервер(ТекДок) Тогда
			СообщитьОДокументе(ТекДок);
			ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
			ЗаписатьДокументСервер(ТекДок, Истина);
		КонецЕсли;
	КонецЕсли;
	
	Док.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

			ЗаписатьДолг(Долги, кодТек, начДолг, ДатаС);
			кодТек = код;
			начДолг = Выборка.ДолгКонечныйОстаток;	
		КонецЕсли;
		Если выборка.Документ <> Null и накладные.НайтиПоЗначению(выборка.Документ.Номер) = Неопределено Тогда
			накладные.Добавить(выборка.Документ.Номер);
			
			долгДок = Выборка.Документ.СуммаВзаиморасчетов;
			Если долгДок > начДолг Тогда
				долгДок = начДолг;
			КонецЕсли;
			Если долгДок < 0 Тогда
				долгДок = 0;	
			КонецЕсли;
			ЗаписатьОтгрузку(Отгрузки, Выборка.Документ, долгДок, ПринятыеЗаказы);
			начДолг = начДолг - долгДок;
		КонецЕсли;
	КонецЦикла;
	
	ЗаписатьДолг(долги, кодТек, начДолг, ДатаС);
	Возврат ЗакрытьБазуДолгов(агент, долги, отгрузки);
КонецФункции
