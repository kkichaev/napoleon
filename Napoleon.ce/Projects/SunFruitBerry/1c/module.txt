// Функция формирует внешнюю печатную форму счета документа "Заказ покупателя"
// Параметры:
//   ЗаказСсылка - ДокументСсылка.ЗаказПокупателя - ссылка на документ для которого необходимо сформировать печатную форму
//
// Возвращаемое значение:
//	 Строка - имя файла сформированной печатной формы, Неопределено - если при формировании ПФ возникли ошибки
//
Функция СформироватьСчетНаОплату(ЗаказСсылка)  
	
	ИмяОбработки = "ВО_СформироватьПФСчетНаОплатуСКорректировками";
	ПутьКВременномуФайлу = КаталогВременныхФайлов() + ИмяОбработки + ".tmp";
	ОбрСсылка = Справочники.ВнешниеОбработки.НайтиПоНаименованию(ИмяОбработки, Истина);
	
	Если ОбрСсылка = Справочники.ВнешниеОбработки.ПустаяСсылка() Тогда
		ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка,,,"Не найдена внешняя обработка 'ВО_СформироватьПФСчетНаОплатуСКорректировками' в справочнике внешних обработок");
		Возврат Неопределено;
	КонецЕсли;
	
	Попытка
		ОбрСсылка.ХранилищеВнешнейОбработки.Получить().Записать(ПутьКВременномуФайлу);
		ОбрОбъект = ВнешниеОбработки.Создать(ПутьКВременномуФайлу, Ложь);
	Исключение
		ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка,,,"Ошибка при записи временного файла " + ПутьКВременномуФайлу);
		Возврат Неопределено;
	КонецПопытки;
	
	ОбрОбъект.СсылкаНаОбъект = ЗаказСсылка;
	
	Попытка
		ТабДокумент = ОбрОбъект.Печать(); 
	Исключение
		ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка,,, ОписаниеОшибки());
		Возврат Неопределено;
	КонецПопытки;
	
	ТабДокумент.Автомасштаб = Истина;
	
	ИмяФайлаПФ = КаталогВременныхФайлов()+"Счет №"+СокрЛП(ЗаказСсылка.Номер)+" от "+Формат(ЗаказСсылка.Дата,"ДФ=dd-MM-yyyy")+".XLS";
	
	Попытка
		ТабДокумент.Записать(ИмяФайлаПФ, ТипФайлаТабличногоДокумента.XLS);
	Исключение
		ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка,,, ОписаниеОшибки());
		Возврат Неопределено;
	КонецПопытки;
	
	Возврат ИмяФайлаПФ;
		
КонецФункции

// Функция выполняет рассылку на email

//   Получатели - Массив - элементы типа Структура:
//					Адрес - Строка, адрес email

Функция ОтправитьEmail(Получатели, ТемаПисьма, ТелоПисьма, Вложения=Неопределено)
	
	УчетнаяЗаписьЭПОтправиля = Справочники.УчетныеЗаписиЭлектроннойПочты.НайтиПоНаименованию("sunfruitberry@sunfruitberry.com");
		
	Если УчетнаяЗаписьЭПОтправиля = Неопределено ИЛИ УчетнаяЗаписьЭПОтправиля = Справочники.УчетныеЗаписиЭлектроннойПочты.ПустаяСсылка() Тогда
		ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка,,, "Не найдена учетная запись электронной почты sunfruitberry@sunfruitberry.com");
		Возврат Ложь;
	КонецЕсли;
	
	ПараметрыПисьма = Новый Структура;
	ПараметрыПисьма.Вставить("Кому", Получатели);
	ПараметрыПисьма.Вставить("Тема", ТемаПисьма);
	ПараметрыПисьма.Вставить("Тело", ТелоПисьма);	
	ПараметрыПисьма.Вставить("Важность", ВажностьИнтернетПочтовогоСообщения.Обычная);	
	
	Если Вложения <> Неопределено Тогда
		ПараметрыПисьма.Вставить("Вложения", Вложения);
	КонецЕсли;	
		
	Попытка
		ЭлектроннаяПочта.ОтправитьСообщение(УчетнаяЗаписьЭПОтправиля, ПараметрыПисьма);
		Возврат Истина;
	Исключение
		ЗаписьЖурналаРегистрации("ОбменНаполеон", УровеньЖурналаРегистрации.Ошибка,,,"Ошибка при отправке почты: " + ОписаниеОшибки());

	КонецПопытки;

		Вложения);

Процедура ВыгрузкаПапокТовара(ADOСоединение)
	ADOСоединение.Execute(
		"IF EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_SCHEMA = 'dbo' AND TABLE_NAME = '#tFolder')
		|BEGIN
		|	DROP table #tFolder;
		|END;
		| SET NOCOUNT ON;
		| select top 0 * into #tFolder from Folder; 
		| if @@trancount > 0 commit;", 1);
	
	rs  = Новый COMОбъект("ADODB.RecordSet");
	rs.CursorLocation = 3; // on client
	rs.Open("select * from #tFolder", ADOСоединение, 2, 4);

	"ВЫБРАТЬ
	|	спрНоменклатура.Код,

	|	спрНоменклатура.Наименование
	|ИЗ
	|	Справочник.Номенклатура КАК спрНоменклатура
	|ГДЕ
	|	НЕ спрНоменклатура.Наименование = """"
	|	И спрНоменклатура.ЭтоГруппа
	|	И НЕ спрНоменклатура.ПометкаУдаления";

	rs.Close();

			Возврат выб.Ссылка;
		КонецЕсли;

Функция СоздатьПроцедуруДобавитьАгента(ADOСоединение) Экспорт
	ТекстЗапроса = 
	"IF EXISTS (SELECT *  FROM   sysobjects WHERE  id = object_id(N'[dbo].[AddAgent]') and OBJECTPROPERTY(id, N'IsProcedure') = 1)
	|BEGIN
	|	DROP PROCEDURE [dbo].[AddAgent];
	|END;";
	ADOСоединение.Execute(ТекстЗапроса);
	
	ТекстЗапроса = 
	"CREATE procedure [AddAgent](@id NVARCHAR(300), @name NVARCHAR(300))
	|as
	|begin tran
	|update [Agents] with (serializable) set [name] = @name where [id] = @id
	|if @@rowcount = 0
	|begin
 	|	insert into [Agents] ([id], [name]) values (@id, @name)
	|end;
	|commit tran;";
	ADOСоединение.Execute(ТекстЗапроса);
	
	добавитьАгента = Новый COMОбъект("ADODB.Command"); 
	добавитьАгента.ActiveConnection = ADOСоединение;
	добавитьАгента.CommandText ="AddAgent";
	добавитьАгента.CommandType=4; 
	добавитьАгента.Parameters.Append (добавитьАгента.CreateParameter("@id", 200, 1, 300));
	добавитьАгента.Parameters.Append (добавитьАгента.CreateParameter("@name", 200, 1, 300));
	добавитьАгента.Prepared = true; 
	
	Возврат добавитьАгента;

Функция ЗаписатьКонтрагента(rs, rscnt, Контрагент, Адрес, АдресКод, кодАгента, Выгружен, ТипыЦен, email="", контактМенеджера = Ложь) Экспорт
	типЦены = 0;
	
	Если ТипыЦен <> Неопределено Тогда
		договор = НайтиДоговор(Контрагент);
		Если договор.Пустая() Или договор.ТипЦен = Неопределено Тогда
			Возврат "";
		КонецЕсли;
				//
		// ТипЦены для контрагента выгружаем как позиция цены в order.cfg
		//
		тц = договор.ТипЦен;
		Если ТипЗнч(тц) = Тип("СправочникСсылка.ТипыЦенНоменклатурыКонтрагентов") Тогда
			тц = тц.ТипЦеныНоменклатуры;	
		КонецЕсли;
		Если тц.Пустая() Тогда
			тц = ТипЦеныПрайс;
		КонецЕсли;
		
		типЦены = ТипыЦен.НайтиПоЗначению(тц);
		сч = 0;
		Если типЦены = Неопределено Тогда
			сч = ТипыЦен.Количество();
			ТипыЦен.Добавить(тц, "cost" + Строка(сч+1));
		Иначе
			сч = ТипыЦен.Индекс(типЦены);
		КонецЕсли;
		типЦены = сч;
	КонецЕсли;
	
	тАдрес = Адрес;
	КодКА = СокрЛП(Контрагент.Код);
	КодОрг = КодКА;
	Если Не ПустаяСтрока(АдресКод) Тогда
		КодОрг = КодОрг + Символ(9) + АдресКод;
	КонецЕсли;
	
	Если выгружен.НайтиПоЗначению(КодОрг) <> Неопределено Тогда
		Возврат КодОрг;
	КонецЕсли;
	выгружен.Добавить(КодОрг);
	
	rs.AddNew();

Процедура СохранитьЗаказ(ord, ordItems, Документ, АдресКод, текАгент, ADOСоединение)
	
	number = СокрЛП(Документ.Номер);
	userid = КодАгента(текАгент);
	
	КодОрг = СокрЛП(Документ.Контрагент.Код) + Символ(9);
	Если Не ПустаяСтрока(АдресКод) Тогда
		КодОрг = КодОрг + АдресКод;	
	КонецЕсли;
	
	ord.AddNew();
	ord.Fields("id").value = КодОрг;
	ord.Fields("userid").value = userid;
	датаСтр = Формат(((Документ.Дата - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	ord.Fields("created").value = датаСтр;
	
	ord.Fields("date").value = датаСтр;
	ord.Fields("docNumber").value = number;
	
	опл = Документ.Дата + Документ.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 24 * 3600;
	ord.Fields("payDate").value = Формат(((опл - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	ord.UpdateBatch();
	
	Для Каждого товар из Документ.Товары Цикл
		ordItems.AddNew();
		ordItems.Fields("id").value = товар.Номенклатура.Код;
		ordItems.Fields("qty").value = товар.Количество;
		ordItems.Fields("cost").value = товар.Цена * Документ.КурсВзаиморасчетов;
		ordItems.Fields("LoadedOrders$docNumber").value = number;
		
		ordItems.UpdateBatch();
	КонецЦикла;
	
	ТекстЗапроса = "
	|BEGIN TRANSACTION Update_Dlv;
	|
	|DELETE FROM LoadedOrders WHERE docNumber='" + number + "';
	|INSERT INTO LoadedOrders (id,created,date,payDate,userid,docNumber)
	|   select id,created,date,payDate,userid,docNumber FROM #tOrd;
	|INSERT INTO LoadedOrders$items (id, qty, cost, LoadedOrders$docNumber)
	|  SELECT id, qty, cost, LoadedOrders$docNumber FROM #tOrdItems;
	|COMMIT TRANSACTION Update_Dlv;
	|
	|delete from #tOrd;
	|delete from #tOrdItems;
	|if @@trancount > 0 commit;
	|";	

	
	ВыполнитьЗапрос(ADOСоединение, ТекстЗапроса);

Функция СохранитьНакладную(dlv, dlvItems, Документ, АдресКод, НомерЗаказа, userid, ADOСоединение)
	
	number = СокрЛП(Документ.Номер);
	
	КодКА = СокрЛП(Документ.Контрагент.Код);
	КодОрг = КодКА + Символ(9);
	Если Не ПустаяСтрока(АдресКод) Тогда
		КодОрг = КодОрг + АдресКод;	
	КонецЕсли;
	
	dlv.AddNew();
	dlv.Fields("id").value = КодОрг;
	dlv.Fields("ido").value = КодКА;
	dlv.Fields("number").value = number;
	dlv.Fields("userid").value = userid;
	dlv.Fields("created").value = НомерЗаказа;
	dlv.Fields("sumD").value = 0;
	dlv.Fields("dogovor").value = СокрЛП(Документ.ДоговорКонтрагента.Код);
	
	датаСтр = Формат(((Документ.Дата - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	dlv.Fields("date").value = датаСтр;
	
	опл = Документ.Дата + Документ.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 24 * 3600;
	dlv.Fields("payDate").value = Формат(((опл - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	dlv.UpdateBatch();
	
	докномер = СокрЛП(Документ.Номер);
	Для Каждого товар из Документ.Товары Цикл
		dlvItems.AddNew();
		dlvItems.Fields("id").value = товар.Номенклатура.Код;
		dlvItems.Fields("qty").value = товар.Количество;
		dlvItems.Fields("sum").value = товар.Сумма * Документ.КурсВзаиморасчетов;
		dlvItems.Fields("Delivery$id").value = КодОрг;
		dlvItems.Fields("Delivery$number").value =докномер;
		dlvItems.Fields("Delivery$userid").value = userid;
		
		dlvItems.UpdateBatch();
	КонецЦикла;
	
	Для Каждого товар из Документ.Возвраты Цикл
		dlvItems.AddNew();
		dlvItems.Fields("id").value = товар.Номенклатура.Код;
		dlvItems.Fields("qty").value = -товар.Количество;
		dlvItems.Fields("sum").value = товар.Сумма * Документ.КурсВзаиморасчетов;
		dlvItems.Fields("Delivery$id").value = КодОрг;
		dlvItems.Fields("Delivery$number").value =докномер;
		dlvItems.Fields("Delivery$userid").value = userid;
		
		dlvItems.UpdateBatch();
	КонецЦикла;
	
	ТекстЗапроса = "
	|BEGIN TRANSACTION Update_Dlv;
	|
	|DELETE FROM Delivery WHERE number='" + number + "' and ido='" + КодКА + "';
	|INSERT INTO Delivery (id,ido,number,sumD,created,date,payDate,userid,dogovor)
	|   select id,ido,number,sumD,created,date,payDate,userid,dogovor FROM #tDlv;
	|INSERT INTO Delivery$items (id, qty, sum, Delivery$id, Delivery$number, Delivery$userid)
	|  SELECT id, qty, sum, Delivery$id, Delivery$number, Delivery$userid FROM #tDlvItems;
	|COMMIT TRANSACTION Update_Dlv;
	|
	|delete from #tDlv;
	|delete from #tDlvItems;
	|if @@trancount > 0 commit;
	|";	

	
	ВыполнитьЗапрос(ADOСоединение, ТекстЗапроса);	
	
	Возврат 0;

			?(статусЗнач = 1, Перечисления.СтатусыЗадачМаршрутногоЛиста.Выполнено,
			?(статусЗнач = 2, Неопределено,
			?(статусЗнач = 3, Перечисления.СтатусыЗадачМаршрутногоЛиста.Отказ,
			?(статусЗнач = 8, Перечисления.СтатусыЗадачМаршрутногоЛиста.ВРаботе,
			?(статусЗнач = 9, Перечисления.СтатусыЗадачМаршрутногоЛиста.Возврат,
			Неопределено))))));
			//+++ Чумичев 24/04/20
			Если статусЗнач = 1 И ЗначениеЗаполнено(IdВозврата) Тогда			
				статус = Перечисления.СтатусыЗадачМаршрутногоЛиста.Возврат;			
			КонецЕсли;
			//--- Чумичев
		Для каждого докЗаказ из докМаршрутОбъект.Заказы Цикл
			Если докЗаказ.ЗаказПокупателя.Номер = текЗаказ Тогда
				//+++ Чумичев 24/04/20
				Если ЗначениеЗаполнено(НаборЗаписей.Fields("status_date").value) И НаборЗаписей.Fields("status_date").value > докЗаказ.ВремяФакт И НЕ докЗаказ.СтатусВыполнения = статус Тогда
					докЗаказ.Широта  = НаборЗаписей.Fields("latitude").value;
					докЗаказ.Долгота = НаборЗаписей.Fields("longitude").value;
					докЗаказ.ВремяФакт   = НаборЗаписей.Fields("status_date").value;
					докЗаказ.СтатусВыполнения = статус;
					докЗаказ.КомментарийТЧ = НаборЗаписей.Fields("remark").value;
					ВМаршрутныйЛистВнесеныИзменения = Истина;
				КонецЕсли;	
				//--- Чумичев
			КонецЕсли;
			КонецЦикла;		
		НаборЗаписей.MoveNext();
	КонецЦикла;	
	НаборЗаписей.Close();
	
	Если ВМаршрутныйЛистВнесеныИзменения И докМаршрутОбъект <> Неопределено Тогда
		ПроверитьИзменитьСтатусМаршрутногоЛиста(докМаршрутОбъект);//Чумичев 07/05/20
		докМаршрутОбъект.Записать();
	КонецЕсли;

Процедура ПроверитьИзменитьСтатусМаршрутногоЛиста(докМаршрут)
	
	Если докМаршрут.СтатусДокумента = Перечисления.СтатусыМаршрутногоЛиста.Закрыт Тогда
		Возврат;
	КонецЕсли; 
	
	ВсеЗаказыВыполнены = Истина;
	Для Каждого Заказ Из докМаршрут.Заказы Цикл
		Если НЕ Заказ.СтатусВыполнения = Перечисления.СтатусыЗадачМаршрутногоЛиста.Выполнено 
			И НЕ Заказ.СтатусВыполнения = Перечисления.СтатусыЗадачМаршрутногоЛиста.Возврат
			И НЕ Заказ.СтатусВыполнения = Перечисления.СтатусыЗадачМаршрутногоЛиста.Закрыто
			И НЕ Заказ.СтатусВыполнения = Перечисления.СтатусыЗадачМаршрутногоЛиста.Отказ Тогда
			ВсеЗаказыВыполнены = Ложь;
		КонецЕсли; 
	КонецЦикла; 
	
	Если ВсеЗаказыВыполнены Тогда
		докМаршрут.СтатусДокумента = Перечисления.СтатусыМаршрутногоЛиста.Закрыт;
	Иначе
		докМаршрут.СтатусДокумента = Перечисления.СтатусыМаршрутногоЛиста.ВРейсе;
	КонецЕсли;

// Функция устанавлиает значения реквизитов из стрктуры настроек
// Параметры:
//   СтруктураНастроек - тип Структура, содержит значения устанавливаемых настроек
//
// Возвращаемое значение:
//	 Тип Булево: Истина - установка значений настроек завершена успешно; Ложь - в противном случае
//
Функция ВосстановитьНастройкиИзСтуктуры(СтруктураНастроек)
	Если ТипЗнч(СтруктураНастроек) <> Тип("ХранилищеЗначения") Тогда
		Возврат Ложь;
	КонецЕсли;
	
	СтруктураНастроек = СтруктураНастроек.Получить();
	
	лЗначениеНастройки = Неопределено;
	СтруктураНастроек.Свойство("СтрокаПодключения", лЗначениеНастройки);
	Если лЗначениеНастройки <> Неопределено Тогда
		СтрокаПодключения = лЗначениеНастройки;
	КонецЕсли;
	
	лЗначениеНастройки = Неопределено;
	СтруктураНастроек.Свойство("СкладОстатков", лЗначениеНастройки);
	Если лЗначениеНастройки <> Неопределено Тогда
		СкладОстатков = лЗначениеНастройки;
	КонецЕсли;
	
	лЗначениеНастройки = Неопределено;
	СтруктураНастроек.Свойство("ИзменятьТипЦены", лЗначениеНастройки);
	Если лЗначениеНастройки <> Неопределено Тогда
		ИзменятьТипЦены = лЗначениеНастройки;
	КонецЕсли;
	
	лЗначениеНастройки = Неопределено;
	СтруктураНастроек.Свойство("ТипЦеныПрайс", лЗначениеНастройки);
	Если лЗначениеНастройки <> Неопределено Тогда
		ТипЦеныПрайс = лЗначениеНастройки;
	КонецЕсли;
	
	лЗначениеНастройки = Неопределено;
	СтруктураНастроек.Свойство("ГлубинаВыгрузкиДолгов", лЗначениеНастройки);
	Если лЗначениеНастройки <> Неопределено Тогда
		ГлубинаВыгрузкиДолгов = лЗначениеНастройки;
	КонецЕсли;
	
	Возврат Истина;
