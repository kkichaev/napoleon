Функция КодОбъекта(эл)

Процедура ПроверитьИнформациюПоМенеджеру(МенеджерDBF, Агент, префикс) 
	МенеджерDBF.Первая();
	пКод = КодОбъекта(Агент);
	
	паспорт = "";
	серия = "";
	выдан = "";
	датаВыдачи = "";
	имя = "";
	
	Если Не Агент.ФизЛицо.Пустая() Тогда
		
		имя = Агент.ФизЛицо.Наименование;
		
		Запрос = Новый Запрос();
		Запрос.Текст = 
		"ВЫБРАТЬ
		|	ПаспортныеДанныеФизЛиц.ДокументСерия,
		|	ПаспортныеДанныеФизЛиц.ДокументНомер,
		|	ПаспортныеДанныеФизЛиц.ДокументДатаВыдачи,
		|	ПаспортныеДанныеФизЛиц.ДокументКемВыдан
		|ИЗ
		|	РегистрСведений.ПаспортныеДанныеФизЛиц КАК ПаспортныеДанныеФизЛиц
		|ГДЕ
		|	ПаспортныеДанныеФизЛиц.ФизЛицо = &ФизЛицо";
		Запрос.Параметры.Вставить("ФизЛицо", Агент.ФизЛицо.Ссылка);
		выб = Запрос.Выполнить().Выбрать();
		Если выб.Следующий() Тогда
			паспорт = выб.ДокументНомер;
			серия = выб.ДокументСерия;
			выдан = выб.ДокументКемВыдан;
			датаВыдачи = выб.ДокументДатаВыдачи;
		КонецЕсли;
	КонецЕсли;
	
	Пока МенеджерDBF.ВКонце() = 0 Цикл
		Если СокрЛП(МенеджерDBF.ID) = пКод Тогда
			МенеджерDBF.NAME = Агент.Наименование;
			МенеджерDBF.FULLNAME = имя;
			МенеджерDBF.PFA = префикс;
			МенеджерDBF.SER = серия;
			МенеджерDBF.NUMBER = паспорт;
			МенеджерDBF.REGION = выдан;
			МенеджерDBF.DATA = датаВыдачи;
			
			МенеджерDBF.Записать();
			Возврат;
		КонецЕсли;
		МенеджерDBF.Следующая();
	КонецЦикла;
	
	МенеджерDBF.Добавить();
	МенеджерDBF.ID = пКод;
	МенеджерDBF.NAME = Агент.Наименование;
	МенеджерDBF.PFA = префикс;
	МенеджерDBF.SER = серия;
	МенеджерDBF.NUMBER = паспорт;
	МенеджерDBF.REGION = выдан;
	МенеджерDBF.DATA = датаВыдачи;
	МенеджерDBF.FULLNAME = имя;
	
	
	МенеджерDBF.Записать();

Процедура ЗаполнитьРасчетныеЦены(стрТовара, цены, ВыгрЦены)
	перем ЦенаБазовая;
	Дата = ТекущаяДата();
	
	Для сч = 0 По ВыгрЦены.Количество() - 1 Цикл
		
		ТипЦен = ВыгрЦены.Получить(сч);
		Если Не (ТипЦен.Рассчитывается и ЗначениеЗаполнено(ТипЦен.БазовыйТипЦен)) Тогда
			Продолжить;
		КонецЕсли;
		
		ЦенаБазовая = цены.Получить(ТипЦен.БазовыйТипЦен);
		Если ЦенаБазовая = Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		ПолученнаяЦена = 0;
		
		ВалютаТекущая = ТипЦен.ВалютаЦены;
		ВалютаБазовая = ТипЦен.БазовыйТипЦен.ВалютаЦены;
		
		Если ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоПроцентнойНаценкеНаБазовыйТип Тогда
			ПолученнаяЦена = ЦенаБазовая * (1 + ТипЦен.ПроцентСкидкиНаценки / 100);
			ВалютаТекущая = ВалютаБазовая;
		ИначеЕсли ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоВхождениюБазовойЦеныВДиапазон Тогда
			ПолученнаяЦена = Ценообразование.ПолучитьЦенуПоЦеновымДиапазонам(ТипЦен, Дата, ЦенаБазовая, ВалютаТекущая, ВалютаБазовая);
		КонецЕсли;
		
		ПолученнаяЦена = Ценообразование.ПересчитатьЦенуПриИзмененииВалюты(ПолученнаяЦена, ВалютаТекущая, ВалютаТекущая, 0, 1, Дата);
		ПолученнаяЦена = Ценообразование.ОкруглитьЦену(ПолученнаяЦена, ТипЦен.ПорядокОкругления, ТипЦен.ОкруглятьВБольшуюСторону);
		
		стрТовара.Цены.Вставить("COST" + Строка(сч+1), ПолученнаяЦена);

	КонецЦикла;
	
	цены.Очистить();

	|	Товары.ПометкаУдаления = ЛОЖЬ
	|	И Товары.ЭтоГруппа = ЛОЖЬ
	|	И Товары.Ссылка В ИЕРАРХИИ(&Товары)
	|
	|УПОРЯДОЧИТЬ ПО
	|	Товар";
	
	папки = ПапкиТоваров.ВыгрузитьКолонку("Папка");
	
	Запрос.Параметры.Вставить("ТипыЦен", ВыгрЦены);
	Запрос.Параметры.Вставить("Дата", ТекущаяДата());
	Запрос.Параметры.Вставить("СкладОстатков", СкладОстатков);
	Запрос.Параметры.Вставить("Товары", папки);
	
	базовыеЦены = Новый Соответствие;
	текТовар = Неопределено;
	стрТовара = Неопределено;
	рез = Запрос.Выполнить().Выгрузить(ОбходРезультатаЗапроса.Прямой);	
	Для Каждого стр из рез Цикл
		Если текТовар <> стр.Товар Тогда
			Если текТовар <> Неопределено Тогда
				ЗаполнитьРасчетныеЦены(стрТовара, базовыеЦены, ВыгрЦены);
			КонецЕсли;
			текТовар = стр.Товар;
			
			стрТовара = таблТоваров.Добавить();
			стрТовара.Товар = стр.Товар;
			стрТовара.Кол_во = стр.Кол_во;
			стрТовара.Цены = Новый Соответствие;
		КонецЕсли;
		
		Цена = стр.Цена;
		Если Цена <> 0 Тогда
			Если стр.ЕдиницаИзмерения.Коэффициент <> 0 Тогда
				Цена = Цена / стр.ЕдиницаИзмерения.Коэффициент;
			КонецЕсли;
			базовыеЦены.Вставить(стр.ТипЦены, Цена);
			
			типЦены = ВыгрЦены.Найти(стр.ТипЦены);		
			Если типЦены <> Неопределено Тогда
				стрТовара.Цены.Вставить("COST" + Строка(типЦены+1), Цена);
			КонецЕсли;
		КонецЕсли;
	КонецЦикла;

	Если текТовар <> Неопределено Тогда
		ЗаполнитьРасчетныеЦены(стрТовара, базовыеЦены, ВыгрЦены);
	КонецЕсли;
	
	
	Возврат таблТоваров;
КонецФункции

		складDBF.COUNTRY = текТовар.СтранаПроисхождения.Наименование;
		складDBF.CTRYCODE = текТовар.СтранаПроисхождения.Код;

		Для каждого кстр из ки Цикл
			Если Найти(кстр.Вид.Наименование, "Юридический") > 0 Тогда
				фмт.ADDRESS = ?(кстр.Представление <> "",

					УправлениеКонтактнойИнформацией.ПолучитьПредставлениеАдреса(кстр));
			КонецЕсли;

				фмт.PHONE = кстр.Представление;
			КонецЕсли;

	Отбор.СтруктурнаяЕдиница = Фирма.Ссылка;

		фмт.CERT = "свидетельство " + Фирма.СвидетельствоСерияНомер + " от " +

Функция СоздатьНакладную(Заказ, Агент)
	Накладная = Документы.РеализацияТоваровУслуг.СоздатьДокумент();
	
	Накладная.Ответственный = 
		УправлениеПользователями.ПолучитьЗначениеПоУмолчанию(глЗначениеПеременной("глТекущийПользователь"), "ОсновнойОтветственный");
	
	Накладная.Дата = Заказ.DATE;
	Накладная.Номер = Заказ.NUMBER;
	Накладная.ВидОперации = Перечисления.ВидыОперацийРеализацияТоваров.ПродажаКомиссия;
		
	тт = ПолучитьОбъект(Справочники.ТорговыеТочки, Заказ.ID);
	контрагент = тт.Владелец;
	
	Накладная.Грузополучатель = контрагент;
	Накладная.Контрагент = контрагент;
	Накладная.ДоговорКонтрагента = Контрагент.ОсновнойДоговорКонтрагента;
	Накладная.Организация = Накладная.ДоговорКонтрагента.Организация;

	Накладная.БанковскийСчетОрганизации = Накладная.Организация.ОсновнойБанковскийСчет;
	Накладная.ОтражатьВУправленческомУчете = Истина;
	
	Накладная.Комментарий = Накладная.Комментарий + " " + Заказ.REMARK;
	
	Накладная.ТипЦен = Накладная.ДоговорКонтрагента.ТипЦен;
	Накладная.ВалютаДокумента = Накладная.ДоговорКонтрагента.ВалютаВзаиморасчетов;
	
	СтруктураКурсаДокумента = МодульВалютногоУчета.ПолучитьКурсВалюты(Накладная.ВалютаДокумента, Накладная.Дата);
	Накладная.КратностьВзаиморасчетов = СтруктураКурсаДокумента.Кратность;
	Накладная.КурсВзаиморасчетов = СтруктураКурсаДокумента.Курс;
	
	Если УчитыватьНДС(Накладная.Дата, Накладная.Организация) Тогда
		Накладная.УчитыватьНДС = Истина;
		Накладная.СуммаВключаетНДС = Истина;
	Иначе
		Накладная.УчитыватьНДС = Ложь;
		Накладная.СуммаВключаетНДС = Ложь;
	КонецЕсли;

	
	Накладная.ВидПередачи = Перечисления.ВидыПередачиТоваров.СоСклада;		
	
	Накладная.Склад = СкладАгента(Агент);
	
	Возврат Накладная;

	Накладная.УчитыватьНДС = УчитыватьНДС(Накладная.Дата, Накладная.Организация);
	Накладная.СуммаВключаетНДС = Истина;

	контрагент = тт.Владелец;

	КонецЕсли;

	Накладная.КурсВзаиморасчетов = Курс.Курс;

Процедура ПриемЗаявок()
	Перем Накладная, агент;
	
	Рез = НайтиФайлы(ПапкаОбмена, "ORD*.DBF");
		
	СчНакл = 0;
	
	Для Каждого файл из Рез Цикл
		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");

		КодСотр = Сред(имяФайла, 4, разделитель-4);
		Агент = ПолучитьОбъект(Справочники.Пользователи, КодСотр);
				
		временныйФайл = ПапкаОбмена + "\TORD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		Заказ = Новый XBase(временныйФайл);
		Заказ.Кодировка = КодировкаXBase.OEM;
		ОбработанныеЗаказы = ФайлОбработанныхДокументов(КодСотр);
		
		ЗаписатьЗаказ = 0;
		ПринятыеЗаявки = ЗагрузитьПринятыеДокументы(КодСотр, Истина);
		Ном = "";
		
		Заказ.Первая();		
		Пока Заказ.ВКонце() = 0 Цикл
			//
			// Проверим признак начала заявки
			//
			Если Заказ.ORDFLAG <> 0 Тогда
				//
				// началась новая заявка, если надо записать заявка, запишем ее.
				//
				Если ЗаписатьЗаказ = 1 Тогда
					ЗаписатьЗаявку(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
					ЗаписатьЗаказ = 0;
				КонецЕсли;
				
				//
				// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
				//
				пз = ПринятыеЗаявки.НайтиПоЗначению(СокрЛП(Заказ.CREATED));
				Если пз <> Неопределено Тогда
					//
					// пропускаем этот заказ
					//
					ЗаписатьЗаказ = 0;
					Заказ.Следующая();
					СообщитьИнт("Повторная заявка - накладная № " + пз.Представление);
					Продолжить;
				КонецЕсли;
				
				//
				// создаем новую заявку
				//
				Накладная = СоздатьЗаказ(Заказ, Агент);
				ЗаписатьЗаказ = 1;
				Ном = СокрЛП(Заказ.CREATED);
				СчНакл = СчНакл + 1;
			КонецЕсли;
			
			Если ЗаписатьЗаказ = 0 Тогда
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
						
			номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);

			товар = Накладная.Товары.Добавить();
			товар.Номенклатура = номенклатура.Ссылка;
			//itrunet 20.05.2015 >>>
			//Если заполнять, то товар ставиться в резерв. Т.к. нет товара на складах - закомментировано
			//товар.Размещение = Накладная.СкладГруппа;
			//itrunet 20.05.2015 <<<
			
			Если ТипЗнч(Накладная) = Тип("ДокументСсылка.ЗаказПокупателя") Тогда
				Накладная.ПриИзмененииНоменклатурыТоваров(товар);
			Иначе
				СтрокаТабличнойЧасти = товар;
				ОбработкаТабличныхЧастей.ЗаполнитьЕдиницуЦенуПродажиТабЧасти(СтрокаТабличнойЧасти, Накладная,
					Накладная.ВалютаДокумента);
			КонецЕсли;
			
			коэф = товар.ЕдиницаИзмерения.Коэффициент;
			Если коэф = 0 Тогда
				коэф = 1;
			КонецЕсли;
			товар.Количество = Заказ.QTY / коэф;
			товар.Коэффициент = коэф;
			
			ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Накладная);
			
			Заказ.Следующая();
		КонецЦикла;
		
		Если ЗаписатьЗаказ = 1 Тогда
			ЗаписатьЗаявку(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
		КонецЕсли;
		
		СохранитьПринятыеДокументы(КодСотр, ПринятыеЗаявки);
		Заказ.ЗакрытьФайл();
		ЗакрытьФайлОбработанныхЗаявок(ОбработанныеЗаказы, КодСотр);
		
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
	СообщитьИнт("Всего заявок " + СчНакл);
КонецПроцедуры

Функция ПриемФайлаРеализации(имяФайла, КодСотр)
	СчНакл = 0;
	Агент = ПолучитьОбъект(Справочники.Пользователи, КодСотр);
	
	врФайл = ПапкаОбмена + "\TSLS.DBF";
	ПереместитьФайл(имяФайла, врФайл);
	Заказ = Новый XBase(врФайл);
	Заказ.Кодировка = КодировкаXBase.OEM;
	ОбработанныеЗаказы = ФайлОбработанныхДокументов(КодСотр);
	ПринятыеЗаявки = ЗагрузитьПринятыеДокументы(КодСотр, Истина, "Sales");
	
	ЗаписатьДокумент = 0;
	Накладная = Неопределено;
	Ном = "";
	текПоз = 1;
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			Если ЗаписатьДокумент = 1 Тогда
				ЗаписатьРеализацию(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);				
				ЗаписатьДокумент = 0;
			КонецЕсли;
			текПоз = Заказ.НомерЗаписи();
			
			Док = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Заказ.NUMBER, Заказ.DATE);
			Если Не Док.Пустая() Тогда
				ЗаписатьДокумент = 0;
				СообщитьИнт("Повторная накладная № " + Заказ.NUMBER);
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
			
			//
			// создаем новую заявку
			//
			ЗаписатьДокумент = 1;
			Ном = СокрЛП(Заказ.CREATED);
			СчНакл = СчНакл + 1;
			
			Накладная = СоздатьНакладную(Заказ, Агент);
		КонецЕсли;
		
		Если ЗаписатьДокумент = 0 Тогда
			Заказ.Следующая();
			Продолжить;
		КонецЕсли;
		
		номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);
		товар = Накладная.Товары.Добавить();
		товар.Номенклатура = номенклатура.Ссылка;
		
		
		товар.ЕдиницаИзмерения = номенклатура.ЕдиницаХраненияОстатков;
		товар.Количество = Заказ.QTY;
		товар.Коэффициент = 1;
		ОбработкаТабличныхЧастей.ЗаполнитьЦенуПродажиТабЧасти(товар, Накладная, Накладная.ВалютаДокумента);
		
		товар.ЕдиницаИзмеренияМест = товар.ЕдиницаИзмерения;
		Если Заказ.PACK <> 0 Тогда
			едВыб = НайтиУпаковку(номенклатура);
			Если едВыб <> Неопределено Тогда
				товар.ЕдиницаИзмеренияМест = едВыб.Ссылка;
			КонецЕсли;
		КонецЕсли;
		
		ОбработкаТабличныхЧастей.РассчитатьКоличествоМестТабЧасти(товар, Накладная);
				
		товар.СпособСписанияОстаткаТоваров = Перечисления.СпособыСписанияОстаткаТоваров.СоСклада;
		товар.СтавкаНДС = номенклатура.СтавкаНДС;
		товар.Сумма = Заказ.SUM_I;
		Если Накладная.УчитыватьНДС Тогда
			товар.СуммаНДС = Заказ.TAX_I;
		Иначе
			товар.СуммаНДС = 0;
		КонецЕсли;
		//товар.СуммаНДС = Заказ.TAX_I;
		товар.ХарактеристикаНоменклатуры = Неопределено;
		Если товар.Количество > 0 Тогда
			товар.Цена = товар.Сумма / товар.Количество;
		Иначе
			товар.Цена = 0;
		КонецЕсли;
		товар.Склад = Накладная.Склад;
		товар.КлючСтроки = УправлениеЗапасами.ПолучитьНовыйКлючСтроки(Накладная);
		товар.КлючСвязи  = 0;

		Заказ.Следующая();
	КонецЦикла;
	
	Если ЗаписатьДокумент = 1 Тогда
		ЗаписатьРеализацию(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	УдалитьФайлы(врФайл);
	СохранитьПринятыеДокументы(КодСотр, ПринятыеЗаявки, "Sales");
	
	Возврат СчНакл;
КонецФункции

				товар.Номенклатура = номенклатура.Ссылка;
				товар.Коэффициент = номенклатура.ЕдиницаДляОтчетов.Коэффициент;
				товар.Количество = Заказ.QTY / номенклатура.ЕдиницаДляОтчетов.Коэффициент;
				товар.ЕдиницаИзмерения = номенклатура.ЕдиницаДляОтчетов;
				
				товар.ЕдиницаИзмеренияМест = номенклатура.ЕдиницаИзмеренияМест;
				Если Не товар.ЕдиницаИзмеренияМест.Пустая() Тогда
					товар.КоличествоМест = 
						Окр(товар.Количество / номенклатура.ЕдиницаИзмеренияМест.Коэффициент, 4, РежимОкругления.Окр15как20);
				КонецЕсли;
					
				товар.Цена = Ценообразование.ПолучитьЦенуКонтрагента(товар.Номенклатура,
					Справочники.ХарактеристикиНоменклатуры.ПустаяСсылка(),
					Накладная.Контрагент,
					Справочники.ТипыЦенНоменклатурыКонтрагентов.НайтиПоКоду("УТ0000001"));
					
				товар.Сумма = товар.Количество * товар.Цена;
			КонецЕсли;
			Заказ.Следующая();
		КонецЦикла;
		
		Если ЗаписатьДокумент = 1 Тогда
			ЗаписатьПоступление(Накладная, ОбработанныеЗаказы, Ном);
		КонецЕсли;
		
		Заказ.ЗакрытьФайл();
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
	СообщитьИнт("Всего заявок на борт (ПеремещениеТоваров): " + СчНакл);
КонецПроцедуры
