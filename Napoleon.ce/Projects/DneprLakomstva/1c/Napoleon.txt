Процедура ЗакрытьФайлыДолгов(ПапкаОбмена, Долги, ДолгиПоНакладным, Накладные, ТекНакладные, ТекАгент)

	код = КодАгента(ТекАгент);
	
	Если Долги <> Неопределено Тогда

		ДолгиПоНакладным.дбф.ЗакрытьФайл();
		ПереместитьФайл(ДолгиПоНакладным.имяФайла, ДолгиПоНакладным.имяДБФ);// ПапкаОбмена + "\BLDV" + СокрЛП(код) + ".DBF");  
	КонецЕсли;
	
	Если Накладные <> Неопределено Тогда
		Накладные.ЗакрытьФайл();
		ПереместитьФайл(ПапкаОбмена + "\TDLV.DBF", ПапкаОбмена + "\DLV" + СокрЛП(код) + ".DBF");  
	КонецЕсли;
	
	Если ТекНакладные <> Неопределено Тогда
		ТекНакладные.ЗакрытьФайл();
		ПереместитьФайл(ПапкаОбмена + "\TDDV.DBF", ПапкаОбмена + "\DDV" + СокрЛП(код) + ".DBF");  
	КонецЕсли;

//
// Ищем в файле USERS.DBF запись о торговом агенте
//
Процедура ПроверитьИнформациюПоМенеджеру(МенеджерDBF, Агент) 
	МенеджерDBF.Первая();
	пКод = КодАгента(Агент);
	Пока МенеджерDBF.ВКонце() = 0 Цикл
		Если СокрЛП(МенеджерDBF.ID) = пКод Тогда
			МенеджерDBF.NAME = Агент.Наименование;
			МенеджерDBF.Записать();
			Возврат;
		КонецЕсли;
		МенеджерDBF.Следующая();
	КонецЦикла;
	
	МенеджерDBF.Добавить();
	МенеджерDBF.ID = пКод;
	МенеджерDBF.NAME = Агент.Наименование;
	
	//
	// В качестве пароля и логина для КПК назначим количество записей + 1. Так для первого пользователя
	// логин и пароль будет 1/1, для второго 2/2 и т.д.
	//
	МенеджерDBF.PASSWORD = МенеджерDBF.КоличествоЗаписей() + 1;
	МенеджерDBF.LOGIN = МенеджерDBF.КоличествоЗаписей() + 1;
	
	МенеджерDBF.Записать();
КонецПроцедуры

Процедура ЗаполнитьРасчетныеЦены(стрТовара, цены, ВыгрЦены)
	перем ЦенаБазовая;
	Дата = ТекущаяДата();
	
	Для каждого стр из ВыгрЦены Цикл
		
		ТипЦен = стр.Значение;
		Если Не (ТипЦен.Рассчитывается и ЗначениеЗаполнено(ТипЦен.БазовыйТипЦен)) Тогда
			Продолжить;
		КонецЕсли;
		
		ЦенаБазовая = цены.Получить(ТипЦен.БазовыйТипЦен);
		Если ЦенаБазовая = Неопределено Тогда
			Продолжить;
		КонецЕсли;
		
		ПолученнаяЦена = 0;
		
		ВалютаТекущая = ТипЦен.ВалютаЦены;
		ВалютаБазовая = ТипЦен.БазовыйТипЦен.ВалютаЦены;
		
		Если ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоПроцентнойНаценкеНаБазовыйТип Тогда
			ПолученнаяЦена = ЦенаБазовая * (1 + стр.Значение.ПроцентСкидкиНаценки / 100);
			ВалютаТекущая = ВалютаБазовая;
		ИначеЕсли ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоВхождениюБазовойЦеныВДиапазон Тогда
			ПолученнаяЦена = Ценообразование.ПолучитьЦенуПоЦеновымДиапазонам(ТипЦен, Дата, ЦенаБазовая, ВалютаТекущая, ВалютаБазовая);
		КонецЕсли;
		
		ПолученнаяЦена = Ценообразование.ПересчитатьЦенуПриИзмененииВалюты(ПолученнаяЦена, ВалютаТекущая, ВалютаТекущая, 0, 1, Дата);
		ПолученнаяЦена = Ценообразование.ОкруглитьЦену(ПолученнаяЦена, ТипЦен.ПорядокОкругления, ТипЦен.ОкруглятьВБольшуюСторону);
		
		стрТовара.Цены.Вставить(стр.Представление, ПолученнаяЦена);

	КонецЦикла;
	
	цены.Очистить();

//

Функция ВыгрузкаТовара(ПапкаОбмена, ТипыЦен, стрАгент)
	Перем цена, едЦены;
	
	СообщитьИнт(ПапкаОбмена, "Выгружаем информацию по номенклатуре...");
	    
	//
	// сформируем список полей цен
	//
	ВыгрЦены = Новый СписокЗначений;
	Для каждого стр из ТипыЦен Цикл
		ВыгрЦены.Добавить(стр.Значение, стр.Представление);
	КонецЦикла;
	
	товары = ВыгрузитьТовары(ВыгрЦены);
	
	// Выгрузим общий прайс
	ВыгрузитьТоварыАгента(ПапкаОбмена, стрАгент, товары, ВыгрЦены);
	ВыгрузкаПапокТовара(ПапкаОбмена);
	
	возврат товары;

Процедура ВыгрузитьТоварыАгента(ПапкаОбмена, СтрАгент, товары, ВыгрЦены)
	
	файлТовара = ПапкаОбмена + "\TWH.DBF";
	Остатки = РегистрыНакопления.ТоварыНаСкладах;
	складDBF = ФайлТоваров(файлТовара, ВыгрЦены);
					
	Для Каждого стр из Товары Цикл
		текТовар = стр.Товар;
		Если текТовар = Null Тогда
			Продолжить;
		КонецЕсли;
		
		складDBF.Добавить();
		//Кол = стр.Кол_во;
	
		//Кол = Кол * текТовар.ЕдиницаДляОтчетов.Коэффициент;
		
		складDBF.ID = текТовар.УникальныйИдентификатор();
		складDBF.FOLDER = текТовар.Родитель.Ссылка.УникальныйИдентификатор();
		складDBF.NAME = текТовар.Наименование;
		кодЕд = СокрЛП(текТовар.БазоваяЕдиницаИзмерения.Код);
		Если кодЕд = "0301" или кодЕд = "815" Тогда
			складDBF.BOXED = 1;
			складDBF.INPACK = текТовар.ЕдиницаДляОтчетов.Коэффициент;
			складDBF.WEIGHT = 1;
		Иначе
			складDBF.INPACK = текТовар.ЕдиницаИзмеренияМест.Коэффициент;
		КонецЕсли;
		
		//складDBF.QTY = Кол;
		складDBF.CHANGES = стр.Изменение;
		Если Не текТовар.ЦеноваяГруппа.Пустая() Тогда
			складDBF.PRC_GRP = текТовар.ЦеноваяГруппа.Ссылка.УникальныйИдентификатор();
		КонецЕсли;
		
		Для Каждого цена Из стр.Цены Цикл
			складDBF.УстановитьЗначениеПоля("COST" + цена.Ключ, цена.Значение);		
		КонецЦикла;
		складDBF.Записать();	
	КонецЦикла;

	складDBF.ЗакрытьФайл();
	
	ПереместитьФайл (файлТовара, ПапкаОбмена + ?(СтрАгент = Неопределено, "\WAREHOUS.DBF", "\W" + КодАгента(СтрАгент.Агент) + ".DBF"));

Функция СоздатьЗаказ(Заказ, Агент, Организация, можетМенятьЦену, СкладОстатков, ПапкаОбмена)
	
	параметрыДоговора = ЗаполнениеДокументов.ПолучитьСтруктуруПараметровДляПолученияДоговораЗаказаПокупателя();
	
	Накладная = Документы.ЗаказПокупателя.СоздатьДокумент();
	
	ТекДата = ТекущаяДата();
	ТекВремя = ТекДата - НачалоДня( ТекДата );
	
	Накладная.Дата = Заказ.DATE + ТекВремя;
	Накладная.ДатаОтгрузки = Заказ.DATE;
	//Накладная.ДатаОплаты = Заказ.PAYDATE;
	
	Накладная.ВидОперации = Перечисления.ВидыОперацийЗаказПокупателя.ПродажаКомиссия;
	Накладная.Организация = организация;
	Накладная.УчитыватьНДС = Истина;
	Накладная.СуммаВключаетНДС = Истина;
	Накладная.АвторасчетНДС = Истина;
	Если Заказ.WILLPAY > 0 Тогда
		Накладная.БезДенегНеОтгружать = Истина;
	КонецЕсли;
	
	уид = Новый УникальныйИдентификатор(Заказ.ID);
	контрагент = Справочники.Контрагенты.ПолучитьСсылку(уид);
	Накладная.Грузополучатель = контрагент;
	Накладная.Контрагент = контрагент;
	Накладная.пкОплачено = Заказ.INCASS;
	Накладная.пкСуммаПогашенияДляМашины = Заказ.WILLSUM;
	
	договор = НайтиДоговор(контрагент, организация);
	
	Накладная.ДоговорКонтрагента = договор;
	Накладная.ТипЦен = Накладная.ДоговорКонтрагента.ТипЦен;
	
	Если можетМенятьЦену или Накладная.ТипЦен.Пустая() Тогда
		типЦен = Справочники.ТипыЦенНоменклатуры.НайтиПоНаименованию(Заказ.CTYPE);
		Если типЦен <> Неопределено Тогда
			Накладная.ТипЦен = типЦен;	
		КонецЕсли;
	КонецЕсли;
	
	Если ЗначениеЗаполнено(Накладная.ДоговорКонтрагента) Тогда

Процедура ЗаполнитьРасшифровкуПлатежей(СтруктураПараметров, док)
	
	СуммаДляПодбора = СтруктураПараметров.СуммаДляПодбора;
	СтавкаНДС=УправлениеПользователями.ПолучитьЗначениеПоУмолчанию(глЗначениеПеременной("глТекущийПользователь"),"ОсновнаяСтавкаНДС");
	ВсегоПлатежей=0;
	
	табл = СформироватьСписокДолговПоРегистру(СтруктураПараметров);
	Для Каждого СтрокаДолг Из табл Цикл
		
		Если СтруктураПараметров.ПодбиратьСумму Тогда
			
			Если ВсегоПлатежей+СтрокаДолг.СуммаПлатежа <= СуммаДляПодбора Тогда
				
				СуммаПлатежа=СтрокаДолг.СуммаПлатежа;
				СуммаВзаиморасчетов=СтрокаДолг.СуммаВзаиморасчетов;
				
				ВсегоПлатежей=ВсегоПлатежей+СуммаПлатежа;
				
			ИначеЕсли ВсегоПлатежей<СуммаДляПодбора Тогда
				
				СуммаПлатежа=СуммаДляПодбора-ВсегоПлатежей;
				СуммаВзаиморасчетов=МодульВалютногоУчета.ПересчитатьИзВалютыВВалюту(СуммаПлатежа,
							док.ВалютаДокумента,СтрокаДолг.ДоговорКонтрагента.ВалютаВзаиморасчетов,
							док.КурсДокумента,СтрокаДолг.КурсВзаиморасчетов,
							док.КратностьДокумента,СтрокаДолг.КратностьВзаиморасчетов);
				ВсегоПлатежей=ВсегоПлатежей+СуммаПлатежа;												
																
			Иначе
				
				Прервать;
				
			КонецЕсли;
			
		Иначе
			
			СуммаПлатежа=СтрокаДолг.СуммаПлатежа;
			СуммаВзаиморасчетов=СтрокаДолг.СуммаВзаиморасчетов;
			
		КонецЕсли;
		
		СтрокаПлатеж=Док.РасшифровкаПлатежа.Добавить();
		СтрокаПлатеж.ДоговорКонтрагента=СтрокаДолг.ДоговорКонтрагента;
		СтрокаПлатеж.Сделка=СтрокаДолг.Сделка;
		СтрокаПлатеж.ДокументРасчетовСКонтрагентом=СтрокаДолг.ДокументРасчетовСКонтрагентом;
		СтрокаПлатеж.СуммаПлатежа=СуммаПлатежа;
		СтрокаПлатеж.КурсВзаиморасчетов=СтрокаДолг.КурсВзаиморасчетов;
		СтрокаПлатеж.КратностьВзаиморасчетов=СтрокаДолг.КратностьВзаиморасчетов;
		СтрокаПлатеж.СуммаВзаиморасчетов=СуммаВзаиморасчетов;
		СтрокаПлатеж.СтатьяДвиженияДенежныхСредств = СтрокаПлатеж.ДоговорКонтрагента.ОсновнаяСтатьяДвиженияДенежныхСредств;
		СтрокаПлатеж.Проект = СтрокаПлатеж.ДоговорКонтрагента.ОсновнойПроект;
		
	КонецЦикла;
	
	Если СтруктураПараметров.ПодбиратьСумму И ВсегоПлатежей<СуммаДляПодбора Тогда
		
		СтрокаПлатеж=Док.РасшифровкаПлатежа.Добавить();
		СтрокаПлатеж.СуммаПлатежа=СуммаДляПодбора-ВсегоПлатежей;
		
		Если Не ЗначениеЗаполнено(док.ДоговорКонтрагента) Тогда
			
			СтрокаПлатеж.КурсВзаиморасчетов=1;
			СтрокаПлатеж.КратностьВзаиморасчетов=1;
			СтрокаПлатеж.СуммаВзаиморасчетов=СтрокаПлатеж.СуммаПлатежа;
			
		Иначе
			
			СтрокаПлатеж.ДоговорКонтрагента=док.ДоговорКонтрагента;
			ВалютаВзаиморасчетов=СтрокаПлатеж.ДоговорКонтрагента.ВалютаВзаиморасчетов;
			СтруктураКурсаВзаиморасчетов = МодульВалютногоУчета.ПолучитьКурсВалюты(ВалютаВзаиморасчетов,док.Дата);
			
			СтрокаПлатеж.КурсВзаиморасчетов      = СтруктураКурсаВзаиморасчетов.Курс;
			СтрокаПлатеж.КратностьВзаиморасчетов = СтруктураКурсаВзаиморасчетов.Кратность;
			
			
			СтрокаПлатеж.СуммаВзаиморасчетов=МодульВалютногоУчета.ПересчитатьИзВалютыВВалюту(СтрокаПлатеж.СуммаПлатежа,
									док.ВалютаДокумента,СтрокаПлатеж.ДоговорКонтрагента.ВалютаВзаиморасчетов,
									док.КурсДокумента,СтрокаПлатеж.КурсВзаиморасчетов,
									док.КратностьДокумента,СтрокаПлатеж.КратностьВзаиморасчетов);
									
			СтрокаПлатеж.СтатьяДвиженияДенежныхСредств = СтрокаПлатеж.ДоговорКонтрагента.ОсновнаяСтатьяДвиженияДенежныхСредств;
		КонецЕсли;
		
	КонецЕсли;

	Для каждого Стр из Накладная.ДокументыРасчетовСКонтрагентом Цикл
		Если стр.СуммаВзаиморасчетов = 0 Тогда
			пустыеСтроки.Добавить(стр);
		КонецЕсли;
	КонецЦикла;
	
	Для каждого стр из пустыеСтроки Цикл
		инд = Накладная.ДокументыРасчетовСКонтрагентом.Индекс(стр.Значение);
		Если инд >= 0 Тогда
			Накладная.ДокументыРасчетовСКонтрагентом.Удалить(инд);
		КонецЕсли;
	КонецЦикла;
	
	Если ПопыткаЗаписатьДокумент(ПапкаОбмена, Накладная, РежимЗаписиДокумента.Запись) = Ложь Тогда
		ответ.ordstatus = "fail";	
		ответ.message = "Ошибка при записи документа";
		СообщитьИнт(ПапкаОбмена, "Ошибка записи документа информация об ошибке в файле " + ПапкаОбмена + "\\log.txt");
		Возврат;
	КонецЕсли;
	
	ответ.ordstatus = "saved";
	ответ.ordnumber = Накладная.Номер;
	
	Если НомНакладной = "" Тогда
		НомНакладной = Накладная.Номер;
	КонецЕсли;			
	типДок = Строка(ТипЗнч(Накладная));

	СообщитьИнт(ПапкаОбмена, типДок + " № " + Накладная.Номер + 
			" от " + Накладная.Дата + 
			" контрагент " + Накладная.Контрагент.Наименование);
			
	Если НомНакладной <> "" Тогда
		ОбработанныеЗаказы.Добавить();
		ОбработанныеЗаказы.CREATED = Ном;
		ОбработанныеЗаказы.TYPE = "Returns";
		ОбработанныеЗаказы.Записать();

		ДобавитьВРегистрДокумент(ТипДокумента, Ном, Накладная, Агент);
	КонецЕсли;
	
	//Если ПопыткаЗаписатьДокумент(ПапкаОбмена, Накладная, РежимЗаписиДокумента.Проведение) Тогда
	//	ответ.ordstatus = "handled";

Функция ПриемПКОАгента(настройки, КодСотр, ПолноеИмя, ответы)
	Перем Документ, агент;
	
	ПапкаОбмена = настройки.ПапкаОбмена;	
	ТипДок = "ПКО";
	СчДок = 0;
	
	ЗапросДок = Новый Запрос;
	ЗапросДок.Текст = 
	"ВЫБРАТЬ
	|	НаполеонПринятыеДокументы.Документ
	|ИЗ
	|	РегистрСведений.НаполеонПринятыеДокументы КАК НаполеонПринятыеДокументы
	|ГДЕ
	|	НаполеонПринятыеДокументы.Агент = &Агент
	|	И НаполеонПринятыеДокументы.КодДокумента = &Код
	|	И НаполеонПринятыеДокументы.ТипДокумента = &Тип";
	
	
	Агент = НайтиАгента(КодСотр, ПолучитьАгентов(настройки));
	КассаАгента = Справочники.Кассы.ПустаяСсылка();
	ОрганизацияАгента = Неопределено;
	
	Если ТипЗнч(Агент) <> Тип("СправочникСсылка.Пользователи") Тогда
		ОрганизацияАгента = Агент.Организация;
		КассаАгента = Агент.Касса;
		Агент = Агент.Агент;
	Иначе         
		ОрганизацияАгента = Справочники.Организации.Выбрать().Следующий();
	КонецЕсли;

	
	временныйФайл = ИмяВременногоФайла(ПапкаОбмена + "\TINC");
	ПереместитьФайл(ПолноеИмя, временныйФайл);
	
	ДДок = Новый XBase(временныйФайл);
	ДДок.Кодировка = КодировкаXBase.ANSI;
	темпФайл = ИмяВременногоФайла(ПапкаОбмена + "\TPOD");
	ОбработанныеДокументы = ФайлОбработанныхЗаказов(ПапкаОбмена, КодСотр, темпФайл);
	
	ЗаписатьДок = 0;
	Ном = "";
	
	ДДок.Первая();		
	Пока ДДок.ВКонце() = 0 Цикл
		//
		// Проверим признак начала заявки
		//
		Если ДДок.ORDFLAG <> 0 или СтрДлина(СокрЛП(ДДок.NUM_I)) = 0 Тогда
			//
			// началась новая заявка, если надо записать заявка, запишем ее.
			//
			Если ЗаписатьДок = 1 Тогда
				ОбработанныеДокументы.Добавить();
				ОбработанныеДокументы.CREATED = Ном;
				ОбработанныеДокументы.TYPE = "Incass";
				ОбработанныеДокументы.Записать();
			КонецЕсли;
			
			ЗаписатьДок = 1;
			
			Ном = СокрЛП(ДДок.CREATED);
			//
			// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
			//
			//
			// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
			//				
			ЗапросДок.Параметры.Вставить("Агент", Агент);
			ЗапросДок.Параметры.Вставить("Код", Ном);
			ЗапросДок.Параметры.Вставить("Тип", ТипДок);
			выб = ЗапросДок.Выполнить().Выбрать();
			Если выб.Следующий() Тогда
				//
				// пропускаем этот заказ
				//
				ДДок.Следующая();
				СообщитьИнт(ПапкаОбмена, "Повторный ПКО " + выб.Документ);
				ЗаписатьДок = 0;
				Продолжить;
			КонецЕсли;
			
		КонецЕсли;
		
		Если ЗаписатьДок = 0 или СтрДлина(СокрЛП(ДДок.NUM_I)) = 0 Тогда
			ДДок.Следующая();
			Продолжить;
		КонецЕсли;
		
		реализация = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(СокрЛП(ДДок.NUM_I), ДДок.DATE_I);
		Если реализация <> Неопределено Тогда
			ТекДата = ТекущаяДата();
			ТекВремя = ТекДата - НачалоДня( ТекДата );
			
			ДатаДок = ДДок.DATE + ТекВремя;
			СоздатьПКО(ПапкаОбмена, реализация, ДДок.SUM_I, ДатаДок, Агент, КассаАгента, Ном, ТипДок, ответы, ДДок.REMARK);
			СчДок = СчДок + 1;
		КонецЕсли;
								
		ДДок.Следующая();
	КонецЦикла;
	
	Если ЗаписатьДок = 1 Тогда
		ОбработанныеДокументы.Добавить();
		ОбработанныеДокументы.CREATED = Ном;
		ОбработанныеДокументы.TYPE = "Incass";
		ОбработанныеДокументы.Записать();
	КонецЕсли;
	
	ДДок.ЗакрытьФайл();
	ЗакрытьФайлОбработанныхЗаявок(ПапкаОбмена, ОбработанныеДокументы, КодСотр, темпФайл);
	
	УдалитьФайлы(временныйФайл);

	Возврат СчДок;
КонецФункции

Процедура ОбновитьКоличестваДляВозврата(ПапкаОбмена, реализацииДляВозвратов, ПринятыеЗаявки, Агент)
	Запрос = Новый Запрос();
	Запрос.Текст = 
	"ВЫБРАТЬ
	|	СписанныеТовары.Регистратор,
	|	СписанныеТовары.Период
	|ИЗ
	|	РегистрСведений.СписанныеТовары КАК СписанныеТовары
	|ГДЕ
	|	СписанныеТовары.ДокументПартии В(&ДокументыПартий)";
	
	Запрос.Параметры.Вставить("ДокументыПартий", реализацииДляВозвратов);
	Выборка = Запрос.Выполнить().Выбрать();
	
	ВозвратыИзНакладных = Новый Соответствие;
	ДатаКон = ТекущаяДата();
	ДобавитьНепроведенныеВозвраты(ВозвратыИзНакладных, ДатаКон, ДатаКон - 7 * 24 * 3600);
	
	Пока Выборка.Следующий() Цикл
		Если ТипЗнч(Выборка.Регистратор) = Тип("ДокументСсылка.ВозвратТоваровОтПокупателя") Тогда
			ЗаполнитьДанныеПоНакладныи(Выборка.Регистратор, ВозвратыИзНакладных);
		КонецЕсли;
	КонецЦикла;
	
	врФайл = ИмяВременногоФайла(ПапкаОбмена+"\TRDV");
	файл = новый Файл(ПапкаОбмена + "\DDV" + КодАгента(Агент) + ".DBF");

		ПереместитьФайл(файл.ПолноеИмя, врФайл);

	Иначе	
		ФайлТекНакладных = СоздатьФайлОтгрузок(врФайл);
	КонецЕсли;
	Для каждого стр из реализацииДляВозвратов Цикл 
		док = стр.Значение;
		
		// добавим разделитель, чтобы при приеме не "склеились" накладные
		ФайлТекНакладных.Добавить();
		ФайлТекНакладных.ID = "";
		ФайлТекНакладных.NUM = "";
		ФайлТекНакладных.Записать();
		
		СохранитьНакладную(ФайлТекНакладных, док.ПолучитьОбъект(), 0, НомерДокументаЗаказа(док, ПринятыеЗаявки), ВозвратыИзНакладных);
	КонецЦикла;
	
	ФайлТекНакладных.ЗакрытьФайл();
	ПереместитьФайл(врФайл, файл.ПолноеИмя);  
КонецПроцедуры

Функция ПриемВозвратаАгента(настройки, КодСотр, ПолноеИмя, ответы)
	СчДок = 0;
	ТипДок = "Возврат";
	
	ЗапросДок = Новый Запрос;
	ЗапросДок.Текст = 
	"ВЫБРАТЬ
	|	НаполеонПринятыеДокументы.Документ
	|ИЗ
	|	РегистрСведений.НаполеонПринятыеДокументы КАК НаполеонПринятыеДокументы
	|ГДЕ
	|	НаполеонПринятыеДокументы.Агент = &Агент
	|	И НаполеонПринятыеДокументы.КодДокумента = &Код
	|	И НаполеонПринятыеДокументы.ТипДокумента = &Тип";
	
	
	ПапкаОбмена = настройки.ПапкаОбмена;
	Агент = НайтиАгента(КодСотр, ПолучитьАгентов(настройки));
	ОрганизацияАгента = Неопределено;
	СкладВозвратов = настройки.СкладВозвратов;
	
	Если ТипЗнч(Агент) <> Тип("СправочникСсылка.Пользователи") Тогда
		ОрганизацияАгента = Агент.Организация;
		Если Не Агент.Склад.Пустая() Тогда
			СкладВозвратов = Агент.Склад;
		КонецЕсли;
		Агент = Агент.Агент;
	Иначе         
		ОрганизацияАгента = Справочники.Организации.Выбрать().Следующий();
	КонецЕсли;
	
	временныйФайл = ИмяВременногоФайла(ПапкаОбмена + "\TVZT");
	ПереместитьФайл(ПолноеИмя, временныйФайл);
	
	ДДок = Новый XBase(временныйФайл);
	ДДок.Кодировка = КодировкаXBase.ANSI;
	темпФайл = ИмяВременногоФайла(ПапкаОбмена + "\TPOD");
	ОбработанныеДокументы = ФайлОбработанныхЗаказов(ПапкаОбмена, КодСотр, темпФайл);
	
	ЗаписатьДок = 0;
	Ном = "";
	Документ = Неопределено;
	
	ДатаНач = ТекущаяДата();
	реализацииДляВозвратов = Новый СписокЗначений;
	ДДок.Первая();		
	Пока ДДок.ВКонце() = 0 Цикл
		//
		// Проверим признак начала заявки
		//
		Если (ДДок.ORDFLAG <> 0 и ДДок.DOCFLAG <> 0) Тогда
			//
			// началась новая заявка, если надо записать заявка, запишем ее.
			//
			Если ЗаписатьДок = 1 Тогда
				ЗаписатьВозврат(ПапкаОбмена, Документ, Ном, ОбработанныеДокументы, Агент, ТипДок, ответы);
			КонецЕсли;
			
			ЗаписатьДок = 0;
			
			Ном = СокрЛП(ДДок.CREATED);
			ЗапросДок.Параметры.Вставить("Агент", Агент);
			ЗапросДок.Параметры.Вставить("Код", Ном);
			ЗапросДок.Параметры.Вставить("Тип", ТипДок);
			выб = ЗапросДок.Выполнить().Выбрать();
			Если выб.Следующий() Тогда
				//
				// пропускаем этот заказ
				//
				ДДок.Следующая();
				СообщитьИнт(ПапкаОбмена, "Повторный возврат " + выб.Документ);
				Продолжить;
			КонецЕсли;
			
			Если СтрДлина(СокрЛП(ДДок.ID_I)) <> 0 Тогда
				//
				// создаем новую заявку
				//
				Документ = СоздатьВозврат(ДДок, Агент, ОрганизацияАгента, СкладВозвратов);
				ЗаписатьДок = 1;
				СчДок = СчДок + 1;
			КонецЕсли;
		КонецЕсли;
		
		Если ЗаписатьДок = 0 или СтрДлина(СокрЛП(ДДок.ID_I)) = 0 Тогда
			ДДок.Следующая();
			Продолжить;
		КонецЕсли;
					
		уид = Новый УникальныйИдентификатор(ДДок.ID_I);
		номенклатура = Справочники.Номенклатура.ПолучитьСсылку(уид);

		товар = Документ.Товары.Добавить();
		товар.Номенклатура = номенклатура.Ссылка;
		товар.Склад = Документ.СкладОрдер;
		
		СтрокаТабличнойЧасти = товар;
		ОбработкаТабличныхЧастей.ЗаполнитьЕдиницуЦенуПродажиТабЧасти(СтрокаТабличнойЧасти, Документ, Документ.ВалютаДокумента);
		
		коэф = товар.ЕдиницаИзмерения.Коэффициент;
		Если коэф = 0 Тогда
			коэф = 1;
		КонецЕсли;
		товар.Количество = ДДок.QTY_I / коэф;
		товар.Коэффициент = коэф;
		товар.Качество = Справочники.Качество.Новый;
		товар.ПричинаВозврата = Справочники.ПричиныВозврата.НайтиПоКоду(ДДок.CAUSE);
		товар.ДатаПроизводства = ДДок.MFRDATE;
		
		докНакл = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(СокрЛП(ДДок.NUM_I), ДДок.DATE_I);
		Если докНакл <> Неопределено Тогда
			реализацииДляВозвратов.Добавить(докНакл);
			товар.ДокументПартии = докНакл;
			ДатаЗаявки = докНакл.Дата - 2 * 24 * 3600;
			Если ДатаНач > ДатаЗаявки Тогда
				ДатаНач = ДатаЗаявки;
			КонецЕсли;
		КонецЕсли;
		
		товар.Цена = ДДок.COST_I;
		
		ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар,  Документ);
		ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Документ);
		ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Документ);
		
		ДДок.Следующая();
	КонецЦикла;
	
	Если ЗаписатьДок = 1 Тогда
		ЗаписатьВозврат(ПапкаОбмена, Документ, Ном, ОбработанныеДокументы, Агент, ТипДок, ответы);
	КонецЕсли;
	
	ДДок.ЗакрытьФайл();
	ЗакрытьФайлОбработанныхЗаявок(ПапкаОбмена, ОбработанныеДокументы, КодСотр, темпФайл);
	
	УдалитьФайлы(временныйФайл);
	
	Если реализацииДляВозвратов.Количество() > 0 Тогда
		Попытка
			ПринятыеЗаявки = ЗагрузитьПринятыеЗаявки(ПапкаОбмена, Агент, ДатаНач);
			ОбновитьКоличестваДляВозврата(ПапкаОбмена, реализацииДляВозвратов, ПринятыеЗаявки, Агент);
		Исключение
			СообщитьИнт(ПапкаОбмена, ОписаниеОшибки());
		КонецПопытки;
	КонецЕсли;
	
	Возврат СчДок;
КонецФункции

Функция ПриемЗаявокАгента(настройки, КодСотр, ПолноеИмя, ответы)
	Перем Накладная, агент, ответ;
	
	ТипДокумента = "Заказ";
	ПапкаОбмена = настройки.ПапкаОбмена;
	СчНакл = 0;
	
	Агент = НайтиАгента(КодСотр, ПолучитьАгентов(настройки));
	ОрганизацияАгента = Неопределено;
	КассаАгента = Справочники.Кассы.ПустаяСсылка();
	СкладАгента = настройки.СкладОстатков;
	
	Если ТипЗнч(Агент) <> Тип("СправочникСсылка.Пользователи") Тогда
		ОрганизацияАгента = Агент.Организация;
		КассаАгента = Агент.Касса;
		Если Не Агент.Склад.Пустая() Тогда
			СкладАгента = Агент.Склад;
		КонецЕсли;
		Агент = Агент.Агент;
	Иначе         
		ОрганизацияАгента = Справочники.Организации.Выбрать().Следующий();
	КонецЕсли;
	
	ЗапросДок = Новый Запрос;
	ЗапросДок.Текст = 
	"ВЫБРАТЬ
	|	НаполеонПринятыеДокументы.Документ
	|ИЗ
	|	РегистрСведений.НаполеонПринятыеДокументы КАК НаполеонПринятыеДокументы
	|ГДЕ
	|	НаполеонПринятыеДокументы.Агент = &Агент
	|	И НаполеонПринятыеДокументы.КодДокумента = &Код
	|	И НаполеонПринятыеДокументы.ТипДокумента = &Тип";
	
	можетМенятьЦену = Ложь;
	
	временныйФайл = ИмяВременногоФайла(ПапкаОбмена + "\TORD");
	ПереместитьФайл(ПолноеИмя, временныйФайл);
	
	Заказ = Новый XBase(временныйФайл);
	Заказ.Кодировка = КодировкаXBase.ANSI;
	темпФайл = ИмяВременногоФайла(ПапкаОбмена + "\TPOD");
	ОбработанныеЗаказы = ФайлОбработанныхЗаказов(ПапкаОбмена, КодСотр, темпФайл);
	
	ЗаписатьЗаказ = 0;
	Ном = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 или СтрДлина(СокрЛП(Заказ.ID_I)) = 0 Тогда
			//
			// началась новая заявка, если надо записать заявка, запишем ее.
			//
			Если ЗаписатьЗаказ = 1 Тогда
				ЗаписатьЗаявку(ПапкаОбмена, Накладная, Ном, ОбработанныеЗаказы, Агент, КассаАгента, ТипДокумента, ответ);
				Если ответы <> Неопределено Тогда
					ответы.Добавить(ответ);
				КонецЕсли;
			КонецЕсли;
			
			ЗаписатьЗаказ = 0;
			
			ответ = СоздатьОтвет();
			Ном = СокрЛП(Заказ.CREATED);
			//
			// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
			//				
			ЗапросДок.Параметры.Вставить("Агент", Агент);
			ЗапросДок.Параметры.Вставить("Код", Ном);
			ЗапросДок.Параметры.Вставить("Тип", ТипДокумента);
			выб = ЗапросДок.Выполнить().Выбрать();
			Если выб.Следующий() Тогда
				//
				// пропускаем этот заказ
				//
				Заказ.Следующая();
				мсг = "Повторная заявка " + выб.Документ;
				СообщитьИнт(ПапкаОбмена, мсг);
				Если ответы <> Неопределено Тогда
					ответ.created = Ном;
					ответ.ordstatus = "repeat";
					ответ.ordnumber = выб.Документ.Номер;
					ответ.message = мсг;
					ответы.Добавить(ответ);
				КонецЕсли;
				Продолжить;
			КонецЕсли;
			
			//
			// создаем новую заявку
			//
			Накладная = СоздатьЗаказ(Заказ, Агент, ОрганизацияАгента, можетМенятьЦену, СкладАгента, ПапкаОбмена);			
			ЗаписатьЗаказ = 1;
			
			СчНакл = СчНакл + 1;
		КонецЕсли;
		
		Если ЗаписатьЗаказ = 0 или СтрДлина(СокрЛП(Заказ.ID_I)) = 0 Тогда
			Заказ.Следующая();
			Продолжить;
		КонецЕсли;
		
		уид = Новый УникальныйИдентификатор(Заказ.ID_I);
		номенклатура = Справочники.Номенклатура.ПолучитьСсылку(уид);

		товар = Накладная.Товары.Добавить();
		товар.Номенклатура = номенклатура.Ссылка;
		//товар.Размещение = Накладная.СкладГруппа;
		
		Если ТипЗнч(Накладная) = Тип("ДокументСсылка.ЗаказПокупателя") Тогда
			Накладная.ПриИзмененииНоменклатурыТоваров(товар);
		Иначе
			СтрокаТабличнойЧасти = товар;
			ОбработкаТабличныхЧастей.ЗаполнитьЕдиницуЦенуПродажиТабЧасти(СтрокаТабличнойЧасти, Накладная,
				Накладная.ВалютаДокумента);
		КонецЕсли;
		
		товар.ЕдиницаИзмеренияМест = номенклатура.ЕдиницаИзмеренияМест;	
		Если Заказ.PACK = 1 Тогда
			товар.ЕдиницаИзмерения = номенклатура.ЕдиницаИзмеренияМест;	
		КонецЕсли;
		
		коэф = товар.ЕдиницаИзмерения.Коэффициент;
		Если коэф = 0 Тогда
			коэф = 1;
		КонецЕсли;
		товар.Количество = Заказ.QTY / коэф;
		товар.Цена = товар.Цена * коэф;
		товар.Коэффициент = коэф;
		
		ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар, Накладная);
		// 13.04.2017 Д.М.
		ОбработкаТабличныхЧастей.РассчитатьКоличествоМестТабЧасти(товар, Накладная);
		// 13.04.2017 Д.М.
		ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Накладная);
		ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Накладная);
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если ЗаписатьЗаказ = 1 Тогда
		ЗаписатьЗаявку(ПапкаОбмена, Накладная, Ном, ОбработанныеЗаказы, Агент, КассаАгента, ТипДокумента, ответ);
		Если ответы <> Неопределено Тогда
			ответы.Добавить(ответ);
		КонецЕсли;
	КонецЕсли;
	
	СохранитьПринятыеДокументы(ПапкаОбмена, Агент, ТипДокумента);
	Заказ.ЗакрытьФайл();
	ЗакрытьФайлОбработанныхЗаявок(ПапкаОбмена, ОбработанныеЗаказы, КодСотр, темпФайл);
	
	УдалитьФайлы(временныйФайл);
	
	Возврат СчНакл;

Функция OrderHandler(КодАгента) Экспорт
	
	ответы = Новый Массив;
	
	настройки = ПолучитьНастройки();
	ПапкаОбмена = настройки.ПапкаОбмена;
	
	Попытка
		
		файлПКО = Новый Файл(ПапкаОбмена + "\Incass" + КодАгента + ".DBF");
		Если файлПКО.Существует() Тогда
			СообщитьИнт(ПапкаОбмена, "Прием ПКО от " + КодАгента);
			ПриемПКОАгента(настройки, КодАгента, файлПКО.ПолноеИмя, ответы);
		КонецЕсли;
		файл = Новый Файл(ПапкаОбмена + "\ORD" + КодАгента + ".DBF");
		Если файл.Существует() Тогда
			СообщитьИнт(ПапкаОбмена, "Прием заявок от " + КодАгента);
			ПриемЗаявокАгента(настройки, КодАгента, файл.ПолноеИмя, ответы);
		КонецЕсли;
		файл = Новый Файл(ПапкаОбмена + "\VZT" + КодАгента + ".DBF");
		Если файл.Существует() Тогда
			СообщитьИнт(ПапкаОбмена, "Прием возвратов от " + КодАгента);
			ПриемВозвратаАгента(настройки, КодАгента, файл.ПолноеИмя, ответы);
		КонецЕсли;
		
		ЗагрузитьОтвергнутыеВозвраты(ответы, настройки, КодАгента);
	Исключение
		СообщитьИнт(ПапкаОбмена, ОписаниеОшибки());
	КонецПопытки;
	
	Возврат ответы;

Процедура ВыгрузитьМаршрутАгента(Параметры)
	КаталогФайла = Параметры.ПапкаОбмена + "\";
	КаталогВременныхФайлов = КаталогВременныхФайлов();
	
	ДБФ = Новый XBase;
	ДБФ.Кодировка = КодировкаXBase.ANSI;
	
	ГУИД_Менеджера = Параметры.Агент.УникальныйИдентификатор();
	ИмяФайла = "FORG"+ГУИД_Менеджера+".dbf";
	ИмяВременногоФайла = "FORG.dbf";
	
	ПолныйПутьКФайлу = КаталогФайла+ИмяФайла;
	ПолныйПутьВременногоФайла = КаталогВременныхФайлов()+ИмяВременногоФайла;
	
	ДБФ.поля.Добавить("ID","S",100);
	ДБФ.поля.Добавить("NAME","S",100);
	ДБФ.СоздатьФайл(ПолныйПутьВременногоФайла);
	ДБФ.Записать();
	
	ТекущаяНеделя = НеделяГода(ТекущаяДата());
	ТекущаяНеделяМесяца = НеделяГода(НачалоМесяца(ТекущаяДата()));
	
	ТекущаяНеделя = ТекущаяНеделя - ТекущаяНеделяМесяца + 1;
	ТекущаяНеделя = ?(ТекущаяНеделя>4,4,ТекущаяНеделя);
	Для Каждого Стр Из Параметры.Маршрут Цикл
		ПрисутствуетДеньНедели = Стр["Неделя"+ТекущаяНеделя];
		Если НЕ ПрисутствуетДеньНедели Тогда
			Продолжить;
		КонецЕсли;
		ДБФ.Добавить();
		ДБФ.ID = Стр.Контрагент.УникальныйИдентификатор();
		ДБФ.NAME = Строка(Стр.ДеньНедели);
		ДБФ.Записать();
	КонецЦикла;
	
	
	ДБФ.ЗакрытьФайл();
	
	ПереместитьФайл(ПолныйПутьВременногоФайла,ПолныйПутьКФайлу);
КонецПроцедуры

Процедура ВыгрузкаМаршрутов() Экспорт
	НастройкаНаполеон = ПолучитьНастройки();
	 
	ВыборкаАгенты = Справочники.НаполеонАгенты.Выбрать(,НастройкаНаполеон);
	Пока ВыборкаАгенты.Следующий() Цикл
		Если ВыборкаАгенты.ПометкаУдаления Тогда
			Продолжить;
		КонецЕсли;
		СтруктураПараметров = Новый Структура; 
		СтруктураПараметров.Вставить("ПапкаОбмена",НастройкаНаполеон.ПапкаОбмена);
		СтруктураПараметров.Вставить("Агент",ВыборкаАгенты.Агент);
		Док = ПоискПоследнегоМаршрута(ВыборкаАгенты.Агент);
		Если Док <> Неопределено Тогда
			ТЗМаршрут = Док.Маршрут.Выгрузить();
			ВремТЗ = ТЗМаршрут.Скопировать();
			ВремТЗ.Свернуть("Контрагент");
			Для Каждого СтрВремТЗ Из ВремТЗ Цикл
				Если НЕ КонтрагентПринадлежитАгенту(СтрВремТЗ.Контрагент,ВыборкаАгенты) Тогда
					УдалитьКонтрагентаИзТЗ(ТЗМаршрут,СтрВремТЗ.Контрагент);
				КонецЕсли;
			КонецЦикла;
			ЧистыйМаршрут = Документы.Маршруты.СоздатьДокумент();
			ЧистыйМаршрут.Дата = ТекущаяДата();
			ЧистыйМаршрут.Менеджер = Док.Менеджер;
			ЧистыйМаршрут.Маршрут.Загрузить(ТЗМаршрут);
			ЧистыйМаршрут.Записать(РежимЗаписиДокумента.Проведение);
			
			СтруктураПараметров.Вставить("Маршрут",ЧистыйМаршрут.Маршрут.Выгрузить());
		Иначе 
			Продолжить;
		КонецЕсли;
		
		ВыгрузитьМаршрутАгента(СтруктураПараметров);
	КонецЦикла;
КонецПроцедуры
