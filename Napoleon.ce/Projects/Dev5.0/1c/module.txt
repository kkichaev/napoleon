Функция СжатьДанные(данные)
	имяФайла = КаталогВременныхФайлов() + "data";
	тд = Новый ТекстовыйДокумент();
	тд.ДобавитьСтроку(данные);
	тд.Записать(имяФайла, КодировкаТекста.UTF8);
	
	поток = Новый ПотокВПамяти();	
	зип = Новый ЗаписьZipФайла(поток,,,МетодСжатияZIP.Сжатие);
	зип.Добавить(имяФайла);
	зип.Записать();
	
	поток.Перейти(0, ПозицияВПотоке.Начало);
	читать = Новый ЧтениеДанных(поток);
	читать.Пропустить(14);
	crc = читать.ПрочитатьЦелое32();
	размерСжатый = читать.ПрочитатьЦелое32();
	размер = читать.ПрочитатьЦелое32();
	
	// fileNameLength + extraLentgh +fileName+extra = 8 for data filename
	читать.Пропустить(8);
	тело = читать.Прочитать(размерСжатый).ПолучитьДвоичныеДанные();
	
	читать.Закрыть();
	поток.Закрыть();
	
	зппоток = Новый ПотокВПамяти();
	запись = Новый ЗаписьДанных(зппоток);
	
	// GZip Header for data fileName
	запись.Записать(Base64Значение("H4sICAAAAAAEAGRhdGEA"));
	запись.Записать(тело);
	запись.ЗаписатьЦелое32(crc);
	запись.ЗаписатьЦелое32(размер);
	запись.Закрыть();

	Возврат зппоток.ЗакрытьИПолучитьДвоичныеДанные();	
КонецФункции

Процедура ВыгрузитьНакладные(сообщения) ЭКСПОРТ
	Сообщения.Добавить("Выгрузка долгов...");
	накладные = Новый Массив;
	
	ДатаКнц = ТекущаяДата();
	ДатаНач = ДобавитьМесяц(ДатаКнц, -ГлубинаВыгрузкиДолгов);
	Для Каждого стр из ПолучитьДанные("ExchangeInfo", """type""='Delivery'") Цикл
		ДатаНач = Дата(стр.time);
	КонецЦикла;
	
	// загружаем заявкм на день раньше до накладных
	Заявки = ПолучитьЗагруженныеЗаявки(ДатаНач - 24 * 3600);
	
	Запрос = Новый Запрос;
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	РеализацияТоваровУслуг.Ссылка КАК Документ
		|ИЗ
		|	Документ.РеализацияТоваровУслуг КАК РеализацияТоваровУслуг
		|ГДЕ
		|	РеализацияТоваровУслуг.Дата МЕЖДУ &ДатаНач И &ДатаКнц
		|	И РеализацияТоваровУслуг.ПометкаУдаления = ЛОЖЬ"
		;
	Запрос.УстановитьПараметр("ДатаНач", ДатаНач);
	Запрос.УстановитьПараметр("ДатаКнц", ДатаКнц);
	выб = Запрос.Выполнить().Выбрать();
	Пока выб.Следующий() Цикл
		
		заявкаАгента = Неопределено;
		Если ЗначениеЗаполнено(выб.Документ.ЗаказКлиента) Тогда
			заказ = выб.Документ.ЗаказКлиента;
			ключ = ДатаВСтроку(заказ.Дата) + "|" + СокрЛП(Заказ.Номер);
			заявкаАгента = Заявки.Получить(ключ);
		КонецЕсли;
		
		товары = Новый Массив;
		
		Для Каждого стр из выб.Документ.Товары Цикл
			товары.Добавить(Новый Структура("id,qty,sum"
				,КодОбъекта(стр.Номенклатура)
				,стр.Количество
				,стр.Сумма
			));
		КонецЦикла;
		
		док = новый Структура("id,date,number,items"
			,КодОбъекта(выб.Документ.Партнер)
			,ДатаВСтроку(выб.Документ.Дата)
			,выб.Документ.Номер
			,товары
		);
		Если заявкаАгента <> Неопределено Тогда
			док.Вставить("created", заявкаАгента.created);
			док.Вставить("userid", заявкаАгента.userid);
		КонецЕсли;
		накладные.Добавить(док);
	КонецЦикла;
	
	данные = Новый Массив;
	Если накладные.Количество() > 0 Тогда
		данные.Добавить(Новый Структура("Тип,Данные", "Delivery", накладные));

Процедура ДобавитьТовар(Документ, ТоварИст, ВидЦеныДокумента)
	Товар = Документ.Товары.Добавить();
	
	Номенклатура = СоздатьОбъект(Справочники.Номенклатура, ТоварИст.id);
	Товар.Номенклатура = Номенклатура;	
	
	Товар.Количество = ТоварИст.qty;	
	Товар.КоличествоУпаковок = ТоварИст.qty;
	Товар.ВидЦены = ВидЦеныДокумента;
	Товар.Упаковка = Номенклатура.ЕдиницаИзмерения;
	Товар.Склад = Документ.Склад;
	
	
	Если ТоварИст.unit <> "" Тогда		
		Упаковка = СоздатьОбъект(Справочники.УпаковкиЕдиницыИзмерения, ТоварИст.unit);
		Товар.Упаковка = Упаковка;
		множитель = МножительУпаковки(Упаковка);
		Товар.КоличествоУпаковок = ТоварИст.qty / множитель;
	КонецЕсли;
	
	ОбработкаСтроки(Документ, Товар);
	
	Если ТипЗнч(Документ.Ссылка) = Тип("ДокументСсылка.ЗаказКлиента") Тогда
		Товар.ВариантОбеспечения = Перечисления.ВариантыОбеспечения.Отгрузить;
	КонецЕсли;
