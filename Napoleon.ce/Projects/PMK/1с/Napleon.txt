Процедура ПриемЗаявок() Экспорт
	Перем Накладная, агент;
	
	ПапкаОбмена=РегистрыСведений.ДополнительныеНастройкиУчета.Получить().КаталогОбменаКПК;
	ГруппаНоменклатурыКПК=РегистрыСведений.ДополнительныеНастройкиУчета.Получить().ГруппаНоменклатурыКПК;
	Рез = НайтиФайлы(ПапкаОбмена, "ORD*.DBF");
		
	СчНакл = 0;
	
	Для Каждого файл из Рез Цикл
		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");

		КодСотр = СокрЛП(Сред(имяФайла, 4, разделитель-4));
		Агент = НайтиАгента(КодСотр);
		
		временныйФайл = ПапкаОбмена + "\TORD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		Заказ = Новый XBase(временныйФайл);
		Заказ.Кодировка = КодировкаXBase.OEM;
		ОбработанныеЗаказы = ФайлОбработанныхЗаказов(КодСотр,ПапкаОбмена);
		
		ЗаписатьЗаказ = 0;
		ПринятыеЗаявки = ЗагрузитьПринятыеЗаявки(КодСотр, Истина, ПапкаОбмена);
		Ном = "";
		
		Заказ.Первая();		
		Пока Заказ.ВКонце() = 0 Цикл
			//
			// Проверим признак начала заявки
			//
			Если Заказ.ORDFLAG <> 0 Тогда
				//
				// началась новая заявка, если надо записать заявка, запишем ее.
				//
				Если ЗаписатьЗаказ = 1 Тогда
					Если СохранитьЗаявку(Накладная,КодСотр,ПапкаОбмена) Тогда;

						ОбработанныеЗаказы.Добавить();
						ОбработанныеЗаказы.CREATED = Ном;
						ОбработанныеЗаказы.Записать();
						ПринятыеЗаявки.Добавить(Ном, Накладная.Номер);
						//СмирныхИА СКФ 07.06.2013
						Если НЕ Накладная.Проведен Тогда
							ОтправитьСообщениеКПК(КодСотр,"Заявка: "+Накладная.Номер+" на "+Накладная.Контрагент+" "+Накладная.АдресДоставки+" не проведена. Свяжитесь с оператором",ПапкаОбмена);
						КонецЕсли;
						//------------------------
					Иначе
						ОтправитьСообщениеКПК(КодСотр,"Не удалось загрузить заявку на "+Накладная.Контрагент+" "+Накладная.АдресДоставки+". Необходимо повторно отправить заявку",ПапкаОбмена);
					КонецЕсли;
					ЗаписатьЗаказ = 0;
				КонецЕсли;
				
				//
				// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
				//
				пз = ПринятыеЗаявки.НайтиПоЗначению(СокрЛП(Заказ.CREATED));
				Если пз <> Неопределено Тогда
					//
					// пропускаем этот заказ
					//
					ЗаписатьЗаказ = 0;
					Заказ.Следующая();
					Продолжить;
				КонецЕсли;
				
				//
				// создаем новую заявку
				//
				ЗаписатьЗаказ = 1;
				Ном = СокрЛП(Заказ.CREATED);
				СчНакл = СчНакл + 1;
				
				Накладная = СоздатьЗаказ(Заказ, Агент);				
				//Накладная = СоздатьНакладную(Заказ, Агент);				
			КонецЕсли;
			
			Если ЗаписатьЗаказ = 0 Тогда
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
			
			товар = Накладная.Товары.Добавить();
			номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);
			Если номенклатура=Справочники.Номенклатура.ПустаяСсылка()ИЛИ НЕ номенклатура.ПринадлежитЭлементу(ГруппаНоменклатурыКПК) Тогда
				//СмирныхИА СКФ 17.03.2015 Возможно вывели из прайса
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
			товар.Номенклатура = номенклатура.Ссылка;
			товар.ЕдиницаИзмерения = номенклатура.ЕдиницаХраненияОстатков;
			//коэф = товар.КоэффициентОтгрузки;
			//Если коэф = 0 Тогда
			//	коэф = 1;
			//КонецЕсли;
			товар.Коэффициент = товар.ЕдиницаИзмерения.Коэффициент;
			Накладная.ПриИзмененииНоменклатурыТоваров(товар);
			Примечание = СокрЛП(Заказ.REM_I);
			Если ЗначениеЗаполнено(Примечание) Тогда
				товар.Примечание=Примечание;
			КонецЕсли;
			Если номенклатура.БазоваяЕдиницаИзмерения=номенклатура.ЕдиницаХраненияОстатков.ЕдиницаПоКлассификатору Тогда //весовой товар
				Если Заказ.INKG=1 Тогда
					Товар.Количество = Заказ.QTY;
				Иначе
					Товар.Количество = Заказ.QTY * номенклатура.ЕдиницаХраненияОстатков.Коэффициент;
				КонецЕсли;
			Иначе	//штучный товар
				Если Заказ.INKG=1 Тогда
					Товар.Количество = Заказ.QTY / номенклатура.ЕдиницаХраненияОстатков.Коэффициент;
				Иначе
					Товар.Количество = Заказ.QTY;
				КонецЕсли;
			КонецЕсли;
			//товар.Количество = Заказ.QTY * коэф;

			//Если ТипЗнч(Накладная) = Тип("ДокументСсылка.ЗаказПокупателя") Тогда
				Накладная.мСписокФильтров = УправлениеДопПравамиПользователей.ПравоНеОтпускатьТоварСЦенойНижеОпределенногоТипа();
				Накладная.ПриИзмененииНоменклатурыТоваров(товар);
				мУчетнаяПолитика = ОбщегоНазначения.ПолучитьПараметрыУчетнойПолитикиУпр(ТекущаяДата(),ложь);
				мСуммаДокументаБезСкидок = Ценообразование.ПолучитьСуммуДокументаБезСкидки(Накладная.Товары) + Ценообразование.ПолучитьСуммуДокументаБезСкидки(Накладная.Услуги);
				мМинимальныеЦены = Новый ТаблицаЗначений;
				мМинимальныеЦены.Колонки.Добавить("Номенклатура");
				мМинимальныеЦены.Колонки.Добавить("ХарактеристикаНоменклатуры");
				мМинимальныеЦены.Колонки.Добавить("ЕдиницаИзмерения");
				мМинимальныеЦены.Колонки.Добавить("Цена");
				мСписокФильтров = УправлениеДопПравамиПользователей.ПравоНеОтпускатьТоварСЦенойНижеОпределенногоТипа();
			
				СтруктураПараметров = Новый Структура;
				СтруктураПараметров.Вставить("ВидРеализации"                      , Перечисления.ВидыСкидок.Оптовая);
				СтруктураПараметров.Вставить("СуммаДокумента"                     , мСуммаДокументаБезСкидок);
				СтруктураПараметров.Вставить("Карта"                              , Накладная.ДисконтнаяКарта);
				СтруктураПараметров.Вставить("УчитыватьНДС"                       , Накладная.УчитыватьНДС);
				СтруктураПараметров.Вставить("СуммаВключаетНДС"                   , НАкладная.СуммаВключаетНДС);
				СтруктураПараметров.Вставить("ВалютаРегламентированногоУчета"     , глЗначениеПеременной("ВалютаРегламентированногоУчета"));
				СтруктураПараметров.Вставить("УчетнаяПолитика",       мУчетнаяПолитика);
				
				ОбработкаТабличныхЧастей.РассчитатьСкидкиПриПродаже(Накладная, Накладная.Товары, СтруктураПараметров, мМинимальныеЦены, мСписокФильтров);
				ОбработкаТабличныхЧастей.РассчитатьСкидкиПриПродаже(Накладная, Накладная.Услуги, СтруктураПараметров, мМинимальныеЦены, мСписокФильтров);
			//Иначе
			//	СтрокаТабличнойЧасти = товар;
			//	ОбработкаТабличныхЧастей.ЗаполнитьЕдиницуЦенуПродажиТабЧасти(СтрокаТабличнойЧасти, Накладная,
			//		Накладная.ВалютаДокумента); 
			//КонецЕсли;
			
			Накладная.ЗаполнитьСчетаУчетаВСтрокеТабЧастиРегл(товар, "Товары", Истина, Истина);
			ОбработкаТабличныхЧастей.РассчитатьКоличествоМестТабЧасти(Товар, Накладная);
			ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Накладная);
			
			Заказ.Следующая();
		КонецЦикла;
		
		Если ЗаписатьЗаказ = 1 Тогда
			Если СохранитьЗаявку(Накладная,КодСотр,ПапкаОбмена) Тогда

				ОбработанныеЗаказы.Добавить();
				ОбработанныеЗаказы.CREATED = Ном;
				ОбработанныеЗаказы.Записать();	
				ПринятыеЗаявки.Добавить(Ном, Накладная.Номер);
				//СмирныхИА СКФ 07.06.2013
				Если НЕ Накладная.Проведен Тогда
					ОтправитьСообщениеКПК(КодСотр,"Заявка: "+Накладная.Номер+" на "+Накладная.Контрагент+" "+Накладная.АдресДоставки+" не проведена. Свяжитесь с оператором",ПапкаОбмена);
				КонецЕсли;
				//------------------------
			Иначе
				ОтправитьСообщениеКПК(КодСотр,"Не удалось загрузить заявку на "+Накладная.Контрагент+" "+Накладная.АдресДоставки+". Необходимо повторно отправить заявку",ПапкаОбмена);
			КонецЕсли;
		КонецЕсли;
		
		СохранитьПринятыеЗаявки(КодСотр, ПринятыеЗаявки, ПапкаОбмена);
		Заказ.ЗакрытьФайл();
		ЗакрытьФайлОбработанныхЗаявок(ОбработанныеЗаказы, КодСотр, ПапкаОбмена);
		
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
КонецПроцедуры

Функция НайтиАгента(КодАгента)
	Возврат Справочники.Пользователи.НайтиПоКоду(КодАгента);
КонецФункции

Функция ФайлОбработанныхЗаказов(кодМенеджера,ПапкаОбмена)
	имяФайла = ПапкаОбмена + "\POD" + СокрЛП(кодМенеджера) + ".DBF";
	темпФайл = ПапкаОбмена + "\TPOD.DBF";

	ДБФ = Новый XBase();
	файл = Новый Файл(имяФайла);
	
	Если файл.Существует() Тогда
		временныйФайл = ПапкаОбмена + "\TORD.DBF";
		ПереместитьФайл(имяФайла, темпФайл);	
		ДБФ.ОткрытьФайл(темпФайл);
	Иначе
		ДБФ.Поля.Добавить("CREATED", "S", 25, 0);	
		ДБФ.СоздатьФайл(темпФайл);
	КонецЕсли;
	
	ДБФ.Кодировка = КодировкаXBase.OEM;
	Возврат ДБФ;
КонецФункции    

Функция ЗагрузитьПринятыеЗаявки(КодАгента, ПоЗаявкам, ПапкаОбмена)
	имяФайла = ПапкаОбмена + "\ODB" + КодАгента + ".DBF";
	файл = Новый Файл(имяФайла);

	списокЗаявок = Новый СписокЗначений;
	
	Если файл.Существует() Тогда
		КопироватьФайл (имяФайла, ПапкаОбмена + "\TODB.DBF");
		ДБФ = Новый XBase();
		ДБФ.ОткрытьФайл(ПапкаОбмена + "\TODB.DBF");
		ДБФ.Кодировка = КодировкаXBase.OEM;
		
		ДБФ.Первая();
		Пока Не ДБФ.ВКонце() Цикл
			Если ПоЗаявкам Тогда
				списокЗаявок.Добавить(СокрЛП(ДБФ.ORDER), СокрЛП(ДБФ.DLV));
			Иначе
				списокЗаявок.Добавить(СокрЛП(ДБФ.DLV), СокрЛП(ДБФ.ORDER));
			КонецЕсли;
			ДБФ.Следующая();
		КонецЦикла;
		
		ДБФ.ЗакрытьФайл();
	КонецЕсли;
	
	Возврат списокЗаявок;
КонецФункции

Функция СохранитьЗаявку(Накладная,КодАгента,ПапкаОбмена)
	
	Если Накладная.Товары.Количество()=0 Тогда
		Возврат Истина;
	КонецЕсли;
	
	//СмирныхИА СКФ 23.07.2013 Ограничеваем по времени приема заявки
	Если Накладная.ДатаОтгрузки=НачалоДня(ТекущаяДата()) Тогда
		Запрос=Новый Запрос;
		Запрос.Текст="ВЫБРАТЬ ПЕРВЫЕ 1
					 |	МенеджерыДоговоров.ЗавершениеРаботыКПК КАК ВремяЗавершения
					 |ИЗ
					 |	РегистрСведений.МенеджерыДоговоров КАК МенеджерыДоговоров
					 |ГДЕ
					 |	МенеджерыДоговоров.Менеджер = &Менеджер
					 |	И МенеджерыДоговоров.Договор = &Договор";
		Запрос.УстановитьПараметр("Менеджер",Накладная.Ответственный);
		Запрос.УстановитьПараметр("Договор",Накладная.ДоговорКонтрагента);
		Выборка=Запрос.Выполнить().Выбрать();
		Если Выборка.Следующий() Тогда
			Если ЗначениеЗаполнено(Выборка.ВремяЗавершения) Тогда
				Если ТекущаяДата()-НачалоДня(ТекущаяДата())>=Выборка.ВремяЗавершения-НачалоДня(Выборка.ВремяЗавершения) Тогда
					ОтправитьСообщениеКПК(КодАгента,"Время приема заявок истекло. Заявка на "+Накладная.Контрагент+" "+Накладная.ДоговорКонтрагента+" не загружена",ПапкаОбмена);
					Возврат Ложь;
				КонецЕсли;
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
	
	Попытка
		Накладная.ОбменДанными.Загрузка=Истина;
		Накладная.Записать(РежимЗаписиДокумента.Запись);
	Исключение
	КонецПопытки;
	
	Отказ=Ложь;
	Накладная.СуммаДокумента=Накладная.Товары.Итог("Сумма");
	//ДополнительныйМПЗ.ПроверитьПревышениеДолга(Накладная.ДоговорКонтрагента,Накладная.МоментВремени(),Накладная.СуммаДокумента,Отказ);
	Если Отказ Тогда
		ОтправитьСообщениеКПК(КодАгента,"Превышен долг. Заявка на "+Накладная.Контрагент+" "+Накладная.ДоговорКонтрагента+" не проведена",ПапкаОбмена);
		Возврат НЕ Накладная.Ссылка.Пустая();
	КонецЕсли;
	
	Попытка
		Накладная.ОбменДанными.Загрузка=Ложь;
		Накладная.Записать(РежимЗаписиДокумента.Проведение, РежимПроведенияДокумента.Оперативный);
	Исключение
		ЗаписьЖурналаРегистрации("Ошибка проведения КПК",,Накладная.Ссылка,Накладная.Ссылка,ОписаниеОшибки());
	КонецПопытки;

	Если Накладная.Проведен Тогда
		НовыйДокумент=Документы.Отгрузка.СоздатьДокумент();
		НовыйДокумент.Дата=ТекущаяДата();
		НовыйДокумент.УстановитьНовыйНомер();
		
		НовыйДокумент.ЗаказПокупателя = Накладная.Ссылка;
		ЗаполнитьПоЗаказуПокупателя(НовыйДокумент);
		УправлениеЗапасами.ЗаполнитьСоставНабораПоОснованию(НовыйДокумент, Накладная);
		
		НовыйДокумент.ОбменДанными.Загрузка=Истина;
		НовыйДокумент.Комментарий=СтрЗаменить(НовыйДокумент.Комментарий,"Создана на КПК.","");
		НовыйДокумент.Записать();
	КонецЕсли;
	
	ЗаписалиНакладную=НЕ Накладная.Ссылка.Пустая();	
	Возврат ЗаписалиНакладную;
			
КонецФункции

Функция ПолучитьКассу(Организация)

	Запрос = Новый Запрос;
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	Кассы.Ссылка КАК Касса
		|ИЗ
		|	Справочник.Кассы КАК Кассы
		|ГДЕ
		|	Кассы.Владелец = &Владелец
		|	И НЕ Кассы.ПометкаУдаления
		|	И Кассы.ВидКассы = ЗНАЧЕНИЕ(Перечисление.ВидыКасс.Операционная)";

	Запрос.УстановитьПараметр("Владелец", Организация);

	Результат = Запрос.Выполнить();

	ВыборкаДетальныеЗаписи = Результат.Выбрать();

	Касса="";
	Пока ВыборкаДетальныеЗаписи.Следующий() Цикл
		Касса=ВыборкаДетальныеЗаписи.Касса;
	КонецЦикла;

	Возврат Касса;

КонецФункции

Функция СоздатьЗаказ(Заказ, Агент)
	
	Накладная = Документы.ЗаказПокупателя.СоздатьДокумент();
	
	Накладная.Дата = ТекущаяДата();
	Накладная.ДатаОтгрузки = Заказ.DATE;
	//Накладная.ДатаОплаты = Заказ.PAYDATE;
	//Накладная.ДоверенностьНаДеньги=Заказ.DVR_PAY=1;
	//Накладная.ДоверенносьтНаТМЦ=Заказ.DVR_RET=1;
	
	кгид = СокрЛП(Заказ.ID);
	табПоз = Найти(кгид, Символ(9));
	
	договор = Справочники.ДоговорыКонтрагентов.НайтиПоКоду(СокрЛП(Заказ.ID));
	
	контрагент = договор.Владелец;
	Накладная.Контрагент = контрагент;
	//Накладная.Грузополучатель = контрагент;
	Накладная.ДоговорКонтрагента = договор;
	Накладная.АдресДоставки = договор.АдресДоставки;
	Накладная.Маршрут = договор.Маршрут;
	
	организация = договор.Организация;
	Накладная.Организация = организация;

Функция СохранитьНакладную(накладная, код, Документ, СуммаДолга, НомерЗаказа)
	СуммаДолгаДок = 0;
	Если СуммаДолга > 0 Тогда
		Если СуммаДолга >= Документ.СуммаДокумента Тогда
			СуммаДолгаДок = Документ.СуммаДокумента;
		Иначе
			СуммаДолгаДок = СуммаДолга;	
		КонецЕсли;
		СуммаДолга = СуммаДолга - СуммаДолгаДок;
	КонецЕсли;
	
	Для Каждого товар из Документ.Товары Цикл
		накладная.Добавить();
		накладная.ID = код;
		накладная.NUM = Документ.Номер;
		накладная.DATE = Документ.Дата;
		накладная.ID_I = товар.Номенклатура.Код;
		накладная.QTY = товар.Количество;
		накладная.SUM = товар.Сумма * Документ.КурсВзаиморасчетов;
		накладная.SUMD = СуммаДолгаДок;
		накладная.CREATED = НомерЗаказа;
		накладная.Записать();
	КонецЦикла;
	
	Возврат СуммаДолга;
