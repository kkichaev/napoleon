	тз = ТипЗнч(эл);

	Возврат СокрЛП(эл.Код);

Процедура ВыгрузкаКонтрагентов()
	агентыТбл = ЗагрузитьАгентов();
		
	кагенты = СоздатьБазуПартнеров("TORGS.DBF");
	
	выгружен = Новый Массив;
	Для Каждого астр из Агенты Цикл
		текАгент = астр.Агент;
		ДобавитьАгента(агентыТбл, текАгент);
				
		партнеры = СоздатьБазуПартнеров("TO.DBF");
		Контакты = СоздатьБазуКонтактов();
		
		ка = Новый Массив;
		строкиКА = Контрагенты.НайтиСтроки(новый Структура("Агент", текАгент));
		Для Каждого кстр из строкиКА Цикл
			ка.Добавить(кстр.Контрагент);
		КонецЦикла;
		
		Запрос = Новый Запрос;
		Запрос.Текст = 
			"ВЫБРАТЬ
			|	Контрагенты.Ссылка КАК Ссылка
			|ИЗ
			|	Справочник.Контрагенты КАК Контрагенты
			|ГДЕ
			|	Контрагенты.Ссылка В ИЕРАРХИИ(&Ссылка)";
		Запрос.УстановитьПараметр("Ссылка", ка);
		выб = Запрос.Выполнить().Выбрать();
		Пока выб.Следующий() Цикл
			код = КодОбъекта(выб.Ссылка);			
			партнеры.Добавить();
			партнеры.ID = код;
			партнеры.NAME = выб.Ссылка.Наименование;
			партнеры.ADDRESS = АдресКонтрагента(выб.Ссылка);
			партнеры.COSTYPE = 0;
			партнеры.Записать();
			
			Если выгружен.Найти(код) = Неопределено Тогда
				выгружен.Добавить(Код);
				
				кагенты.Добавить();
				кагенты.ID = код;
				кагенты.NAME = выб.Ссылка.Наименование;
				кагенты.ADDRESS = АдресКонтрагента(выб.Ссылка);
				кагенты.COSTYPE = 0;
				кагенты.Записать();
			КонецЕсли;			
			списокКонтактов = ВыгрузитьКонтакты(выб.Ссылка);
			ЗаписатьКонтакты(контакты, списокКонтактов, код);
		КонецЦикла;
		
		ЗакрытьБазуПартнеров(партнеры, контакты, текАгент);
	КонецЦикла;
		
	кагенты.ЗакрытьФайл();
	ПереместитьФайл(ПапкаОбмена + "\TORGS.DBF", ПапкаОбмена + "\ORGS.DBF");
	
	ЗаписатьАгентов(агентыТбл);

	Если Документ.НалогообложениеНДС = Перечисления.ТипыНалогообложенияНДС.НеОблагаетсяНДС Тогда
		Товар.СтавкаНДС = УправлениеНебольшойФирмойПовтИсп.ПолучитьСтавкуНДСБезНДС();
	ИначеЕсли ЗначениеЗаполнено(Товар.Номенклатура.ВидСтавкиНДС) Тогда
		Товар.СтавкаНДС = Справочники.СтавкиНДС.СтавкаНДС(Товар.Номенклатура.ВидСтавкиНДС, Документ.Дата);
	Иначе
		Товар.СтавкаНДС = Справочники.СтавкиНДС.СтавкаНДС(Документ.Организация.ВидСтавкиНДСПоУмолчанию, Документ.Дата);
	КонецЕсли;
	
	СтруктураДанныеНоменклатуры = Новый Структура();
	СтруктураДанныеНоменклатуры.Вставить("Номенклатура", Товар.Номенклатура);
	СтруктураДанныеНоменклатуры.Вставить("Коэффициент", 1);
	СтруктураДанныеНоменклатуры.Вставить("Характеристика", Товар.Характеристика);
	СтруктураДанныеНоменклатуры.Вставить("ДатаОбработки", Документ.Дата);
	СтруктураДанныеНоменклатуры.Вставить("Организация", Документ.Организация);
	СтруктураДанныеНоменклатуры.Вставить("НалогообложениеНДС", Документ.НалогообложениеНДС);
	СтруктураДанныеНоменклатуры.Вставить("СуммаВключаетНДС", Документ.СуммаВключаетНДС);
	СтруктураДанныеНоменклатуры.Вставить("ВалютаДокумента", Документ.ВалютаДокумента);
	СтруктураДанныеНоменклатуры.Вставить("ВидЦен", ТипЦеныПрайс);
	СтруктураДанныеНоменклатуры.Вставить("СтавкаНДС", Товар.СтавкаНДС);
	
	Товар.Цена = ЦенообразованиеСервер.ПолучитьЦенуНоменклатурыПоВидуЦен(СтруктураДанныеНоменклатуры);
	
	Товар.Сумма = Товар.Цена * Товар.Количество;
	Товар.ЕдиницаИзмерения = Номенклатура.ЕдиницаИзмерения;
	Товар.КлючСвязи = Документ.Запасы.Количество() + 1;
	Товар.ТипНоменклатурыЗапас = Истина;
		
	ЗначенияРеквизитов = ОбщегоНазначения.ЗначенияРеквизитовОбъекта(Товар.Номенклатура, "ТипНоменклатуры, Вес, Объем, ЭтоНабор, СпособРасчетаЦеныНабора");
		
	ВесЕдиницыТовара = ЗначенияРеквизитов.Вес;
	Если ВесЕдиницыТовара <> Неопределено Тогда
		Товар.Вес = ВесЕдиницыТовара * Товар.Количество;
	КонецЕсли;
	ОбъемЕдиницыТовара = ЗначенияРеквизитов.Объем;
	Если ОбъемЕдиницыТовара <> Неопределено Тогда
		Товар.Объем = ОбъемЕдиницыТовара * Товар.Количество;
	КонецЕсли;

	
	СтавкаНДС = УправлениеНебольшойФирмойПовтИсп.ПолучитьЗначениеСтавкиНДС(Товар.СтавкаНДС);
	Товар.СуммаНДС = ?(Документ.СуммаВключаетНДС, 
							  		Товар.Сумма - (Товар.Сумма) / ((СтавкаНДС + 100) / 100),
							  		Товар.Сумма * СтавкаНДС / 100);
	Товар.Всего = Товар.Сумма + ?(Документ.СуммаВключаетНДС, 0, Товар.СуммаНДС);
КонецПроцедуры

Функция ЗагрузкаЗаявокАгента(временныйФайл, КодАгента, ПринятыеЗаказы, РезультатДок)

	ВидЦеныДокумента = Неопределено;

	Заказ.Кодировка = КодировкаXBase.OEM;
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				ЗаписатьДокумент(Документ, ВидЦеныДокумента);
				
				СчДок = СчДок + 1;
				СообщитьОДокументе(Документ, РезультатДок);
				
				ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);

			КонецЕсли;

			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				//Сообщить("Повторный заказ " + значЗаказ);
			Иначе			
				Документ = СоздатьЗаказ(Заказ, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ);
		КонецЕсли;

	КонецЦикла;

		ЗаписатьДокумент(Документ, ВидЦеныДокумента);
		
		СчДок = СчДок + 1;
		СообщитьОДокументе(Документ, РезультатДок);
		
		ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);

	Возврат СчДок;

	ДатаПо = ТекущаяДата();
	ДатаС = ДобавитьМесяц(ДатаПо, -ГлубинаВыгрузкиДолгов);
	
	Для Каждого астр из Агенты Цикл
		текАгент = астр.Агент;
		
		ПринятыеЗаказы = ЗагрузитьПринятыеДокументы(КодОбъекта(текАгент), Ложь);
		Отгрузки = СоздатьБазуОтгрузок();
		Долги = СоздатьБазуДолгов();			
		
		ка = Новый Массив;
		строкиКА = Контрагенты.НайтиСтроки(новый Структура("Агент", текАгент));
		Для Каждого кстр из строкиКА Цикл
			ка.Добавить(кстр.Контрагент);
		КонецЦикла;
		
		Запрос = Новый Запрос;
		Запрос.Текст = 
			"ВЫБРАТЬ
			|	РасчетыСПокупателямиОстатки.Контрагент КАК Контрагент,
			|	РасчетыСПокупателямиОстатки.СуммаОстаток КАК СуммаДолга,
			|	РасчетыСПокупателямиОстатки.Документ КАК Документ,
			|	ДОБАВИТЬКДАТЕ(РасчетыСПокупателямиОстатки.Документ.Дата, ДЕНЬ, РасчетыСПокупателямиОстатки.Договор.СрокОплатыПокупателя) КАК СрокОплаты,
			|	РасчетыСПокупателямиОстатки.ТипРасчетов КАК ТипРасчетов
			|ИЗ
			|	РегистрНакопления.РасчетыСПокупателями.Остатки КАК РасчетыСПокупателямиОстатки
			|ГДЕ
			|	РасчетыСПокупателямиОстатки.Контрагент В ИЕРАРХИИ(&Контрагенты)
			|
			|УПОРЯДОЧИТЬ ПО
			|	Контрагент";
			//"ВЫБРАТЬ
			//|	РасчетыСПокупателямиОстатки.Контрагент КАК Контрагент,
			//|	РасчетыСПокупателямиОстатки.СуммаОстаток КАК СуммаДолга,
			//|	РасчетыСПокупателямиОстатки.Документ КАК Документ,
			//|	ДОБАВИТЬКДАТЕ(РасчетыСПокупателямиОстатки.Документ.Дата, ДЕНЬ, РасчетыСПокупателямиОстатки.Договор.СрокОплатыПокупателя) КАК СрокОплаты,
			//|	РасчетыСПокупателямиОстатки.ТипРасчетов КАК ТипРасчетов
			//|ИЗ
			//|	РегистрНакопления.РасчетыСПокупателями.Остатки КАК РасчетыСПокупателямиОстатки
			//|ГДЕ
			//|	РасчетыСПокупателямиОстатки.Контрагент В ИЕРАРХИИ(&Контрагенты)
			//|	И РасчетыСПокупателямиОстатки.ТипРасчетов = ЗНАЧЕНИЕ(Перечисление.ТипыРасчетов.Долг)";


		Запрос.УстановитьПараметр("Контрагенты", ка);
		
		Результат = Запрос.Выполнить();
		Выборка = Результат.Выбрать(ОбходРезультатаЗапроса.Прямой);
		
		
		СуммаДолга = 0;
		ТекКА = Неопределено;
		Пока Выборка.Следующий() Цикл
			Если текКА <> Выборка.Контрагент Тогда
				Если СуммаДолга <> 0 Тогда
					Долги.Добавить();
					Долги.ID = КодОбъекта(текКА);
					Долги.SUM = СуммаДолга;
					Долги.NUM = "Нач.Сальдо";
					Долги.DATE = ДатаС;
					Долги.Записать();
				КонецЕсли;
				текКА = Выборка.Контрагент;
				СуммаДолга = 0;
			КонецЕсли;
			Если ЗначениеЗаполнено(Выборка.Документ) и ТипЗнч(Выборка.Документ) = Тип("ДокументСсылка.РасходнаяНакладная") Тогда
				ЗаписатьОтгрузку(Отгрузки, Выборка.Документ, Выборка.СуммаДолга, ПринятыеЗаказы, Выборка.СрокОплаты);
			Иначе	
				СуммаДолга = СуммаДолга + Выборка.СуммаДолга;
			КонецЕсли;
		
		КонецЦикла;
		
		Если СуммаДолга <> 0 Тогда
			Долги.Добавить();
			Долги.ID = КодОбъекта(текКА);
			Долги.SUM = СуммаДолга;
			Долги.NUM = "Нач.Сальдо";
			Долги.DATE = ДатаС;
			Долги.Записать();
		КонецЕсли;
		ЗакрытьБазуДолгов(текАгент, Долги, отгрузки);		
	КонецЦикла;
