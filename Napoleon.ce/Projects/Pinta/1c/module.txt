		Если текАгент <> стр.Агент Тогда

			видыЦенАгентов.Вставить(текАгент, видыЦен);

		Если стр.Контрагент = Null Тогда
			Продолжить;
		КонецЕсли;
		
		Если кодКонтрагента <> стр.Контрагент.Код Тогда

			//Наименование = стр.Контрагент.НаименованиеПолное;
			Наименование = стр.Контрагент.Наименование;
			
			кагенты.Добавить();
			кагенты.ID = кодКонтрагента;
			кагенты.NAME = Наименование;
			
			кагенты.ADDRESS = АдресКонтрагента(стр.Контрагент);
			
			кагенты.COSTYPE = 0;
			кагенты.Записать();
			
			Если выгружено.Найти(кодКонтрагента) = Неопределено Тогда
				выгружено.Добавить(кодКонтрагента);
				
				общийФайл.Добавить();
				общийФайл.ID = кодКонтрагента;
				общийФайл.NAME = Наименование;
				общийФайл.ADDRESS = кагенты.ADDRESS;
				общийФайл.Записать();
			КонецЕсли;
			
		КонецЕсли;

		ценаИнд = 0;
		Если стр.ТипЦен <> Null и не стр.ТипЦен.Пустая() Тогда
			тц = видыЦен.НайтиПоЗначению(стр.ТипЦен);
			Если тц = неопределено Тогда

		договоры.ID = кодКонтрагента;
		договоры.ID_DG = стр.Договор.Код;
		договоры.ID_ORG = стр.Организация.Код;

		договоры.NAME = стр.Организация.Наименование + " " + стр.Договор.Наименование;

	Если прм <> Неопределено Тогда
		рет.Контрагенты.Добавить(прм);	
	КонецЕсли;
	рет.Агенты = ЗаписатьАгентов(агенты, КодировкаБазы);

	рет.ОбщийФайл = ПоместитьВоВременноеХранилище(Новый ДвоичныеДанные(КаталогВременныхФайлов() + "TRGS.DBF"));

Функция ЗагрузитьПринятыеДокументы(КодАгента, ПоЗаявкам, ТипДок = Неопределено)
	Кодировка = ?(КодировкаБазыANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	списокЗаявок = Новый СписокЗначений;	
	Если ТипДок = Неопределено Тогда
		ТипДок = "Order";
	КонецЕсли;
	
	имяФайла = ПапкаОбмена + "\PRD" + КодАгента + ".DBF";
	файл = Новый Файл(имяФайла);
	Если файл.Существует() Тогда
		КопироватьФайл (имяФайла, ПапкаОбмена + "\TPRD.DBF");
		ДБФ = Новый XBase();
		ДБФ.ОткрытьФайл(ПапкаОбмена + "\TPRD.DBF");
		//ДБФ.Кодировка = КодировкаXBase.OEM;
		ДБФ.Кодировка = Кодировка;
		
		ДБФ.Первая();
		Пока Не ДБФ.ВКонце() Цикл
			Если СокрЛП(ДБФ.TYPE) = ТипДок Тогда
				Если ПоЗаявкам Тогда
					списокЗаявок.Добавить(СокрЛП(ДБФ.CREATED), СокрЛП(ДБФ.NUMBER));
				Иначе
					списокЗаявок.Добавить(СокрЛП(ДБФ.NUMBER), СокрЛП(ДБФ.CREATED));
				КонецЕсли;
			КонецЕсли;

Функция ЗагрузкаЗаявокАгента(АдресЗаказ, КодАгента, ПринятыеЗаказы,КодировкаANSI)
	Кодировка = ?(КодировкаANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(КодАгента);
	Документ = Неопределено;
	ВидЦеныДокумента = Неопределено;
	
	СчДок = 0;
	
	Заказ_имя = КаталогВременныхФайлов() + "TORD.dbf";
	ЗаказДД = ПолучитьИзВременногоХранилища(АдресЗаказ);
	ЗаказДД.Записать(Заказ_имя);
	Заказ = Новый XBase(Заказ_имя);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Заказ.Кодировка = Кодировка;
	
	КодЗаказа = "";
	
	Заказ.Первая();		
	Пока Заказ.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Заказ.ORDFLAG <> 0 Тогда
			//
			// Запишем предыдущий документ
			//
			Если Документ <> Неопределено Тогда
				Если ЗаписатьДокумент(Документ, ВидЦеныДокумента) Тогда
					СчДок = СчДок + 1;
					СообщитьОДокументе(Документ);
					ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
				КонецЕсли;
				Документ = Неопределено;	
			КонецЕсли;
			
			КодЗаказа = СокрЛП(Заказ.CREATED);
			значЗаказ = ПринятыеЗаказы.НайтиПоЗначению(КодЗаказа);
			Если значЗаказ <> Неопределено Тогда			
				СообщитьИнт("Повторный заказ " + значЗаказ);
			Иначе			
				код = Заказ.ID;
				Документ = СоздатьЗаказ(Заказ, код, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если Документ <> Неопределено Тогда
			ДобавитьТовар(Документ, Заказ, ВидЦеныДокумента);
		КонецЕсли;
		
		Заказ.Следующая();
	КонецЦикла;
	
	Если Документ <> Неопределено Тогда
		Если ЗаписатьДокумент(Документ, ВидЦеныДокумента) Тогда
			СчДок = СчДок + 1;
			СообщитьОДокументе(Документ);
			ПринятыеЗаказы.Добавить(КодЗаказа, Документ.Номер);
		КонецЕсли;
	КонецЕсли;
	
	Заказ.ЗакрытьФайл();
	
	Возврат СчДок;
КонецФункции

Функция ЗагрузкаИнкассацийАгента(АдресДок, КодАгента, ПринятыеДок, КодировкаANSI)
	Кодировка = ?(КодировкаANSI,КодировкаXBase.ANSI,КодировкаXBase.OEM); 
	
	Агент = НайтиАгентаПоКоду(КодАгента);
	Документ = Неопределено;
	
	СчДок = 0;
	
	Док_имя = КаталогВременныхФайлов() + "TINC.dbf";
	ДокД = ПолучитьИзВременногоХранилища(АдресДок);
	ДокД.Записать(Док_имя);
	Док = Новый XBase(Док_имя);
	
	//Заказ.Кодировка = КодировкаXBase.OEM;
	Док.Кодировка = Кодировка;
	
	КодДок = "";
	ТекСум = 0;
	КодОрг = "";
	ТекДок = Неопределено;
	
	Док.Первая();		
	Пока Док.ВКонце() = 0 Цикл			
		//
		// Проверим признак начала заявки
		//
		Если Док.ORDFLAG <> 0 или ПустаяСтрока(Док.NUM_I) Тогда
			//
			// Запишем предыдущий документ
			//
			Если ТекДок <> Неопределено Тогда
				Если ТекСум > 0 Тогда
					ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
				КонецЕсли;
				СчетаУчетаВДокументах.Заполнить(ТекДок, Новый Структура);
				Если ЗаписатьДокументСервер(ТекДок) Тогда
					СообщитьОДокументе(ТекДок);
					ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
					ЗаписатьДокументСервер(ТекДок, Истина);
				КонецЕсли;
			КонецЕсли;
			
			ТекДок = Неопределено;
			ТекСум = Док.SUM;
			КодДок = СокрЛП(Док.CREATED);
			значДок = ПринятыеДок.НайтиПоЗначению(КодДок);
			Если значДок <> Неопределено Тогда			
				СообщитьИнт("Повторный документ " + значДок);
				ТекСум = 0;
			Иначе
				ТекДок = СоздатьИнкассацию(Док, Агент);
			КонецЕсли;
		КонецЕсли;
		
		Если ТекДок <> Неопределено Тогда 
			КодОрг = Док.ID;
			Сумма = Док.SUM_I;
			ДокОсн = Документы.РеализацияТоваровУслуг.ПустаяСсылка();
			Если ПустаяСтрока(Док.NUM_I) Тогда
				Сумма = Док.SUM;
			Иначе
				ДокОсн = Документы.РеализацияТоваровУслуг.НайтиПоНомеру(Док.NUM_I, Док.DATE_I);
			КонецЕсли;
			
			ДобавитьДокументОснование(ТекДок, ДокОсн, Сумма);
			ТекСум = ТекСум - Сумма;
		КонецЕсли;	
		Док.Следующая();
	КонецЦикла;
	
	Если ТекДок <> Неопределено Тогда
		Если ТекСум > 0 Тогда
			ДобавитьДокументОснование(ТекДок, Документы.РеализацияТоваровУслуг.ПустаяСсылка(), ТекСум);
		КонецЕсли;
		СчетаУчетаВДокументах.Заполнить(ТекДок, Новый Структура);
		Если ЗаписатьДокументСервер(ТекДок) Тогда
			СообщитьОДокументе(ТекДок);
			ПринятыеДок.Добавить(КодДок, ТекДок.Номер);
			ЗаписатьДокументСервер(ТекДок, Истина);
		КонецЕсли;
	КонецЕсли;
	
	Док.ЗакрытьФайл();
	
	Возврат СчДок;
