Процедура СохранитьПринятыеДокументы(ПапкаОбмена, КодАгента, списокДок, ТипДок)
	новФайл = ПапкаОбмена + "PRD" + КодАгента + ".DBF";
	имяФайла = ПапкаОбмена + "TPRD.DBF";
	
	ДБФ = Новый XBase();
	ДБФ.Поля.Добавить("CREATED",   "S", 20, 0);
	ДБФ.Поля.Добавить("NUMBER", "S", 40, 0);
	ДБФ.Поля.Добавить("TYPE", "S", 20, 0);
	ДБФ.СоздатьФайл(имяФайла);
	ДБФ.Кодировка = КодировкаXBase.OEM;
	
	ДатаНач = ДобавитьМесяц(ТекущаяДата(), -1);
	нач = Формат(ДатаНач, "ДФ=""ггггММдд000000""");
	
	Если ТипДок = Неопределено Тогда
		ТипДок = "Order";
	КонецЕсли;
	
	датаПров = НачалоМесяца(ДобавитьМесяц(ТекущаяДата(), -1));
	
	файл = Новый Файл(новФайл);
	Если файл.Существует() Тогда
		Ист = Новый XBase();
		врФайл = ПапкаОбмена + "\TPRDSC.DBF";
		КопироватьФайл (новФайл, врФайл);
		Ист.ОткрытьФайл(врФайл);
		Пока Не Ист.ВКонце() Цикл
			Если Дата(СокрЛП(Ист.CREATED)) > датаПров и СокрЛП(Ист.TYPE) <> ТипДок Тогда
				ДБФ.Добавить();
				ЗаполнитьЗначенияСвойств(ДБФ, Ист, "CREATED,NUMBER,TYPE");
				ДБФ.Записать();         
			КонецЕсли;
			Ист.Следующая();
		КонецЦикла;
		Ист.ЗакрытьФайл();
	КонецЕсли;
	
	Для Каждого стр из списокДок Цикл
		Если стр.Значение > нач Тогда
			ДБФ.Добавить();
			ДБФ.CREATED = стр.Значение;
			ДБФ.NUMBER = стр.Представление;
			ДБФ.TYPE = ТипДок;
			ДБФ.Записать();         
		КонецЕсли;
	КонецЦикла;
	
	ДБФ.ЗакрытьФайл();

	ПереместитьФайл(имяФайла, новФайл);

Функция СоздатьНакладную(ДокИст)
	Накладная = Документы.РеализацияТоваровУслуг.СоздатьДокумент();
	
	Накладная.Ответственный = 
		УправлениеПользователями.ПолучитьЗначениеПоУмолчанию(глЗначениеПеременной("глТекущийПользователь"), "ОсновнойОтветственный");
	
	Накладная.Дата = ДокИст.DATE;
	Накладная.ВидОперации = Перечисления.ВидыОперацийРеализацияТоваров.ПродажаКомиссия;
		
	контрагент = Справочники.Контрагенты.НайтиПоКоду(СокрЛП(ДокИст.ID));
	
	Накладная.Грузополучатель = контрагент;
	Накладная.Контрагент = контрагент;
	Накладная.ДоговорКонтрагента = Контрагент.ОсновнойДоговорКонтрагента;
	Накладная.Организация = Справочники.Организации.НайтиПоКоду(СокрЛП(ДокИст.FIRMA));
	Если Накладная.Организация = Неопределено или Накладная.Организация.Пустая() Тогда
		Накладная.Организация = Накладная.ДоговорКонтрагента.Организация;
	КонецЕсли;
	Если Накладная.ДоговорКонтрагента.Организация <> Накладная.Организация Тогда
		выб = Справочники.ДоговорыКонтрагентов.Выбрать(, Контрагент.Ссылка);
		Пока выб.Следующий() Цикл
			Если (Не выб.ПометкаУдаления) и выб.Организация.Ссылка = Накладная.Организация.Ссылка Тогда
				Накладная.ДоговорКонтрагента = выб.Ссылка;
				Прервать;	
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	
	Накладная.БанковскийСчетОрганизации = Накладная.Организация.ОсновнойБанковскийСчет;
	Накладная.ОтражатьВУправленческомУчете = Истина;
	
	Накладная.Комментарий = Накладная.Комментарий + " " + ДокИст.REMARK;
	
	Накладная.ТипЦен = Накладная.ДоговорКонтрагента.ТипЦен;
	Накладная.ВалютаДокумента = Накладная.ДоговорКонтрагента.ВалютаВзаиморасчетов;
	
	СтруктураКурсаДокумента = МодульВалютногоУчета.ПолучитьКурсВалюты(Накладная.ВалютаДокумента, Накладная.Дата);
	Накладная.КратностьВзаиморасчетов = СтруктураКурсаДокумента.Кратность;
	Накладная.КурсВзаиморасчетов = СтруктураКурсаДокумента.Курс;
	
	Накладная.УчитыватьНДС = УчетНДС.ОрганизацияУчитываетНДС(Накладная.Организация, ?(ЗначениеЗаполнено(Накладная.Дата), Накладная.Дата, ТекущаяДата()));
	Накладная.СуммаВключаетНДС = Истина;
	
	Накладная.ВидПередачи = Перечисления.ВидыПередачиТоваров.СоСклада;
	
	//Накладная.Склад = Склад;
	
	Возврат Накладная;

Процедура SalesHandler(КодАгента) Экспорт
	ПапкаОбмена = ПапкаОбмена();
	имяФайла = ПапкаОбмена + "SLS" + КодАгента + ".DBF";
	
	временныйФайл = ПапкаОбмена + "TSLS.DBF";
	ПереместитьФайл(имяФайла, временныйФайл);
	
	ДокРеал = Новый XBase(временныйФайл);
	ДокРеал.Кодировка = КодировкаXBase.OEM;
	ОбработанныеДокументы = ФайлОбработанныхДокументов(ПапкаОбмена, КодАгента);
	ПринятыеДок = ЗагрузитьПринятыеДокументы(ПапкаОбмена, КодАгента, "Sales");
	
	ЗаписатьДокумент = 0;
	Накладная = Неопределено;
	КодДок = "";
	
	ДокРеал.Первая();		
	Пока ДокРеал.ВКонце() = 0 Цикл
		//
		// Проверим признак начала заявки
		//
		Если ДокРеал.ORDFLAG <> 0 Тогда
			Если ЗаписатьДокумент = 1 Тогда
				ЗаписатьРеализацию(Накладная, КодДок, ПринятыеДок, ОбработанныеДокументы)
			КонецЕсли;
			
			ЗаписатьДокумент = 0;
			КодДок = СокрЛП(ДокРеал.CREATED);
			Если ПринятыеДок.НайтиПоЗначению(КодДок) = Неопределено Тогда
				ЗаписатьДокумент = 1;
				Накладная = СоздатьНакладную(ДокРеал);
			КонецЕсли;
		КонецЕсли;
		
		Если ЗаписатьДокумент = 0 Тогда
			ДокРеал.Следующая();
			Продолжить;
		КонецЕсли;
		
		номенклатура = Справочники.Номенклатура.НайтиПоКоду(ДокРеал.ID_I);
		товар = Накладная.Товары.Добавить();
		товар.Номенклатура = номенклатура.Ссылка;
		
		товар.ЕдиницаИзмерения = номенклатура.ЕдиницаХраненияОстатков;
		товар.Количество = ДокРеал.QTY;
		товар.Коэффициент = 1;
		ОбработкаТабличныхЧастей.ЗаполнитьЦенуПродажиТабЧасти(товар, Накладная, Накладная.ВалютаДокумента);
		
		товар.ЕдиницаИзмеренияМест = товар.ЕдиницаИзмерения;
		Если ДокРеал.PACK <> 0 Тогда
			едВыб = НайтиУпаковку(номенклатура);
			Если едВыб <> Неопределено Тогда
				товар.ЕдиницаИзмеренияМест = едВыб.Ссылка;
			КонецЕсли;
		КонецЕсли;
		
		ОбработкаТабличныхЧастей.РассчитатьКоличествоМестТабЧасти(товар, Накладная);
				
		товар.СпособСписанияОстаткаТоваров = Перечисления.СпособыСписанияОстаткаТоваров.СоСклада;
		товар.СтавкаНДС = ?(ДокРеал.TAX_I = 0, Перечисления.СтавкиНДС.БезНДС, номенклатура.СтавкаНДС);
		
		товар.Сумма = ДокРеал.SUM_I;
		товар.СуммаНДС = ДокРеал.TAX_I;
		товар.ХарактеристикаНоменклатуры = Неопределено;
		Если товар.Количество > 0 Тогда
			товар.Цена = товар.Сумма / товар.Количество;
		Иначе
			товар.Цена = 0;
		КонецЕсли;
		товар.Склад = Накладная.Склад;
		
		ДокРеал.Следующая();
	КонецЦикла;
	
	Если ЗаписатьДокумент = 1 Тогда
		ЗаписатьРеализацию(Накладная, КодДок, ПринятыеДок, ОбработанныеДокументы)
	КонецЕсли;
	
	ДокРеал.ЗакрытьФайл();
	УдалитьФайлы(временныйФайл);
	СохранитьОбработанныеДокументы(ПапкаОбмена, КодАгента, ОбработанныеДокументы);
	СохранитьПринятыеДокументы(ПапкаОбмена, КодАгента, ПринятыеДок, "Sales");
КонецПроцедуры
