Процедура ЗаполнитьПодразделения()
	подключение = СоздатьПодключение();
	Если подключение = "" Тогда
		Возврат;
	КонецЕсли;

	Попытка
		rs  = СоздатьОбъект("ADODB.RecordSet");
		rs.CursorLocation = 3; // on client 
		stmt = "select id, name from Division order by name";
		rs.Open(stmt, подключение, 2, 4);
	
		Пока rs.EOF = 0 Цикл
			Подразделение.ДобавитьЗначение(rs.Fields("id").value, rs.Fields("name").value);
			rs.MoveNext();
		КонецЦикла;
		
		rss  = СоздатьОбъект("ADODB.RecordSet");
		rss.CursorLocation = 3; // on client 
		stmt = "select id, division from Sklads where base='" + ИДБазы + "' order by division, index";
		rss.Open(stmt, подключение, 2, 4);
	
		текП = "";
		скл = СоздатьОбъект("Справочник.МестаХранения");
		спЗнач = "";
		Пока rss.EOF = 0 Цикл
			Если текП <> rss.Fields("division").value Тогда
				Если текП <> "" Тогда
					СкладыПодразделений.НоваяСтрока();
					СкладыПодразделений.Подразделение = текП;
					СкладыПодразделений.Склады = спЗнач;
				КонецЕсли;
				
				текП = rss.Fields("division").value;
				спЗнач = СоздатьОбъект("СписокЗначений");
			КонецЕсли;
			
			id = rss.Fields("id").value;
			рзд = Найти(id, "|");
			id = Прав(id, СтрДлина(id)-рзд);
			Если скл.НайтиПоКоду(id) = 1 Тогда
				спЗнач.ДобавитьЗначение(скл.ТекущийЭлемент());
			КонецЕсли;
			rss.MoveNext();
		КонецЦикла;		
		Если текП <> "" Тогда
			СкладыПодразделений.НоваяСтрока();
			СкладыПодразделений.Подразделение = текП;
			СкладыПодразделений.Склады = спЗнач;
		КонецЕсли;

		Если Подразделение.РазмерСписка() > 0 Тогда
			Подразделение.ТекущаяСтрока(1);
			текПодразделение = Подразделение.ПолучитьЗначение(Подразделение.ТекущаяСтрока());
			СкладыВСписок(текПодразделение);
		КонецЕсли;
	Исключение	
		
	КонецПопытки;

Функция глРазложить(Знач Стр, Разделитель = ",") Экспорт
	СЗ = СоздатьОбъект("СписокЗначений");
	Если Разделитель = " " Тогда
		Стр = СокрЛП(Стр);
		Пока 1=1 Цикл
			Поз = Найти(Стр,Разделитель);
			Если Поз=0 Тогда
				СЗ.ДобавитьЗначение(Стр);
				Возврат СЗ;
			КонецЕсли;
			СЗ.ДобавитьЗначение(Лев(Стр,Поз-1));
			Стр = СокрЛ(Сред(Стр,Поз));
		КонецЦикла;
	Иначе
		ДлинаРазделителя = СтрДлина(Разделитель);
		Пока 1=1 Цикл
			Поз = Найти(Стр,Разделитель);
			Если Поз=0 Тогда
				СЗ.ДобавитьЗначение(Стр);
				Возврат СЗ;
			КонецЕсли;
			СЗ.ДобавитьЗначение(Лев(Стр,Поз-1));
			Стр = Сред(Стр,Поз+ДлинаРазделителя);
		КонецЦикла;
	КонецЕсли;
