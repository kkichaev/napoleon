Функция АдресКонтрагента(Контрагент)
	Адреса = РегистрыСведений.КонтактнаяИнформация.СоздатьНаборЗаписей();
	УправлениеКонтактнойИнформацией.ПрочитатьКонтактнуюИнформацию(Адреса, Контрагент.Ссылка);
	
	АдресСтр = "";
	
	Попытка
		Для каждого Адрес из Адреса Цикл
			Если ТипЗнч(Адрес.Вид) = Тип("Строка") Тогда
				Продолжить;
			КонецЕсли;
			Если Найти(Адрес.Вид.Наименование, "доставки") > 0 Тогда
				Если Адрес.Представление <> "" Тогда
					Возврат Адрес.Представление;
				КонецЕсли;
				Возврат УправлениеКонтактнойИнформацией.ПолучитьПредставлениеАдреса(Адрес);
			КонецЕсли;
			Если Найти(Адрес.Вид.Наименование, "Фактический") > 0 Тогда
				Если Адрес.Представление <> "" Тогда
					Возврат Адрес.Представление;
				КонецЕсли;
				Возврат УправлениеКонтактнойИнформацией.ПолучитьПредставлениеАдреса(Адрес);
			КонецЕсли;
			Если Найти(Адрес.Вид.Наименование, "Юридический") > 0 Тогда
				Если Адрес.Представление <> "" Тогда
					АдресСтр = Адрес.Представление;
				Иначе
					АдресСтр = УправлениеКонтактнойИнформацией.ПолучитьПредставлениеАдреса(Адрес);
				КонецЕсли;
			КонецЕсли;
		КонецЦикла;
	Исключение
		Сообщить("ошибка АдресКонтрагента " + Контрагент + ": " + ОписаниеОшибки());
	КонецПопытки;
	
	Возврат АдресСтр;

		|
		|УПОРЯДОЧИТЬ ПО
		|	Склад";

Процедура СохранитьЗаказ(ord, ordItems, Документ, АдресКод, текАгент, ADOСоединение)
	
	number = СокрЛП(Документ.Номер);
	userid = КодАгента(текАгент);
	
	КодОрг = СокрЛП(Документ.Контрагент.Код) + Символ(9);
	Если Не ПустаяСтрока(АдресКод) Тогда
		КодОрг = КодОрг + АдресКод;	
	КонецЕсли;
	
	ord.AddNew();
	ord.Fields("id").value = КодОрг;
	ord.Fields("userid").value = userid;
	датаСтр = Формат(((Документ.Дата - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	ord.Fields("created").value = датаСтр;
	
	ord.Fields("date").value = датаСтр;
	ord.Fields("docNumber").value = number;
	
	опл = Документ.Дата + Документ.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 24 * 3600;
	ord.Fields("payDate").value = Формат(((опл - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	ord.UpdateBatch();
	
	Для Каждого товар из Документ.Товары Цикл
		ordItems.AddNew();
		ordItems.Fields("id").value = товар.Номенклатура.Код;
		ordItems.Fields("qty").value = товар.Количество;
		ordItems.Fields("cost").value = товар.Цена * Документ.КурсВзаиморасчетов;
		ordItems.Fields("LoadedOrders$docNumber").value = number;
		
		ordItems.UpdateBatch();
	КонецЦикла;
	
	ТекстЗапроса = "
	|BEGIN TRANSACTION Update_Dlv;
	|
	|DELETE FROM LoadedOrders WHERE docNumber='" + number + "';
	|INSERT INTO LoadedOrders (id,created,date,payDate,userid,docNumber)
	|   select id,created,date,payDate,userid,docNumber FROM #tOrd;
	|INSERT INTO LoadedOrders$items (id, qty, cost, LoadedOrders$docNumber)
	|  SELECT id, qty, cost, LoadedOrders$docNumber FROM #tOrdItems;
	|COMMIT TRANSACTION Update_Dlv;
	|
	|delete from #tOrd;
	|delete from #tOrdItems;
	|if @@trancount > 0 commit;
	|";	

	
	ВыполнитьЗапрос(ADOСоединение, ТекстЗапроса);

Функция СохранитьНакладную(dlv, dlvItems, Документ, долг, ADOСоединение)
	
	докДолг = ?(Документ.СуммаДокумента < долг, Документ.СуммаДокумента, долг);
	Если докДолг < 0 Тогда
		докДолг = 0;	
	КонецЕсли;
	
	номер = СокрЛП(Документ.Номер);
	датаСтр = Формат(((Документ.Дата - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	КодОрг = Строка(Документ.Контрагент.УникальныйИдентификатор());
	
	dlv.AddNew();
	dlv.Fields("id").value = КодОрг;
	dlv.Fields("number").value = номер;
	dlv.Fields("sumD").value = докДолг;
	dlv.Fields("date").value = датаСтр;
	
	опл = Документ.Дата + Документ.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 24 * 3600;
	dlv.Fields("payDate").value = Формат(((опл - Дата(1970,1,1,0,0,0)) + 11644473600) * 10000000, "ЧГ=0");
	
	dlv.UpdateBatch();
	
	Для Каждого товар из Документ.Товары Цикл
		dlvItems.AddNew();
		dlvItems.Fields("id").value = товар.Номенклатура.Код;
		dlvItems.Fields("qty").value = товар.Количество;
		dlvItems.Fields("sum").value = товар.Сумма * Документ.КурсВзаиморасчетов;
		dlvItems.Fields("Delivery$id").value = КодОрг;
		dlvItems.Fields("Delivery$number").value =номер;
		dlvItems.Fields("Delivery$date").value = датаСтр;
		
		dlvItems.UpdateBatch();
	КонецЦикла;
		
	ТекстЗапроса = "
	|BEGIN;
	|
	|DELETE FROM ""Delivery"" WHERE ""number""='" + номер + "' and ""id""='" + КодОрг + "' and ""date""=" + датаСтр + ";
	|INSERT INTO ""Delivery"" (""id"",""number"",""sumD"",""date"",""payDate"")
	|   select ""id"",""number"",""sumD"",""date"",""payDate"" FROM tDlv;

	|  SELECT ""id"", ""qty"", ""sum"", ""Delivery$id"", ""Delivery$number"", ""Delivery$date"" FROM tDlvI;
	|delete from tDlv;
	|delete from tDlvI;	
	|COMMIT;";
	
	ВыполнитьЗапрос(ADOСоединение, ТекстЗапроса);	
	Возврат долг - докДолг;
