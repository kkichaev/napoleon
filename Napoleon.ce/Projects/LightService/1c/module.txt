Процедура ПроверитьИнформациюПоМенеджеру(МенеджерDBF, Агент) 
	МенеджерDBF.Первая();
	пКод = КодАгента(Агент);
	Пока МенеджерDBF.ВКонце() = 0 Цикл
		Если СокрЛП(МенеджерDBF.ID) = пКод Тогда
			МенеджерDBF.NAME = Агент.Наименование;
			МенеджерDBF.Записать();
			Возврат;
		КонецЕсли;
		МенеджерDBF.Следующая();
	КонецЦикла;
	
	МенеджерDBF.Добавить();
	МенеджерDBF.ID = пКод;
	МенеджерDBF.NAME = Агент.Наименование;
	
	//
	// В качестве пароля и логина для КПК назначим количество записей + 1. Так для первого пользователя
	// логин и пароль будет 1/1, для второго 2/2 и т.д.
	//
	МенеджерDBF.PASSWORD = МенеджерDBF.КоличествоЗаписей() + 1;
	МенеджерDBF.LOGIN = МенеджерDBF.КоличествоЗаписей() + 1;
	
	МенеджерDBF.Записать();

Процедура ВыгрузитьМаршрут(Маршрут, Выборка, КодОрг)
	ДеньНедели = Выборка.Маршрут.ДеньНедели;
	ДеньНедели2 = "";
	
	Если Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.Каждая1НеделяМесяца Тогда
		ДеньНедели = "1"+ДеньНедели;
	ИначеЕсли Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.Каждая2НеделяМесяца Тогда
		ДеньНедели = "2"+ДеньНедели;
	ИначеЕсли Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.Каждая3НеделяМесяца Тогда
		ДеньНедели = "3"+ДеньНедели;
	ИначеЕсли Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.Каждая4НеделяМесяца Тогда
		ДеньНедели = "4"+ДеньНедели;
	ИначеЕсли Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.Каждая5НеделяМесяца Тогда
		ДеньНедели = "1"+ДеньНедели;
	ИначеЕсли Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.НечетныеНедели Тогда
		ДеньНедели2 = "3"+ДеньНедели;
		ДеньНедели = "1"+ДеньНедели;
	ИначеЕсли Выборка.Маршрут.ВидЦикла = Перечисления.ВидыЦикличностиМаршрутов.ЧетныеНедели Тогда
		ДеньНедели2 = "4"+ДеньНедели;
		ДеньНедели = "2"+ДеньНедели;
	КонецЕсли;
	
	Маршрут.Добавить();
	Маршрут.ID = КодОрг;
	Маршрут.NAME = ДеньНедели;
	Маршрут.Записать();
	
	Если Не ПустаяСтрока(ДеньНедели2) Тогда
		Маршрут.Добавить();
		Маршрут.ID = КодОрг;
		Маршрут.NAME = ДеньНедели2;
		Маршрут.Записать();
	КонецЕсли;

		КонецЕсли;

		ВалютаТекущая = ТипЦен.ВалютаЦены;
		ВалютаБазовая = ТипЦен.БазовыйТипЦен.ВалютаЦены;
		
		Если ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоПроцентнойНаценкеНаБазовыйТип Тогда
			ПолученнаяЦена = ЦенаБазовая * (1 + стр.Значение.ПроцентСкидкиНаценки / 100);
			ВалютаТекущая = ВалютаБазовая;
		ИначеЕсли ТипЦен.СпособРасчетаЦены = Перечисления.СпособыРасчетаЦены.ПоВхождениюБазовойЦеныВДиапазон Тогда
			ПолученнаяЦена = Ценообразование.ПолучитьЦенуПоЦеновымДиапазонам(ТипЦен, Дата, ЦенаБазовая, ВалютаТекущая, ВалютаБазовая);
		КонецЕсли;
		
		ПолученнаяЦена = Ценообразование.ПересчитатьЦенуПриИзмененииВалюты(ПолученнаяЦена, ВалютаТекущая, ВалютаТекущая, 0, 1, Дата);
		ПолученнаяЦена = Ценообразование.ОкруглитьЦену(ПолученнаяЦена, ТипЦен.ПорядокОкругления, ТипЦен.ОкруглятьВБольшуюСторону);
		
		стрТовара.Цены.Вставить(стр.Представление, ПолученнаяЦена);

	КонецЦикла;
	
	цены.Очистить();

Функция СоздатьЗаказ(Заказ, Агент, можетМенятьЦену)
	
	параметрыДоговора = ЗаполнениеДокументов.ПолучитьСтруктуруПараметровДляПолученияДоговораЗаказаПокупателя();
	
	Накладная = Документы.ЗаказПокупателя.СоздатьДокумент();
	
	Накладная.Дата = ТекущаяДата();
	Накладная.ДатаОтгрузки = Заказ.DATE;
	Накладная.АдресДоставки = Заказ.ADDRESS;
	//Накладная.ДатаОплаты = Заказ.PAYDATE;
	
	//Накладная.ИнициализироватьНовыйДокумент(Неопределено, Неопределено);
	
	кодКА = Заказ.ID;
	рзд = Найти(КодКА, Символ(9));
	Если рзд > 0 Тогда
		кодКА = Лев(КодКА, рзд);
	КонецЕсли;
	контрагент = Справочники.Контрагенты.НайтиПоКоду(КодКА);
	договор = Справочники.ДоговорыКонтрагентов.НайтиПоКоду(СокрЛП(Заказ.ID_D));
	Если договор = Неопределено или договор.Пустая() Тогда
		договор = Контрагент.ОсновнойДоговорКонтрагента;
	КонецЕсли;
	Накладная.УчитыватьНДС = Истина;
	Накладная.СуммаВключаетНДС = Истина;
	
	Накладная.Грузополучатель = контрагент;
	Накладная.Контрагент = контрагент;
	
	Накладная.Организация = договор.Организация;
	Накладная.ДоговорКонтрагента = договор;
	Накладная.ТипЦен = Накладная.ДоговорКонтрагента.ТипЦен;
	Накладная.КаналПоступленияЗАказа = Перечисления.КаналПоступленияЗаказа.МобильнаяТорговля;
	
	Если можетМенятьЦену Тогда
		тц = Заказ.CTYPE;
		рзд = Найти(тц, Символ(9));
		Если рзд > 0 Тогда
			кодТипаЦены = СокрЛП(Прав(тц, СтрДлина(тц) - рзд+1));
			Накладная.ТипЦен = Справочники.ТипыЦенНоменклатуры.НайтиПоКоду(кодТипаЦены);
		Иначе
			Накладная.ТипЦен = Справочники.ТипыЦенНоменклатуры.НайтиПоНаименованию(тц);
		КонецЕсли;
	КонецЕсли;
	
	//ЗаполнениеДокументов.ПриИзмененииЗначенияКонтрагента(Накладная, параметрыДоговора);
	
	Если ЗначениеЗаполнено(Накладная.ДоговорКонтрагента) Тогда
		Накладная.ДатаОплаты = Накладная.ДатаОтгрузки + Накладная.ДоговорКонтрагента.ДопустимоеЧислоДнейЗадолженности * 86400;
	КонецЕсли;
	
	Если ЗначениеЗаполнено(Накладная.ДоговорКонтрагента) Тогда
		Накладная.ВалютаДокумента = Накладная.ДоговорКонтрагента.ВалютаВзаиморасчетов;
	Иначе
		Накладная.ВалютаДокумента = глЗначениеПеременной("ВалютаРегламентированногоУчета");;
	КонецЕсли;
	Курс = МодульВалютногоУчета.ПолучитьКурсВалюты(Накладная.ВалютаДокумента, Накладная.Дата);
	Накладная.КурсВзаиморасчетов = Курс.Курс;
	Накладная.КратностьВзаиморасчетов = Курс.Кратность;
	
	Накладная.Ответственный = УправлениеПользователями.ПолучитьЗначениеПоУмолчанию(глЗначениеПеременной("глТекущийПользователь"), "ОсновнойОтветственный");
	
	склад = Справочники.Склады.НайтиПоКоду(Заказ.ID_WH);
	Если склад.Пустая() Тогда
		склад = СкладОстатков;
	КонецЕсли;
	Накладная.СкладГруппа = склад;
	Если Агент <> Неопределено Тогда
		Накладная.ТорговыйПредставитель = Агент;
	КонецЕсли;
	
	Накладная.Комментарий = Заказ.REMARK;
	Накладная.СтруктурнаяЕдиница = Накладная.Организация.ОсновнойБанковскийСчет;
	
	Накладная.ВидОтгрузки = Перечисления.ВидОтгрузки.Доставка;
	Накладная.ВидОперации = Перечисления.ВидыОперацийЗаказПокупателя.ПродажаКомиссия;
	
	Если Накладная.Организация.ВидыРеализаций.Количество() > 0 Тогда
		вр = Накладная.Организация.ВидыРеализаций.Получить(0);
		Накладная.ВидРеализации = вр.ВидРеализации;
	Иначе
		Накладная.ВидРеализации = Перечисления.ВидРеализации.Розница;

Процедура ПриемЗаявок() Экспорт
	Перем Накладная, агент;
	
	Рез = НайтиФайлы(ПапкаОбмена, "ORD*.DBF");
		
	СчНакл = 0;
	
	Для Каждого файл из Рез Цикл
		имяФайла = файл.Имя;
		разделитель = Найти(ВРег(имяФайла), ".DBF");

		КодСотр = Сред(имяФайла, 4, разделитель-4);
		Агент = НайтиАгента(КодСотр);
		
		можетМенятьЦену = Истина;
		
		временныйФайл = ПапкаОбмена + "\TORD.DBF";
		ПереместитьФайл(файл.ПолноеИмя, временныйФайл);
		
		Заказ = Новый XBase(временныйФайл);
		Заказ.Кодировка = КодировкаXBase.OEM;
		ОбработанныеЗаказы = ФайлОбработанныхДокументов(КодСотр);
		
		ЗаписатьЗаказ = 0;
		ПринятыеЗаявки = ЗагрузитьПринятыеДокументы(КодСотр, Истина);
		Ном = "";
		
		Заказ.Первая();		
		Пока Заказ.ВКонце() = 0 Цикл
			//
			// Проверим признак начала заявки
			//
			Если Заказ.ORDFLAG <> 0 Тогда
				//
				// началась новая заявка, если надо записать заявка, запишем ее.
				//
				Если ЗаписатьЗаказ = 1 Тогда
					ЗаписатьЗаявку(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
					ЗаписатьЗаказ = 0;
				КонецЕсли;
				
				//
				// Проверим, не принимали ли мы уже эту заявку. Может быть передали одну заявку дважды
				//
				пз = ПринятыеЗаявки.НайтиПоЗначению(СокрЛП(Заказ.CREATED));
				Если пз <> Неопределено Тогда
					//
					// пропускаем этот заказ
					//
					ЗаписатьЗаказ = 0;
					Заказ.Следующая();
					СообщитьИнт("Повторная заявка - накладная № " + пз.Представление);
					Продолжить;
				КонецЕсли;
				
				//
				// создаем новую заявку
				//
				Накладная = СоздатьЗаказ(Заказ, Агент, можетМенятьЦену);
				ЗаписатьЗаказ = 1;
				Ном = СокрЛП(Заказ.CREATED);
				СчНакл = СчНакл + 1;
			КонецЕсли;
			
			Если ЗаписатьЗаказ = 0 Тогда
				Заказ.Следующая();
				Продолжить;
			КонецЕсли;
						
			номенклатура = Справочники.Номенклатура.НайтиПоКоду(Заказ.ID_I);

			товар = Накладная.Товары.Добавить();
			товар.Номенклатура = номенклатура.Ссылка;
			товар.Размещение = Накладная.СкладГруппа;
			
			Если ТипЗнч(Накладная) = Тип("ДокументСсылка.ЗаказПокупателя") Тогда
				Накладная.ПриИзмененииНоменклатурыТоваров(товар);
			Иначе
				СтрокаТабличнойЧасти = товар;
				ОбработкаТабличныхЧастей.ЗаполнитьЕдиницуЦенуПродажиТабЧасти(СтрокаТабличнойЧасти, Накладная,
					Накладная.ВалютаДокумента);
			КонецЕсли;
			
			коэф = товар.ЕдиницаИзмерения.Коэффициент;
			Если коэф = 0 Тогда
				коэф = 1;
			КонецЕсли;
			товар.Количество = Заказ.QTY / коэф;
			товар.Коэффициент = коэф;
			Если Заказ.DISCOUNT = 0 Тогда
				товар.Цена = Заказ.COST * коэф;
			Иначе
				товар.Цена = Заказ.COST_ND * коэф;
				товар.ПроцентСкидкиНаценки = Заказ.DISCOUNT;
			КонецЕсли;
			
			ОбработкаТабличныхЧастей.ЗаполнитьСтавкуНДСТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуТабЧасти(товар, Накладная);
			ОбработкаТабличныхЧастей.РассчитатьСуммуНДСТабЧасти(товар, Накладная);
			
			Заказ.Следующая();
		КонецЦикла;
		
		Если ЗаписатьЗаказ = 1 Тогда
			ЗаписатьЗаявку(Накладная, Ном, ПринятыеЗаявки, ОбработанныеЗаказы);
		КонецЕсли;
		
		СохранитьПринятыеДокументы(КодСотр, ПринятыеЗаявки);
		Заказ.ЗакрытьФайл();
		ЗакрытьФайлОбработанныхЗаявок(ОбработанныеЗаказы, КодСотр);
		
		УдалитьФайлы(временныйФайл);
	КонецЦикла;
	
	СообщитьИнт("Всего заявок " + СчНакл);

Функция ЗагрузитьПробегССервера(ДатаС, ДатаПо)
	server = Новый COMОбъект("GRSoft.Server");
	рзд = Найти(АдресСервера,":");
	адрСтр = Лев(АдресСервера, рзд-1);
	порт = Число(Прав(АдресСервера,СтрДлина(АдресСервера) - рзд));

	res = server.Connect(адрСтр, порт);

	Если res = 0 Тогда
		Возврат Неопределено;
	КонецЕсли;

	рез = Новый Соответствие;
	Пока ДатаС < ДатаПо Цикл
		дс = Формат(ДатаС, "ДФ=дд/ММ/гггг");
		ДатаСл = ДатаС + 24 * 3600;
		дпо = Формат(ДатаСл, "ДФ=дд/ММ/гггг");
		ДатаС = ДатаСл;
		парам = "start:" + дс + ",finish:" + дпо + ",gsm:0";
		docs = server.ReportStrParam("daily_agent_distance", парам);
		Если docs = Неопределено Тогда
			Продолжить;
		КонецЕсли;

		Для сч = 0 По docs.Count-1 Цикл
			tdoc = docs.Get(сч);
			Если tdoc.Type = "ReportData" Тогда
				Для сч1 = 0 По tdoc.Count-1 Цикл
					doc = tdoc.Get(сч1);
					ддат = рез.Получить(doc.userid);
					Если ддат = Неопределено Тогда
						рез.Вставить(doc.userid, Новый Соответствие);
						ддат = рез.Получить(doc.userid);
					КонецЕсли;
					
					Если doc.length <> 0 Тогда
						ддат.Вставить(doc.date, doc.length);
					КонецЕсли;
				КонецЦикла;
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
	Возврат рез;

Функция ПреобразоватьЦвет(рИсходныйЦвет,рВидИтоговогоЦвета) Экспорт
Попытка
	тЦветов=ПрочитатьТаблицуЦветов();
    Если ТипЗнч(рИсходныйЦвет)=Тип("Цвет") Тогда
        Если рИсходныйЦвет.Вид=ВидЦвета.ЭлементСтиля Тогда
            стрЦвет=ПолучитьЦветИзСтиляПлатформы(рИсходныйЦвет); // вернёт hex, т.е. 16-ричный
            Если не ПустаяСтрока(стрЦвет) Тогда
                Если рВидИтоговогоЦвета=1 Тогда // нужен web-цвет
					строц=тЦветов.Найти(стрЦвет,"Шестнадцатиричный");
                    Если строц<>Неопределено Тогда
                        Возврат строц.Название;
                    КонецЕсли;
                ИначеЕсли рВидИтоговогоЦвета=2 Тогда // нужен RGB
                    строц=тЦветов.Найти(стрЦвет,"Шестнадцатиричный");
                    Если строц<>Неопределено Тогда
                        Возврат Новый Структура("Красный,Синий,Зеленый",строц.Красный,строц.Синий,строц.Зеленый);
                    КонецЕсли;
                ИначеЕсли рВидИтоговогоЦвета=3 Тогда // нужен 16-ричный
                    Возврат стрЦвет; // это он и есть
                КонецЕсли;
            КонецЕсли;

        ИначеЕсли рИсходныйЦвет.Вид=ВидЦвета.WebЦвет Тогда
            // вытащим из скобок англоязычное написание цвета
            стрЦвет=СокрЛП(Строка(рИсходныйЦвет));
            пози1=Найти(стрЦвет,"(");
            Если пози1<>0 Тогда
                стрЦвет=НРег(СтрЗаменить(Сред(стрЦвет,пози1+1),")",""));
                Если рВидИтоговогоЦвета=1 Тогда // нужен web-цвет
                    Возврат стрЦвет; // это он и есть
                ИначеЕсли рВидИтоговогоЦвета=2 Тогда // нужен RGB
                    строц=тЦветов.Найти(стрЦвет,"Название");
                    Если строц<>Неопределено Тогда
                        Возврат Новый Структура("Красный,Синий,Зеленый",строц.Красный,строц.Синий,строц.Зеленый);
                    КонецЕсли;
                ИначеЕсли рВидИтоговогоЦвета=3 Тогда // нужен 16-ричный
                    строц=тЦветов.Найти(стрЦвет,"Название");
                    Если строц<>Неопределено Тогда
                        Возврат строц.Шестнадцатиричный;
                    КонецЕсли;
                КонецЕсли;
            КонецЕсли;

        ИначеЕсли рИсходныйЦвет.Вид=ВидЦвета.Абсолютный Тогда
            Если рВидИтоговогоЦвета=1 Тогда // нужен web-цвет
                отб=Новый Структура("Красный,Синий,Зеленый",рИсходныйЦвет.Красный,рИсходныйЦвет.Синий,рИсходныйЦвет.Зеленый);
                мстроц=тЦветов.НайтиСтроки(отб);
                Если мстроц.Количество()<>0 Тогда
                    Возврат мстроц[0].Название;
                КонецЕсли;
            ИначеЕсли рВидИтоговогоЦвета=2 Тогда // нужен RGB
                // фактически, он и есть
                Возврат Новый Структура("Красный,Синий,Зеленый",рИсходныйЦвет.Красный,рИсходныйЦвет.Синий,рИсходныйЦвет.Зеленый);
            ИначеЕсли рВидИтоговогоЦвета=3 Тогда // нужен 16-ричный
                отб=Новый Структура("Красный,Синий,Зеленый",рИсходныйЦвет.Красный,рИсходныйЦвет.Синий,рИсходныйЦвет.Зеленый);
                мстроц=тЦветов.НайтиСтроки(отб);
                Если мстроц.Количество()<>0 Тогда
                    Возврат мстроц[0].Шестнадцатиричный;
                КонецЕсли;
            КонецЕсли;

        КонецЕсли; // по вариантам видов цвета как типа Цвет

    ИначеЕсли ТипЗнч(рИсходныйЦвет)=Тип("Строка") Тогда
        // ищем во всех колонках таблицы, т.к. заведомо не знаем, какой это цвет
        строц=тЦветов.Найти(СокрЛП(рИсходныйЦвет));
        Если строц<>Неопределено Тогда
            Если рВидИтоговогоЦвета=1 Тогда // нужен web-цвет
                Возврат строц.Название;
            ИначеЕсли рВидИтоговогоЦвета=2 Тогда // нужен RGB
                Возврат Новый Структура("Красный,Синий,Зеленый",строц.Красный,строц.Синий,строц.Зеленый);
            ИначеЕсли рВидИтоговогоЦвета=3 Тогда // нужен 16-ричный
                Возврат строц.Шестнадцатиричный;
            КонецЕсли;
        КонецЕсли;

    КонецЕсли;

    Возврат Неопределено;
Исключение
    Сообщить("ПреобразоватьЦвет, ошибка: "+ОписаниеОшибки(),СтатусСообщения.Важное);
    Возврат Неопределено;
КонецПопытки;
